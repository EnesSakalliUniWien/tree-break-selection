# ===== protocol.md =====
# # Protocol: diffusion operator against linkage on the cancer_pc cases (frozen 2026-11-12, before the run)
#
# - Question: `tbs_family_diffusion` (operator "symmetric", linkage "average") and
#   `tbs_family_diffusion_markov_ward` (operator "markov", linkage "ward") differ in two parameters. Which of
#   the two makes the balanced splits inside one true group and the accepted extra splits of today's test
#   (`results/analyses/cancer_pc_over_splits_20261112`)?
# - Cells: method `tbs_family_diffusion` with its registered cell and the overrides
#   `diffusion_coordinate_operator` in {"symmetric", "markov"} x `tree_linkage_method` in {"average", "ward"}:
#   4 cells x 4 cases = 16 runs, today's test only (no opt-in).
# - `measure.py` is `cancer_pc_over_splits_20261112/measure.py` with the loop over methods replaced by the loop
#   over cells; same records per internal node. `summarize.py` uses the classification of that folder's
#   deviations.md (pass-through, inside, between) and the same Part 2 (nodes with at least 40 cells, all from
#   one true group).
# - Self-check (asserts): symmetric/average and markov/ward reproduce found clusters and ARI of the two
#   registered methods in the 11 November `results.csv` (this also checks that the method id carries nothing
#   besides the two parameters); split nodes equal found clusters minus 1 plus pass-through nodes.
# - Hypothesis, stated before running: the linkage decides. Both ward cells show balanced splits inside one
#   true group, significant edges there and accepted inside splits; both average cells do not. If the mixed
#   cells do not follow the linkage, the hypothesis is reported as not supported.
# - Limits known before running: one seed per case, 4 synthetic cases, nodes with at least 40 cells.


# ===== deviations.md =====
# # Deviations (written 2026-11-12, during the run)
#
# - The self-check "split nodes equal found clusters minus 1 plus pass-through nodes" stopped the first run at
#   the 15th fit (nohub_connected, markov/average: 14 split nodes, 15 clusters, 1 pass-through node). The
#   identity is not general: a pass-through node adds a split node beyond found clusters minus 1 only when cells
#   below it rejoin another cluster. The check was weakened to
#   found - 1 <= split nodes <= found - 1 + pass-through nodes and all 16 fits were rerun. Nothing else changed.


# ===== measure.py =====
"""protocol.md: every internal node of today's fit on the 4 cancer_pc cases, operator x linkage, with truth counts."""
import json
import sys
from pathlib import Path

import networkx as nx
import numpy as np
import pandas as pd

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parents[2]))

import benchmarks.shared.runners.tbs_runner as tbs_runner  # noqa: E402
from benchmarks.shared.benchmark_runs.runtime import (
    apply_single_thread_runtime_defaults,  # noqa: E402
)
from benchmarks.shared.cases import get_test_cases_by_suite  # noqa: E402
from benchmarks.shared.generators import generate_case_data  # noqa: E402
from benchmarks.shared.pipeline import benchmark_cluster_algorithm  # noqa: E402
from benchmarks.shared.runners.method_registry import METHOD_SPECS  # noqa: E402
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.best_flat_cut_reference import (  # noqa: E402
    best_flat_cut_reference,
    split_rho,
)

METHOD = "tbs_family_diffusion"
CELLS = [(operator, linkage) for operator in ("symmetric", "markov") for linkage in ("average", "ward")]
REGISTERED = {("symmetric", "average"): "tbs_family_diffusion", ("markov", "ward"): "tbs_family_diffusion_markov_ward"}
SIBLING = ["Sibling_Test_Statistic", "Sibling_Raw_Test_Statistic", "Sibling_Reference_Scale", "Sibling_Inflation_Factor",
           "Sibling_Local_Inflation_Factor", "Sibling_Family_Baseline_Inflation_Factor", "Sibling_Degrees_of_Freedom",
           "Sibling_Divergence_P_Value", "Sibling_Divergence_P_Value_Corrected", "Sibling_BH_Different",
           "Sibling_Divergence_Skipped", "Sibling_Test_Method", "Sibling_Projection_Dimension"]
EDGE = ["Child_Parent_Divergence_P_Value_BH", "Child_Parent_Divergence_Significant", "Child_Parent_Divergence_Tested"]
seen = {}
Original = tbs_runner.TreeDecomposition
original_labels = tbs_runner.labels_and_report_from_decomposition


class Recording(Original):
    def decompose_tree(self):
        out = super().decompose_tree()
        seen.update(tree=self.tree, annotations=self.annotations_df, counters=out["traversal_counters"])
        return out


def recording_labels(decomposition, index):
    labels, report = original_labels(decomposition, index)
    seen.update(labels=dict(zip(index, np.asarray(labels).tolist())))
    return labels, report


def plain(value):
    if isinstance(value, (bool, np.bool_)):
        return bool(value)
    if isinstance(value, (int, np.integer)):
        return int(value)
    if isinstance(value, (float, np.floating)):
        return None if not np.isfinite(value) else float(value)
    return None if value is None else str(value)


tbs_runner.TreeDecomposition = Recording
tbs_runner.labels_and_report_from_decomposition = recording_labels
apply_single_thread_runtime_defaults()
earlier = pd.read_csv(HERE.parent / "best_flat_cut_reference_benchmark_20261111" / "results.csv", float_precision="round_trip")
earlier = earlier[~earlier["params"].str.contains("sibling_best_flat_cut_reference")].set_index(["case_id", "method"])
with open(HERE / "nodes.jsonl", "w") as out:
    for case in get_test_cases_by_suite("cancer_pc"):
        data, truth, _, _ = generate_case_data(case)
        matrix = data.to_numpy(dtype=np.float64)
        row_of = {name: row for row, name in enumerate(data.index)}
        for operator, linkage in CELLS:
            method, cell = METHOD, f"{operator}/{linkage}"
            params = {**METHOD_SPECS[METHOD].param_grid[0], "diffusion_coordinate_operator": operator, "tree_linkage_method": linkage}
            seen.clear()
            res = benchmark_cluster_algorithm(
                test_cases=[case], methods=[method], method_params={method: [params]},
                verbose=False, plot_umap=False, include_cover_pages=False, include_validation_page=False,
                allow_method_errors=False)
            found, ari = int(res["found_clusters"].iloc[0]), float(res["ari"].iloc[0])
            if (operator, linkage) in REGISTERED:
                before = earlier.loc[(res["case_id"].iloc[0], REGISTERED[operator, linkage])]
                assert found == int(before["found_clusters"]) and abs(ari - float(before["ari"])) < 1e-12, (case["name"], cell)
            tree, table = seen["tree"], seen["annotations"]
            cluster = np.array([seen["labels"][name] for name in data.index])
            rows_under = {}
            for node in reversed(list(nx.topological_sort(tree))):
                children = list(tree.successors(node))
                rows_under[node] = [row_of[tree.nodes[node]["label"]]] if not children else sum((rows_under[c] for c in children), [])
            splits = 0
            for node in tree.nodes:
                children = list(tree.successors(node))
                if not children:
                    continue
                left, right = (np.array(rows_under[c]) for c in children)
                rows = np.r_[left, right]
                in_left = np.r_[np.ones(len(left), bool), np.zeros(len(right), bool)]
                record = dict(case=case["name"], method=cell, found=found, ari=ari, node=str(node), children=[str(c) for c in children],
                              n=len(rows), n_left=len(left), n_right=len(right),
                              truth_left={int(g): int((truth[left] == g).sum()) for g in np.unique(truth[left])},
                              truth_right={int(g): int((truth[right] == g).sum()) for g in np.unique(truth[right])},
                              split=bool(len(np.unique(cluster[rows])) > 1), is_root=node == tree.root(),
                              **{name: plain(table.loc[node, name]) for name in SIBLING},
                              **{f"{side}_{name}": plain(table.loc[c, name]) for side, c in zip(("left", "right"), children) for name in EDGE})
                splits += record["split"]
                reference = best_flat_cut_reference(matrix[rows], in_left)
                record.update(reference_reason=reference.reason, reference_k=reference.invariant_directions,
                              reference_rho=plain(reference.rho), reference_p=reference.p_value,
                              rho_30_features=plain(split_rho(matrix[rows], in_left)) if len(rows) >= 40 else None)
                out.write(json.dumps(record) + "\n")
            counters = seen["counters"]
            # deviations.md: a pass-through node is a split node of its own only when its cells rejoin a cluster
            assert found - 1 <= splits <= found - 1 + counters["live_pass_through_count"], (splits, found, counters)
            print(case["name"], cell, found, round(ari, 4), "split nodes", splits, counters, flush=True)


# ===== summarize.py =====
"""protocol.md Parts 1 and 2 from nodes.jsonl."""
import json
from pathlib import Path

import numpy as np

HERE = Path(__file__).resolve().parent
nodes = [json.loads(line) for line in open(HERE / "nodes.jsonl")]
TREE = {f"{operator}/{linkage}": f"{operator}/{linkage}" for operator in ("symmetric", "markov") for linkage in ("average", "ward")}


def divided(node):
    groups = []
    for g in set(node["truth_left"]) | set(node["truth_right"]):
        a, b = node["truth_left"].get(g, 0), node["truth_right"].get(g, 0)
        if min(a, b) >= 5 and min(a, b) >= 0.1 * (a + b):
            groups.append(int(g))
    return sorted(groups)


def majority(counts):
    return max(counts, key=counts.get)


def kind_of(node):
    """deviations.md: pass-through, or inside (a divided group, or the same majority group on both sides), or between."""
    if not node["Sibling_BH_Different"]:
        return "pass-through"
    if divided(node) or majority(node["truth_left"]) == majority(node["truth_right"]):
        return "inside"
    return "between"


def med(values, digits=3):
    values = [v for v in values if v is not None]
    return "-" if not values else f"{np.median(values):.{digits}f} (min {min(values):.{digits}f}, max {max(values):.{digits}f}, {len(values)} nodes)"


def num(value):
    return "-" if value is None else f"{value:.3g}"


print("== Part 1: accepted splits ==")
totals = {}
for case in dict.fromkeys(n["case"] for n in nodes):
    for method, tree in TREE.items():
        run = [n for n in nodes if n["case"] == case and n["method"] == method]
        splits = sorted((n for n in run if n["split"]), key=lambda n: -n["n"])
        counts = {k: sum(kind_of(n) == k for n in splits) for k in ("between", "inside", "pass-through")}
        print(f"\n{case} | {tree}: found {run[0]['found']}, ARI {run[0]['ari']:.4f}, split nodes {len(splits)}, {counts}")
        key = totals.setdefault(tree, {})
        for n in splits:
            kind = kind_of(n)
            key[kind] = key.get(kind, 0) + 1
            key[kind + ", reference p > 0.01"] = key.get(kind + ", reference p > 0.01", 0) + (n["reference_p"] > 0.01)
            print(f"  {n['node']:>5} n={n['n']:>3} {n['n_left']:>3}|{n['n_right']:<3} {str(n['truth_left']):<28} | {str(n['truth_right']):<28} "
                  f"{kind:<12} stat={num(n['Sibling_Test_Statistic'])} df={num(n['Sibling_Degrees_of_Freedom'])} "
                  f"c={num(n['Sibling_Inflation_Factor'])} p={num(n['Sibling_Divergence_P_Value'])} "
                  f"p_bh={num(n['Sibling_Divergence_P_Value_Corrected'])} bh={n['Sibling_BH_Different']} | "
                  f"ref {n['reference_reason']} k={n['reference_k']} rho={num(n['reference_rho'])} p={n['reference_p']:.4f} "
                  f"rho30={num(n['rho_30_features'])}")
print("\ntotals over the 4 cases:", json.dumps(totals))

print("\n== Part 2: nodes with >= 40 cells, all from one true group ==")
for method, tree in TREE.items():
    pure = [n for n in nodes if n["method"] == method and n["n"] >= 40 and len(set(n["truth_left"]) | set(n["truth_right"])) == 1]
    tested = [n for n in pure if n["Sibling_Divergence_P_Value"] is not None]
    share = [min(n["n_left"], n["n_right"]) / n["n"] for n in pure]
    print(f"\n{tree}: nodes {len(pure)}")
    print(f"  smaller side share: median {np.median(share):.3f}, quartiles {np.quantile(share, 0.25):.3f} to {np.quantile(share, 0.75):.3f}; "
          f"smaller side under 10 cells: {sum(min(n['n_left'], n['n_right']) < 10 for n in pure)}")
    print(f"  rho in 30 features: {med([n['rho_30_features'] for n in pure])}")
    print(f"  edge gate: an edge of the node was tested in {sum(bool(n['left_Child_Parent_Divergence_Tested']) or bool(n['right_Child_Parent_Divergence_Tested']) for n in pure)}, "
          f"significant for at least one child in {sum(bool(n['left_Child_Parent_Divergence_Significant']) or bool(n['right_Child_Parent_Divergence_Significant']) for n in pure)}, "
          f"for both children in {sum(bool(n['left_Child_Parent_Divergence_Significant']) and bool(n['right_Child_Parent_Divergence_Significant']) for n in pure)}")
    print(f"  sibling test: p-value present in {len(tested)}, Sibling_Divergence_Skipped in {sum(bool(n['Sibling_Divergence_Skipped']) for n in pure)}, "
          f"raw p <= 0.01 in {sum(n['Sibling_Divergence_P_Value'] <= 0.01 for n in tested)}, Sibling_BH_Different in {sum(bool(n['Sibling_BH_Different']) for n in pure)}, "
          f"split in the final clusters {sum(n['split'] for n in pure)}")
    print(f"  tested nodes: statistic per degree of freedom {med([n['Sibling_Test_Statistic'] / n['Sibling_Degrees_of_Freedom'] for n in tested])}; "
          f"inflation factor {med([n['Sibling_Inflation_Factor'] for n in tested])}; degrees of freedom {med([n['Sibling_Degrees_of_Freedom'] for n in tested], 0)}")
    print(f"  reference p <= 0.01: {sum(n['reference_p'] <= 0.01 for n in pure)} of {len(pure)} "
          f"(reasons {json.dumps({r: sum(n['reference_reason'] == r for n in pure) for r in sorted({n['reference_reason'] for n in pure})})})")
    for lo, hi in ((0, 0.1), (0.1, 0.3), (0.3, 0.5001)):
        part = [n for n in pure if lo <= min(n["n_left"], n["n_right"]) / n["n"] < hi]
        print(f"  smaller side share {lo:.1f} to {min(hi, 0.5):.1f}: {len(part)} nodes, edge significant for at least one child in "
              f"{sum(bool(n['left_Child_Parent_Divergence_Significant']) or bool(n['right_Child_Parent_Divergence_Significant']) for n in part)}, "
              f"sibling p present in {sum(n['Sibling_Divergence_P_Value'] is not None for n in part)}, BH different in {sum(bool(n['Sibling_BH_Different']) for n in part)}, "
              f"rho30 {med([n['rho_30_features'] for n in part])}")

print("\n== Part 2b (added after the first output, deviations.md): the largest node of each true group with all cells from that group ==")
for case in dict.fromkeys(n["case"] for n in nodes):
    for method, tree in TREE.items():
        run = [n for n in nodes if n["case"] == case and n["method"] == method]
        for g in range(6):
            pure = [n for n in run if set(n["truth_left"]) | set(n["truth_right"]) == {str(g)}]
            if not pure:
                continue
            n = max(pure, key=lambda n: n["n"])
            if n["n"] < 40:
                continue
            print(f"  {case} | {tree} | group {g}: {n['node']} n={n['n']} {n['n_left']}|{n['n_right']} rho30={num(n['rho_30_features'])} "
                  f"edges tested {bool(n['left_Child_Parent_Divergence_Tested'])},{bool(n['right_Child_Parent_Divergence_Tested'])} "
                  f"significant {bool(n['left_Child_Parent_Divergence_Significant'])},{bool(n['right_Child_Parent_Divergence_Significant'])} "
                  f"edge p_bh {num(n['left_Child_Parent_Divergence_P_Value_BH'])},{num(n['right_Child_Parent_Divergence_P_Value_BH'])} "
                  f"sibling p={num(n['Sibling_Divergence_P_Value'])} skipped={n['Sibling_Divergence_Skipped']} bh={n['Sibling_BH_Different']} split={n['split']}")

print("\n== inflation factor per run (all tested nodes) ==")
for case in dict.fromkeys(n["case"] for n in nodes):
    for method, tree in TREE.items():
        run = [n for n in nodes if n["case"] == case and n["method"] == method]
        tested = [n for n in run if n["Sibling_Divergence_P_Value"] is not None]
        edge_open = sum(bool(n["left_Child_Parent_Divergence_Significant"]) or bool(n["right_Child_Parent_Divergence_Significant"]) for n in run)
        print(f"{case} | {tree}: sibling p present in {len(tested)} of {len(run)} nodes, edge significant for at least one child in {edge_open}, "
              f"inflation factor {med([n['Sibling_Inflation_Factor'] for n in tested])}, BH different {sum(bool(n['Sibling_BH_Different']) for n in run)}")

print("\n== Part 3 (added after the first output): per cell over the 4 cases ==")
for cell in TREE:
    runs = {n["case"]: (n["found"], n["ari"]) for n in nodes if n["method"] == cell}
    accepted_inside = [n for n in nodes if n["method"] == cell and n["split"] and kind_of(n) == "inside"]
    one_group = sum(len(set(n["truth_left"]) | set(n["truth_right"])) == 1 for n in accepted_inside)
    by_case = {case.replace("cancer_pc_", ""): sum(n["case"] == case for n in accepted_inside) for case in runs}
    print(f"{cell}: found {[f for f, _ in runs.values()]}, mean ARI {np.mean([a for _, a in runs.values()]):.4f}, "
          f"accepted inside splits {len(accepted_inside)} (all cells from one true group: {one_group}, cells from several groups: "
          f"{len(accepted_inside) - one_group}), by case {by_case}")
