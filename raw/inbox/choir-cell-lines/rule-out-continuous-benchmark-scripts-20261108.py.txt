# ===== protocol.md =====
# Protocol: split-count rule-out on the continuous benchmark suite (frozen 2026-11-08)

User choice: one comparison run of the continuous suite, benchmark defaults unchanged.

- Cases: `get_test_cases_by_suite("continuous")`, all 9 (45-200 cells, 24-20,000 features, 1-6 true groups).
- Methods: `tbs_family_diffusion` and `tbs_family_diffusion_markov_ward`, each with its registered parameter
  cell and the same cell plus `sibling_split_count_rule_out=True`; run through the shared
  `benchmark_cluster_algorithm` (default alphas), plots off, method errors recorded rather than raised.
- Endpoints (descriptive): ARI, NMI and found clusters against the true number per case, method and cell;
  for rule-out cells, the reasons recorded in `Sibling_Test_Method` where available.
- Expected limit, stated before running: a subtree of n <= p + 1 cells has no invariant directions, so no
  split inside it can be accepted. Every case has more features than cells per true group, so at most the
  first splits can pass; in the 20,000-feature case (40 cells) none can.

# ===== run.py =====
"""protocol.md: continuous benchmark suite, family-diffusion methods, today vs split-count rule-out."""
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parents[2]))

from benchmarks.shared.benchmark_runs.runtime import apply_single_thread_runtime_defaults  # noqa: E402
from benchmarks.shared.cases import get_test_cases_by_suite  # noqa: E402
from benchmarks.shared.pipeline import benchmark_cluster_algorithm  # noqa: E402
from benchmarks.shared.runners.method_registry import METHOD_SPECS  # noqa: E402

METHODS = ["tbs_family_diffusion", "tbs_family_diffusion_markov_ward"]

apply_single_thread_runtime_defaults()
params = {m: [dict(METHOD_SPECS[m].param_grid[0]), {**METHOD_SPECS[m].param_grid[0], "sibling_split_count_rule_out": True}]
          for m in METHODS}
results = benchmark_cluster_algorithm(test_cases=get_test_cases_by_suite("continuous"), methods=METHODS,
                                      method_params=params, verbose=False, plot_umap=False, include_cover_pages=False,
                                      include_validation_page=False, allow_method_errors=True)
results.to_csv(HERE / "results.csv", index=False)
print(results.columns.tolist())
