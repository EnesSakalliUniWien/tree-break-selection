# ===== protocol.md =====
# # Protocol: a fit that splits each subtree at its best flat cut (frozen 2026-11-11, before the run)
#
# Trial, descriptive; no production change. It replaces the tree (inherited geometry) by the rebuilt geometry at
# every node, so it is a different procedure from Tree-Break Selection: no diffusion distances, no linkage tree,
# no edge gate, no pass-through, no BH step.
#
# ## Procedure
#
# Data: 48,879 CHOIR cells, 30 PCs (`choir_cell_lines_tbs_lbfgsb_20261106/pca.npy`). Start with all cells. For a
# set of cells:
#
# 1. k = number of `invariant_directions` of these cells. Stop if k = 0 or fewer than 15 cells.
# 2. Find the best flat cut in those directions with the table's search (grid, top 50 re-sorts, single-cell
#    moves); its `rho` is the statistic.
# 3. p-value under one Gaussian population at (n, k), n rounded down to the table grid (largest value 800):
#    k <= 10 from the stored table (1,999 null datasets); k > 10 from 399 null datasets drawn at run time and
#    searched with the same search (seed [20261111, 7, n, k, b]). Because the whitened Gaussian sample has the
#    same distribution for every covariance, the searched value has the same null distribution whether or not
#    the search finds the true best cut; so the run-time null is a valid reference for the searched value at
#    fixed (n, k), also above k = 10.
# 4. If p <= 0.01, split at the cut and repeat on both sides; otherwise the set is one cluster.
#
# No further rule: no smallest side, no cap on depth. A guard stops the run after 6,000 tested sets and reports
# it.
#
# ## Endpoints
#
# Found clusters, ARI and NMI against the 190 cell lines, lines recovered at 90/90, ARI against published
# CHOIR_1, clusters at least 90% pure, clusters under 20 cells, largest cluster, lines split only / merged only /
# both; per tested set: n, k, `rho`, p, smaller side, accepted. Comparison rows from the earlier runs: today,
# rule-out, ceiling 3, reference (Markov/Ward and default tree).
#
# ## Expected before running
#
# - `split-vs-best-flat-cut-20261111.md`: with two or more lines in a subtree the best flat cut passed in 192 of
#   520 subtrees and mostly respected the lines, so fewer lines should stay merged than under the reference fit
#   (91 merged only on Markov/Ward).
# - Inside single lines the best flat cut passed in 6.7% and 24.6% of subtrees, so lines will also be split:
#   more than 190 clusters and a number of lines split only.
# - Some best flat cuts isolate fewer than 5 cells; clusters under 20 cells are expected.
# - Whether ARI exceeds ceiling 3 on Markov/Ward (0.833) is not predicted.
#
# ## Limits known before running
#
# - Per-test level 0.01 without a family-wise or FDR step, so the number of false splits over the whole fit is
#   not controlled. The lowest possible p-values are 0.0005 (k <= 10) and 0.0025 (k > 10).
# - k is chosen from the same cells; the sides of an accepted cut are not Gaussian samples even when the groups
#   are (they are cut along a flat boundary), which the reference does not cover.
# - One dataset; cell lines are the only truth.


# ===== measure_lib.py (from split_vs_best_flat_cut_20261111) =====
"""Best flat cut of a subtree with its partition, and the flat cut nearest to a given split."""
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parents[3] / "scripts"))
from build_best_flat_cut_reference import _along_one, directions, resort, swap  # noqa: E402

from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.best_flat_cut_reference import (  # noqa: E402
    to_rho,
    whiten,
)


def r2_of(y, mask):
    s = int(mask.sum())
    return float((y[mask].sum(0) ** 2).sum() / (s * (len(y) - s)))


def best_flat_cut(coordinates, top=50):
    """The table's search (grid, top re-sorts, single-point moves), returning rho and the partition."""
    y = whiten(coordinates)
    n, k = y.shape
    dirs = directions(k)
    s = np.arange(1, n)[:, None]
    chunk = max(1, min(2000, int(2e7 // (n * k))))
    best = np.empty(len(dirs))
    for a in range(0, len(dirs), chunk):
        g = dirs[a : a + chunk]
        order = np.argsort(y @ g.T, axis=0)
        prefix = np.cumsum(y[order], axis=0)[:-1]
        best[a : a + chunk] = ((prefix**2).sum(-1) / (s * (n - s))).max(0)
    out_r2, out_mask = -1.0, None
    for j in np.argsort(-best, kind="stable")[:top]:
        r2, mask = _along_one(y, dirs[j])
        r2, mask = resort(y, mask, r2)
        r2, mask = swap(y, mask, r2)
        if r2 > out_r2:
            out_r2, out_mask = float(r2), mask
    return float(to_rho(out_r2, n)), out_mask


def nearest_flat_cut(coordinates, in_left):
    """Start at the given split, re-sort along its mean difference and move single cells until stable."""
    y = whiten(coordinates)
    start = r2_of(y, in_left)
    r2, mask = resort(y, in_left, start)
    r2, mask = swap(y, mask, r2)
    return float(to_rho(start, len(y))), float(to_rho(r2, len(y))), mask


# ===== divisive.py =====
"""protocol.md: split each set of cells at its best flat cut while the Gaussian reference rejects one population."""
import json
import sys
import time
from multiprocessing import Pool
from pathlib import Path

import numpy as np
import pandas as pd

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parents[2]))
sys.path.insert(0, str(HERE.parent / "split_vs_best_flat_cut_20261111"))

from measure_lib import best_flat_cut  # noqa: E402

from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.best_flat_cut_reference import (  # noqa: E402
    MAX_DIRECTIONS,
    MIN_CELLS,
    load_table,
    reference_p_value,
)
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import invariant_directions  # noqa: E402

ALPHA, NULL_DATASETS, MAX_SETS, SEED = 0.01, 399, 6000, 20261111
SOURCE = HERE.parent / "choir_cell_lines_tbs_lbfgsb_20261106"
STATE = {}


def init(path=None):
    STATE["x"] = np.load(path or SOURCE / "pca.npy")


def node_job(args):
    node, idx = args
    start = time.perf_counter()
    points = STATE["x"][idx]
    d = invariant_directions(points) if len(idx) >= MIN_CELLS else np.zeros((0, points.shape[1]))
    if len(d) == 0:
        return node, 0, None, None, time.perf_counter() - start
    rho, mask = best_flat_cut(points @ d.T)
    return node, len(d), rho, mask, time.perf_counter() - start


def null_job(args):
    n, k, b = args
    return best_flat_cut(np.random.default_rng([SEED, 7, n, k, b]).standard_normal((n, k)))[0]


def grid_n(n):
    grid = load_table()["n_grid"]
    return int(grid[np.searchsorted(grid, n, side="right") - 1])


def run(x_path, out_prefix, pool):
    n_cells = len(np.load(x_path, mmap_mode="r"))
    nulls, nodes, labels = {}, [], np.full(n_cells, -1)
    frontier, next_id, tested = [(0, np.arange(n_cells, dtype=np.int32))], 1, 0
    parent_of = {0: None}
    while frontier:
        frontier.sort(key=lambda a: -len(a[1]))
        cells = dict(frontier)
        results = list(pool.imap_unordered(node_job, frontier))
        need = sorted({(grid_n(len(cells[node])), k) for node, k, rho, _, _ in results if k > MAX_DIRECTIONS} - set(nulls))
        for n, k in need:
            nulls[(n, k)] = np.sort(pool.map(null_job, [(n, k, b) for b in range(NULL_DATASETS)]))
            print("null", n, k, "99% point", round(float(np.quantile(nulls[(n, k)], 0.99)), 3), flush=True)
        frontier = []
        for node, k, rho, mask, sec in sorted(results, key=lambda r: r[0]):
            idx = cells[node]
            row = {"node": node, "parent": parent_of[node], "n": len(idx), "k": k, "rho": rho, "sec": round(sec, 2)}
            split = False
            if k > 0:
                tested += 1
                if k <= MAX_DIRECTIONS:
                    p = reference_p_value(rho, len(idx), k)[0]
                else:
                    ref = nulls[(grid_n(len(idx)), k)]
                    p = (1 + len(ref) - np.searchsorted(ref, rho, side="left")) / (len(ref) + 1)
                row.update(p=float(p), small=int(min(mask.sum(), (~mask).sum())))
                split = p <= ALPHA and tested <= MAX_SETS
            row["split"] = bool(split)
            nodes.append(row)
            if split:
                for side in (idx[mask], idx[~mask]):
                    parent_of[next_id] = node
                    frontier.append((next_id, side))
                    next_id += 1
            else:
                labels[idx] = node
        print("round done: sets so far", len(nodes), "tested", tested, "open", len(frontier), flush=True)
    assert (labels >= 0).all()
    pd.DataFrame(nodes).to_json(f"{out_prefix}_nodes.jsonl", orient="records", lines=True)
    return pd.factorize(labels)[0], nodes, tested


if __name__ == "__main__":
    with Pool(10, initializer=init) as pool:
        start = time.perf_counter()
        labels, nodes, tested = run(SOURCE / "pca.npy", str(HERE / "choir"), pool)
    metadata = pd.read_csv(SOURCE / "cell_metadata.csv", index_col=0, dtype=str)
    pd.DataFrame({"cell": metadata.index, "Cell_line": metadata.Cell_line.to_numpy(), "cluster": labels}).to_csv(
        HERE / "assignments.csv.gz", index=False)
    print(json.dumps({"clusters": int(labels.max() + 1), "sets": len(nodes), "tested": tested,
                      "guard_hit": tested > MAX_SETS, "sec": round(time.perf_counter() - start)}), flush=True)
    print("DONE", flush=True)


# ===== selfcheck.py =====
"""Self-check of divisive.py on simulated data: three separated Gaussian groups, and one Gaussian population."""
import sys
import tempfile
from multiprocessing import Pool
from pathlib import Path

import numpy as np
from sklearn.metrics import adjusted_rand_score

sys.path.insert(0, str(Path(__file__).resolve().parent))
import divisive  # noqa: E402

if __name__ == "__main__":
    rng = np.random.default_rng(5)
    truth = np.repeat([0, 1, 2], [300, 150, 150])
    centres = np.zeros((3, 30))
    centres[1, 0], centres[2, 1] = 8, 8
    cases = {"three groups": (centres[truth] + rng.standard_normal((600, 30)), truth),
             "one population": (rng.standard_normal((600, 30)), np.zeros(600, int))}
    for name, (x, y) in cases.items():
        with tempfile.TemporaryDirectory() as tmp:
            path = Path(tmp) / "x.npy"
            np.save(path, x)
            with Pool(4, initializer=divisive.init, initargs=(path,)) as pool:
                labels, nodes, tested = divisive.run(path, str(Path(tmp) / "check"), pool)
        print(name, "clusters", int(labels.max() + 1), "ARI", round(adjusted_rand_score(y, labels), 3) if name == "three groups" else "-",
              "tested", tested, [(r["n"], r["k"], None if r["rho"] is None else round(r["rho"], 2), r.get("p")) for r in nodes][:6], flush=True)
