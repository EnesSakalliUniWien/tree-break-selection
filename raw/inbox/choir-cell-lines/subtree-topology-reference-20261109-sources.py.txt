"""Locked fixed-leaf Ward bootstrap experiment."""

import hashlib
import json
import math
import sys
from collections import Counter
from pathlib import Path

import numpy as np
from scipy.cluster.hierarchy import linkage
from scipy.spatial.distance import pdist

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[3]
sys.path.insert(0, str(ROOT))

from benchmarks.diagnostics.calibration.selected.hierarchy import subtree_topology_geometry as geo

CASES = (
    ("planted", [(-7, 0), (-7, 1), (1, -1), (1, 1), (5, -1), (5, 2)], ((0, 1), ((2, 3), (4, 5)))),
    ("rectangle", [(-2, -1), (-2, 1), (2, -1), (2, 1)], ((0, 3), (1, 2))),
    ("square", [(-1, -1), (-1, 1), (1, -1), (1, 1)], None),
)
SCHEMES = ("feature", "noise_008", "noise_080", "row_permutation")


def ward(values, labels):
    nodes = list(map(int, labels))
    for a, b in linkage(values, method="ward")[:, :2].astype(int):
        nodes.append((nodes[a], nodes[b]))
    return nodes[-1]


def key(tree):
    return tuple(sorted(tuple(sorted(clade)) for clade in geo.tree_clades(tree)))


def rates(trees, nominal, target, ties):
    representatives = {key(t): t for t in trees}
    count = Counter(map(key, trees))
    splits = Counter(
        (cut.left, cut.right)
        for t in trees
        for cut in geo.tree_edge_splits(t)
        if min(len(cut.left), len(cut.right)) > 1
    )
    n = len(trees)
    result = dict(
        attempts=n,
        distinct_topologies=len(count),
        entropy_bits=-sum(c / n * math.log2(c / n) for c in count.values()),
        topologies=[dict(clades=k, count=c, frequency=c / n) for k, c in sorted(count.items())],
        split_frequencies=[
            dict(left=a, right=b, count=c, frequency=c / n) for (a, b), c in sorted(splits.items())
        ],
        rf_to_nominal=dict(
            sorted(Counter(geo.rooted_robinson_foulds(t, nominal) for t in trees).items())
        ),
        outside_exact_tie_family=sum(c for k, c in count.items() if k not in ties),
        min_rf_to_tie_family=dict(
            sorted(
                Counter(
                    min(
                        geo.rooted_robinson_foulds(representatives[k], tie) for tie in ties.values()
                    )
                    for k in map(key, trees)
                ).items()
            )
        ),
    )
    if target is not None:
        result["rf_to_comparator"] = dict(
            sorted(Counter(geo.rooted_robinson_foulds(t, target) for t in trees).items())
        )
        result["comparator_recovery"] = count[key(target)]
    assert sum(c for c in count.values()) == n
    return result


def main():
    result = {}
    for case_idx, (name, points, target) in enumerate(CASES):
        planar = np.asarray(points, dtype=float)
        a, b = planar.T
        x = np.column_stack([a, b, a, b, a + b, a - b])
        np.testing.assert_array_equal(pdist(x, "sqeuclidean"), 4 * pdist(planar, "sqeuclidean"))
        ids = np.arange(len(x))
        nominal = ward(x, ids)
        ties = {key(t): t for t in geo.ward_topology_family(x, rtol=0)}
        assert set(ties) == {key(t) for t in geo.ward_topology_family(x)}
        if name == "square":
            assert set(ties) == {key(((0, 1), (2, 3))), key(((0, 2), (1, 3)))}
        assert key(nominal) in ties
        assert len(ties) == (2 if name == "square" else 1)
        if name == "planted":
            assert geo.rooted_robinson_foulds(nominal, target) == 0
        result[name] = dict(
            points=x.tolist(),
            nominal_clades=key(nominal),
            exact_tied_family=sorted(ties),
            comparator=None if target is None else key(target),
            comparator_is_truth=name == "planted",
            schemes={},
        )
        for i, scheme in enumerate(SCHEMES):
            trees, raw_order = [], []
            for j in range(1000):
                rng = np.random.default_rng([5811260911, case_idx, i, j])
                if scheme == "feature":
                    values = x[:, rng.integers(x.shape[1], size=x.shape[1])]
                elif scheme.startswith("noise"):
                    sigma = 0.08 if scheme == "noise_008" else 0.8
                    values = x + rng.normal(scale=sigma, size=x.shape)
                else:
                    order = rng.permutation(ids)
                    values = x[order][np.argsort(order)]
                    np.testing.assert_array_equal(pdist(values), pdist(x))
                    raw_order.append(ward(x[order], order))
                trees.append(ward(values, ids))
            outcome = rates(trees, nominal, target, ties)
            if raw_order:
                assert outcome["rf_to_nominal"] == {0: 1000}
                outcome["unrestored_order_ward_with_restored_labels"] = rates(
                    raw_order, nominal, target, ties
                )
            result[name]["schemes"][scheme] = outcome
    paths = (Path(__file__), HERE / "protocol.md", Path(geo.__file__))
    evidence = dict(
        cases=result,
        sha256={
            str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest() for p in paths
        },
    )
    with (HERE / "evidence.json").open("x") as stream:
        json.dump(evidence, stream, indent=2, allow_nan=False)
        stream.write("\n")


if __name__ == "__main__":
    main()

"""Exact max-over-labels score on a fixed or median-selected eight-row subtree."""
import json
from pathlib import Path
from time import perf_counter

import numpy as np
from threadpoolctl import threadpool_limits

from run import fixed_subtree_contrast_test, subtree_gaussian_chain_test, summary

HERE = Path(__file__).resolve().parent
LEFT = ((np.arange(127)[:, None] >> np.arange(8)) & 1) == 0
# Fix the last label to the left side, remove the empty complement.
LEFT = LEFT[1:]
LEFT = np.r_[LEFT, [[False] * 7 + [True]]]
CONTRASTS = LEFT - LEFT.mean(axis=1, keepdims=True)
CONTRASTS /= np.linalg.norm(CONTRASTS, axis=1, keepdims=True)
assert len(LEFT) == len(np.unique(LEFT, axis=0)) == 127
assert LEFT[:, -1].all() and np.all(LEFT.sum(axis=1) < 8)


def record(panel_idx, replicate):
    ds, cs = np.random.SeedSequence([81620261109, panel_idx, replicate]).spawn(2)
    seed = int(cs.generate_state(1, dtype=np.uint64)[0])
    factor = np.linalg.cholesky([[1, 1.2], [1.2, 4]])
    x = np.random.default_rng(ds).standard_normal((16, 2)) @ factor.T
    order = np.argsort(x[:, 0])
    if panel_idx:
        rows, outside = np.sort(order[:8]), order[8:]
        boundary = x[outside, 0].min()
    else:
        rows = np.arange(8)
        boundary = None

    def scores(values):
        y = values[rows] - values[rows].mean(axis=0)
        q, _ = np.linalg.qr(y, mode="reduced")
        return ((CONTRASTS @ q)**2).sum(axis=1)

    def score(values):
        return float(scores(values).max())

    def event(values):
        if values[rows, 0].max() == boundary:
            raise FloatingPointError("median membership tied")
        return bool(values[rows, 0].max() < boundary)

    max_index = int(np.argmax(scores(x)))
    fixed = fixed_subtree_contrast_test(
        x, subtree_indices=rows, left_indices=rows[LEFT[max_index]], all_labels=True
    )
    assert fixed["status"] == "complete" and fixed["number_of_labels"] == 127
    np.testing.assert_allclose(fixed["statistic"], score(x), rtol=1e-12)
    result = dict(panel=("fixed_max" if panel_idx == 0 else "selected_descendant"),
                  replicate=replicate, seed=seed, score=score(x),
                  naive_analytic=fixed["unadjusted_p_value"], fixed_beta=fixed)
    for name, test_event in [("naive_chain", None), *([("selected_chain", event)] if panel_idx else [])]:
        start = perf_counter()
        chain = subtree_gaussian_chain_test(
            x, subtree_indices=rows, score=score, selection_event=test_event, steps=199, seed=seed
        )
        chain["seconds"] = perf_counter() - start
        if chain["status"] == "complete":
            assert chain["p_value"] == np.mean(np.asarray(chain["chain_scores"]) >= score(x))
        result[name] = chain
    return result


def main():
    out = {}
    with threadpool_limits(limits=1), (HERE / "all_labels.jsonl").open("x") as stream:
        for panel_idx, count in enumerate((320, 160)):
            rows = []
            for replicate in range(count):
                row = record(panel_idx, replicate)
                stream.write(json.dumps(row, allow_nan=False) + "\n")
                rows.append(row)
            stream.flush()
            out[rows[0]["panel"]] = summary(rows)
    (HERE / "all_labels_summary.json").write_text(json.dumps(out, indent=2) + "\n")


if __name__ == "__main__":
    main()

"""Fresh independent fixed and selected Gaussian subtree checks."""
import json
import sys
from pathlib import Path
from time import perf_counter

import numpy as np
from scipy.stats import beta, binomtest
from threadpoolctl import threadpool_limits

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[3]
sys.path.insert(0, str(ROOT))

from benchmarks.diagnostics.calibration.selected.hierarchy.subtree_gaussian_reference import (
    fixed_subtree_contrast_test,
    subtree_gaussian_chain_test,
)

PANELS = (("fixed_null", 3, 1200), ("selected_null", 3, 1200),
          ("full_span_null", 12, 300), ("selected_signal", 3, 200))
ROWS = np.arange(32).reshape(4, 8)
CONTRAST = np.repeat([1.0, -1.0], 4) / np.sqrt(8)


def statistic(values, rows):
    feature = values[rows, 0]
    z = feature - feature.mean()
    return float((CONTRAST @ z)**2 / (z @ z))


def record(panel, panel_idx, p, replicate, factor):
    data_seed, chain_seed = np.random.SeedSequence([2026110917, panel_idx, replicate]).spawn(2)
    seed = int(chain_seed.generate_state(1, dtype=np.uint64)[0])
    x = np.random.default_rng(data_seed).standard_normal((32, p)) @ factor
    if panel == "selected_signal":
        x[4:8, 0] += 2.5 * np.linalg.norm(factor[:, 0])
    scores = [statistic(x, rows) for rows in ROWS]
    selected = panel.startswith("selected")
    winner = int(np.argmax(scores)) if selected else 0
    rows = ROWS[winner]
    maximum_other = max(value for i, value in enumerate(scores) if i != winner)
    ties_win = winner < next(i for i, v in enumerate(scores) if v == maximum_other and i != winner)

    def score(values):
        return statistic(values, rows)

    def event(values):
        q = score(values)
        return q >= maximum_other if ties_win else q > maximum_other

    tail = float(beta.sf(scores[winner], 0.5, 3))
    result = dict(panel=panel, replicate=replicate, seed=seed, winner=winner,
                  score=scores[winner], largest_other=maximum_other, naive_analytic=tail)
    if selected:
        result["selected_analytic"] = float(tail / beta.sf(maximum_other, 0.5, 3))
    for name, test_event in [("naive_chain", None), *([("selected_chain", event)] if selected else [])]:
        start = perf_counter()
        result[name] = subtree_gaussian_chain_test(
            x, subtree_indices=rows, score=score, selection_event=test_event, steps=119, seed=seed
        )
        result[name]["seconds"] = perf_counter() - start
        chain = result[name]
        if chain["status"] == "complete":
            assert chain["p_value"] == np.mean(np.asarray(chain["chain_scores"]) >= scores[winner])
            assert chain["chain_scores"][chain["anchor_index"]] == scores[winner]
    if not selected:
        fixed = fixed_subtree_contrast_test(x, subtree_indices=rows, left_indices=rows[:4])
        result["fixed_beta"] = fixed
        if panel == "full_span_null":
            assert fixed["decision"] == "unresolved" and fixed["rank"] == 7
            reduced = fixed_subtree_contrast_test(x, subtree_indices=rows, left_indices=rows[:4], rank=2)
            assert reduced["status"] == "complete"
            result["rank2_beta"] = reduced
    return result


def summary(rows):
    result = []
    for endpoint in ("naive_analytic", "selected_analytic", "naive_chain", "selected_chain",
                     "fixed_beta", "rank2_beta"):
        if endpoint not in rows[0]:
            continue
        endpoint_rows = [r[endpoint] for r in rows]
        chain = endpoint.endswith("chain")
        p = [r["p_value"] for r in endpoint_rows] if isinstance(endpoint_rows[0], dict) else endpoint_rows
        for alpha in (0.05, 0.01):
            rejected = [value is not None and value <= alpha and (
                not chain or endpoint_rows[i]["sampling_status"] == "moved"
            ) for i, value in enumerate(p)]
            ci = binomtest(sum(rejected), len(rows)).proportion_ci()
            result.append(dict(
                endpoint=endpoint, alpha=alpha, rejected=sum(rejected), datasets=len(rows),
                lower95=ci.low, upper95=ci.high,
                unresolved=sum(r["decision"] == "unresolved" for r in endpoint_rows)
                if isinstance(endpoint_rows[0], dict) else 0,
                median_moves=float(np.median([r["accepted_moves"] for r in endpoint_rows])) if chain else None,
                median_seconds=float(np.median([r["seconds"] for r in endpoint_rows])) if chain else None,
                stalled=sum(r["sampling_status"] == "stalled" for r in endpoint_rows) if chain else None,
                signal_selected=sum(r["winner"] == 0 for r in rows) if rows[0]["panel"] == "selected_signal" else None,
                signal_selected_rejected=sum(hit and r["winner"] == 0 for hit, r in zip(rejected, rows))
                if rows[0]["panel"] == "selected_signal" else None,
            ))
    return result


def main():
    out = {}
    with threadpool_limits(limits=1), (HERE / "results.jsonl").open("x") as stream:
        for i, (panel, p, count) in enumerate(PANELS):
            q, _ = np.linalg.qr(np.random.default_rng(731190).normal(size=(p, p)))
            factor = np.sqrt(np.geomspace(1, 16, p))[:, None] * q.T
            rows = []
            for rep in range(count):
                row = record(panel, i, p, rep, factor)
                stream.write(json.dumps(row, allow_nan=False) + "\n")
                rows.append(row)
            stream.flush()
            assert len(rows) == count
            out[panel] = summary(rows)
            print(panel, count, flush=True)
    (HERE / "summary.json").write_text(json.dumps(out, indent=2, allow_nan=False) + "\n")


if __name__ == "__main__":
    if sys.argv[1:] == ["summarize"]:
        panels = {name: [] for name, _, _ in PANELS}
        with (HERE / "results.jsonl").open() as stream:
            for row in map(json.loads, stream):
                panels[row["panel"]].append(row)
        for name, _, count in PANELS:
            assert sorted(r["replicate"] for r in panels[name]) == list(range(count))
        out = {name: summary(rows) for name, rows in panels.items()}
        (HERE / "summary.json").write_text(json.dumps(out, indent=2, allow_nan=False) + "\n")
    else:
        main()

"""Gaussian common-mean references for a specified subtree."""

import math
from numbers import Integral, Real

import numpy as np
from scipy.linalg import helmert
from scipy.stats import beta


def _rows(indices, n, *, name, minimum):
    try:
        rows = tuple(indices)
    except TypeError as error:
        raise ValueError(f"{name} must be a sequence of row indices") from error
    if (
        len(rows) < minimum
        or any(isinstance(i, bool) or not isinstance(i, Integral) or not 0 <= i < n for i in rows)
        or len(set(rows)) != len(rows)
    ):
        raise ValueError(f"{name} must contain at least {minimum} distinct valid row indices")
    return np.asarray(rows, dtype=np.intp)


def _inputs(values, subtree_indices, alpha):
    if isinstance(alpha, bool) or not isinstance(alpha, Real) or not 0 < alpha < 1:
        raise ValueError("alpha must lie strictly between zero and one")
    x = np.asarray(values, dtype=float)
    if x.ndim != 2 or x.shape[1] < 1 or not np.isfinite(x).all():
        raise ValueError("values must be a finite matrix with at least one feature")
    return x, _rows(subtree_indices, len(x), name="subtree_indices", minimum=2)


def _rotated(state, rng):
    proposed = state.copy()
    if len(state) == 1:
        sign = 1 if rng.integers(2) else -1
        proposed[0] *= sign
        return proposed, (0, 0, 0.0, sign)
    i, j = sorted(map(int, rng.choice(len(state), size=2, replace=False)))
    angle = float(rng.uniform(-np.pi, np.pi))
    sign = 1 if rng.integers(2) else -1
    c, s = math.cos(angle), math.sin(angle)
    proposed[i] = c * state[i] - s * state[j]
    proposed[j] = sign * (s * state[i] + c * state[j])
    return proposed, (i, j, angle, sign)


def subtree_gaussian_chain_test(
    values, *, subtree_indices, score, selection_event=None, steps=999, seed=0, alpha=0.05
):
    """Rank a deterministic full-matrix score under the common-mean Gaussian null.

    Condition on the subtree mean, centered feature Gram matrix, outside rows
    and, if supplied, the caller's selection event. Subtree rows must be iid
    Gaussian with one common mean and covariance and independent of outside
    rows. Unknown covariance is allowed. The event must describe how the tested
    membership and any other conditioned selection were obtained. Omitting it
    requires prespecified membership. Callbacks receive read-only matrices.
    A failed evaluation invalidates the rank. No descendant or traversal event
    is inferred here, and native floating-point event membership is uncertified.

    Chain moves store accepted transforms as (i, j, angle, sign), relative to
    the preceding state when moving outward from the observed anchor. An empty
    entry keeps the preceding state. Equal indices denote the one-axis case.
    """
    x, rows = _inputs(values, subtree_indices, alpha)
    for name, value, minimum in (("steps", steps, 1), ("seed", seed, 0)):
        if isinstance(value, bool) or not isinstance(value, Integral) or value < minimum:
            raise ValueError(f"{name} must be an integer >= {minimum}")
    if not callable(score) or (selection_event is not None and not callable(selection_event)):
        raise ValueError("score and any selection_event must be callable")

    def in_event(matrix):
        if selection_event is None:
            return True
        accepted = selection_event(matrix)
        if not isinstance(accepted, (bool, np.bool_)):
            raise ValueError("selection_event must return a boolean")
        return bool(accepted)

    def score_value(matrix):
        value = score(matrix)
        if isinstance(value, bool) or not isinstance(value, Real) or not math.isfinite(value):
            raise ValueError("score must return a finite real number")
        return float(value)

    rng = np.random.default_rng(seed)
    anchor = int(rng.integers(steps + 1))
    scores = [None] * (steps + 1)
    moves = [None] * (steps + 1)
    result = dict(
        status="unresolved",
        decision="unresolved",
        p_value=None,
        error=None,
        sampling_status="unresolved",
        observed_score=None,
        anchor_index=anchor,
        accepted_moves=0,
        completed_steps=0,
        event_evaluations=0,
        chain_scores=scores,
        chain_moves=moves,
        alpha=float(alpha),
        seed=int(seed),
        steps_requested=int(steps),
        resolution_sufficient=1 / (steps + 1) <= alpha,
        subtree_indices=rows.tolist(),
        numerical_event_certified=False,
        selection_event="fixed_membership" if selection_event is None else "caller_supplied",
        observation_model="independent_gaussian_common_mean_covariance",
    )
    observed = x.copy()
    observed.setflags(write=False)
    phase = "observation"
    try:
        if not in_event(observed):
            raise ValueError("observed matrix does not satisfy selection_event")
        scores[anchor] = result["observed_score"] = score_value(observed)
        with np.errstate(over="raise", invalid="raise"):
            basis = helmert(len(rows), full=False)
            mean = x[rows].mean(axis=0)
            original = basis @ (x[rows] - mean)
            phase = "sampling"
            for positions in (range(anchor - 1, -1, -1), range(anchor + 1, steps + 1)):
                current, current_score = original, scores[anchor]
                for index in positions:
                    proposed, move = _rotated(current, rng)
                    if not np.array_equal(proposed, current):
                        candidate = x.copy()
                        candidate[rows] = mean + basis.T @ proposed
                        candidate.setflags(write=False)
                        if selection_event is not None:
                            result["event_evaluations"] += 1
                        if in_event(candidate):
                            current_score = score_value(candidate)
                            current = proposed
                            moves[index] = move
                            result["accepted_moves"] += 1
                    scores[index] = current_score
                    result["completed_steps"] += 1
    except Exception as error:
        result.update(error_phase=phase, error=f"{type(error).__name__}: {error}")
        return result

    p_value = sum(value >= scores[anchor] for value in scores) / (steps + 1)
    moved = result["accepted_moves"] > 0
    result.update(
        status="complete", p_value=p_value, sampling_status="moved" if moved else "stalled"
    )
    if moved and result["resolution_sufficient"]:
        result["decision"] = "reject" if p_value <= alpha else "do_not_reject"
    return result


def fixed_subtree_contrast_test(
    values, *, subtree_indices, left_indices, rank=None, all_labels=False, alpha=0.05
):
    """Test a fixed binary row contrast on a prespecified common-mean subtree.

    The statistic is squared projection of the unit row contrast onto the
    leading rank-k left singular subspace of the centered subtree. The right
    directions and numerical rank are functions of its feature Gram matrix.
    For fixed membership and labels the conditional law is Beta(k/2,(m-1-k)/2).
    An optional Bonferroni factor counts all unordered nonempty binary splits.
    It permits label search on the specified subtree. Selected membership and
    contrast-selected rank are outside the reference.
    """
    x, rows = _inputs(values, subtree_indices, alpha)
    left = _rows(left_indices, len(x), name="left_indices", minimum=1)
    left_mask = np.isin(rows, left)
    if left_mask.sum() != len(left) or len(left) == len(rows):
        raise ValueError("left_indices must form a nonempty proper subset of subtree_indices")
    if rank is not None and (isinstance(rank, bool) or not isinstance(rank, Integral) or rank < 0):
        raise ValueError("rank must be a nonnegative integer or None")
    if not isinstance(all_labels, (bool, np.bool_)):
        raise ValueError("all_labels must be a boolean")

    m = len(rows)
    count = (1 << (m - 1)) - 1 if all_labels else 1
    result = dict(
        status="unresolved",
        decision="unresolved",
        error=None,
        p_value=None,
        unadjusted_p_value=None,
        statistic=None,
        rank=None,
        subtree_indices=rows.tolist(),
        left_indices=left.tolist(),
        number_of_labels=count,
        alpha=float(alpha),
        selection_event="fixed_membership",
        all_labels=bool(all_labels),
        observation_model="independent_gaussian_common_mean_covariance",
    )
    try:
        with np.errstate(over="raise", invalid="raise"):
            basis = helmert(m, full=False)
            centered = x[rows] - x[rows].mean(axis=0)
            u, singular, _ = np.linalg.svd(basis @ centered, full_matrices=False)
        tolerance = singular[0] * max(centered.shape) * np.finfo(float).eps
        full_rank = int(np.count_nonzero(singular > tolerance))
        k = full_rank if rank is None else int(rank)
        result["rank"] = k
        if not 0 < k < m - 1 or k > full_rank:
            raise ValueError(
                "rank is zero, exceeds the numerical rank, or spans the centered row space"
            )
        if k < full_rank and singular[k - 1] - singular[k] <= tolerance:
            raise ValueError("rank cuts a tied Gram eigenspace")
        contrast = np.where(left_mask, 1 / len(left), -1 / (m - len(left)))
        contrast /= np.linalg.norm(contrast)
        q = float(np.linalg.norm(u[:, :k].T @ (basis @ contrast)) ** 2)
        if not 0 <= q < 1:
            raise FloatingPointError("projection statistic is outside numerical Beta support")
        log_tail = float(beta.logsf(q, k / 2, (m - 1 - k) / 2))
        if not math.isfinite(log_tail):
            raise FloatingPointError("Beta tail underflowed or is nonfinite")
    except (ValueError, FloatingPointError, np.linalg.LinAlgError) as error:
        result["error"] = f"{type(error).__name__}: {error}"
        return result

    p_value = math.exp(min(0.0, log_tail + math.log(count)))
    return result | dict(
        status="complete",
        decision="reject" if p_value <= alpha else "do_not_reject",
        p_value=p_value,
        unadjusted_p_value=math.exp(log_tail),
        statistic=q,
    )

def ward_family_examples():

            str(rtol): sorted(geo.ward_topology_family(points, rtol=rtol), key=repr)

        "inputs": {name: points.tolist() for name, points in point_sets.items()},
        "topologies_by_rtol": families,
        "axis_union": len(left | right), "axis_intersection": len(left & right),
        "axis_pair_orders": pair_orders,
        "axis_strict_reversals": sum(np.prod(r["orders_by_axis"]) == -1 for r in pair_orders).item(),

    }


def additive_fit_examples():
    """Fit all 15 rooted topologies to each of five fixed distance geometries."""
    geometries = {
        "rectangle_first_axis": pdist(examples["rectangle"][:, :1]),
        "rectangle_second_axis": pdist(examples["rectangle"][:, 1:]),
        "rectangle_full": pdist(examples["rectangle"]),
        "square_full": pdist(square),
        "simplex": pdist(np.eye(4)),
    }

    def triangle_violation(distances):
        d = squareform(distances)
        return float(np.max(d[:, None, :] - (d[:, :, None] + d[None, :, :])))

    out = {}
    for name, observed in geometries.items():
        observed_violation = triangle_violation(observed)
        assert observed_violation <= 1e-12
        rows = []
        for tree in geo.rooted_binary_topologies(4):
            fit = geo.fit_additive_distances(tree, observed)
            fitted_violation = triangle_violation(fit.fitted_distances)
            assert fitted_violation <= 1e-12
            rows.append({
                "tree": tree,
                "edge_splits": [asdict(split) for split in fit.edge_splits],
                "lengths": fit.branch_lengths.tolist(),
                "fitted_distances": fit.fitted_distances.tolist(),
                "rmse": fit.rmse,
                "max_abs_residual": fit.max_abs_residual,
                "fitted_max_triangle_violation": fitted_violation,
            })
        out[name] = {
            "observed_distances": observed.tolist(),
            "observed_max_triangle_violation": observed_violation,
            "fits": rows,
        }
    return out

"""Bounded geometry diagnostics on a fixed labeled subtree.

Leaf labels are row positions. Cut enumeration, Ward trees in spectral spans,
and the combinatorial topology family are separate objects.
"""

from __future__ import annotations

from dataclasses import dataclass
from functools import cache
from itertools import combinations, product

import numpy as np
from scipy.cluster.hierarchy import cophenet, linkage
from scipy.spatial.distance import pdist, squareform

from tree_break_selection.eigenvalue_groups import tied_eigenvalue_neighbors
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import (
    is_separable,
)

MAX_CUT_LEAVES = 10
MAX_TOPOLOGY_LEAVES = 6

type Tree = int | tuple[Tree, Tree]


@dataclass(frozen=True)
class Cut:
    """Unordered split with the smallest leaf on the left."""

    left: tuple[int, ...]
    right: tuple[int, ...]


@dataclass(frozen=True, eq=False)
class SpectralSubspace:
    """Descending Gram eigenblock ranges and the resulting sample geometry."""

    eigenblocks: tuple[tuple[int, int], ...]
    rank: int
    projector: np.ndarray
    coordinates: np.ndarray


@dataclass(frozen=True, eq=False)
class SpectralGeometry:
    subspace: SpectralSubspace
    cuts: tuple[Cut, ...]
    distances: np.ndarray
    ward_linkage: np.ndarray
    ward_tree: Tree


def unordered_cuts(n: int) -> tuple[Cut, ...]:
    """All nonempty complementary splits, with row zero on the left."""
    if not 2 <= n <= MAX_CUT_LEAVES:
        raise ValueError(f"Cut enumeration requires 2 through {MAX_CUT_LEAVES} leaves.")
    return tuple(
        Cut(
            (0, *(j for j in range(1, n) if mask & (1 << (j - 1)))),
            tuple(j for j in range(1, n) if not mask & (1 << (j - 1))),
        )
        for mask in range((1 << (n - 1)) - 1)
    )


def spectral_subspaces(points: np.ndarray) -> tuple[SpectralSubspace, ...]:
    """Return each positive whole Gram block and each distinct leading block union.

    Each projector acts on row space. The coordinates retain Euclidean geometry
    within the projected feature space. SVD avoids explicitly squaring the
    condition number or allocating the feature Gram matrix.
    """
    x = np.asarray(points, dtype=np.float64)
    if x.ndim != 2 or min(x.shape) < 1 or not np.isfinite(x).all():
        raise ValueError("Expected a finite, nonempty rows-by-features matrix.")
    if not 2 <= len(x) <= MAX_CUT_LEAVES:
        raise ValueError(f"Subtree geometry requires 2 through {MAX_CUT_LEAVES} leaves.")

    # Scale before centering and SVD to keep the Gram spectrum finite.
    scale = float(np.max(np.abs(x)))
    if scale == 0:
        return ()
    normalized = x / scale
    normalized -= normalized[0].copy()
    u, singular, _ = np.linalg.svd(normalized - normalized.mean(axis=0), full_matrices=False)
    threshold = np.finfo(float).eps * max(x.shape) * singular[0]
    rank = min(len(x) - 1, int(np.count_nonzero(singular > threshold)))
    if rank == 0:
        return ()

    values = (singular[:rank] / singular[0]) ** 2
    ends = [*(np.flatnonzero(~tied_eigenvalue_neighbors(values)) + 1).tolist(), rank]
    blocks = tuple(zip([0, *ends[:-1]], ends, strict=True))
    groups = [(block,) for block in blocks] + [blocks[:k] for k in range(2, len(blocks) + 1)]
    subspaces = []
    for group in groups:
        start, stop = group[0][0], group[-1][1]
        row_basis = u[:, start:stop]
        coordinates = (row_basis * singular[start:stop]) * scale
        if not np.isfinite(coordinates).all():
            raise ValueError("Projected coordinates must fit in finite float64.")
        subspaces.append(
            SpectralSubspace(group, stop - start, row_basis @ row_basis.T, coordinates)
        )
    return tuple(subspaces)


def spectral_geometry(points: np.ndarray) -> tuple[SpectralGeometry, ...]:
    """Enumerate strict affine cuts and compute a fixed-row-order Ward tree per span."""
    subspaces = spectral_subspaces(points)
    records = []
    for subspace in subspaces:
        x = subspace.coordinates
        # Column rescaling preserves affine feasibility and avoids scaling the LP
        # by the Gram eigenvalues or measurement units.
        lp_points = x / np.max(np.abs(x), axis=0)
        cuts = tuple(
            cut
            for cut in unordered_cuts(len(x))
            if is_separable(lp_points, np.isin(np.arange(len(x)), cut.left))
        )
        scale = np.max(np.abs(x))
        scaled_distances = pdist(x / scale)
        distances = scaled_distances * scale
        if not np.isfinite(distances).all():
            raise ValueError("Projected distances must fit in finite float64.")
        # shortcut: round relative input distances at 1e-12 for deterministic numerical
        # ties in fixed row order, replace if exact near-tie Ward decisions are required.
        max_distance = scaled_distances.max()
        ward = linkage(np.round(scaled_distances / max_distance, 12), method="ward")
        ward[:, 2] *= max_distance * scale
        if not np.isfinite(ward).all():
            raise ValueError("Ward linkage heights must fit in finite float64.")
        nodes: list[Tree] = list(range(len(x)))
        first_leaf = list(range(len(x)))
        for a, b in ward[:, :2].astype(int):
            if first_leaf[b] < first_leaf[a]:
                a, b = b, a
            nodes.append((nodes[a], nodes[b]))
            first_leaf.append(first_leaf[a])
        records.append(SpectralGeometry(subspace, cuts, distances, ward, nodes[-1]))
    return tuple(records)


def rooted_binary_topologies(n: int) -> tuple[Tree, ...]:
    """Enumerate all unordered rooted full binary trees of at most six labeled leaves."""
    if not 2 <= n <= MAX_TOPOLOGY_LEAVES:
        raise ValueError(f"Topology enumeration requires 2 through {MAX_TOPOLOGY_LEAVES} leaves.")

    def recurse(labels: tuple[int, ...]) -> tuple[Tree, ...]:
        if len(labels) == 1:
            return (labels[0],)
        trees = []
        for cut in unordered_cuts(len(labels)):
            left = recurse(tuple(labels[i] for i in cut.left))
            right = recurse(tuple(labels[i] for i in cut.right))
            trees.extend(product(left, right))
        return tuple(trees)

    return recurse(tuple(range(n)))


def ward_topology_family(
    points_or_condensed: np.ndarray, *, rtol: float = 1e-10
) -> frozenset[Tree]:
    """All Ward trees reachable by tied merges on two through six leaves.

    Accepts Euclidean coordinates or condensed Euclidean distances. At each
    step include every increase in within-cluster SSE within `rtol` times the
    largest original squared pair distance of the minimum. Zero uses exact
    floating-point comparisons.
    """
    x = np.asarray(points_or_condensed, dtype=float)
    if not np.isfinite(rtol) or rtol < 0:
        raise ValueError("rtol must be finite and nonnegative.")
    if not x.size or not np.isfinite(x).all() or x.ndim not in (1, 2):
        raise ValueError("Expected finite coordinates or condensed Euclidean distances.")
    if x.ndim == 1 and (x < 0).any():
        raise ValueError("Euclidean distances must be nonnegative.")
    scale = np.max(np.abs(x))
    if scale > 0:
        x = x / scale
    if x.ndim == 2 and not 2 <= len(x) <= MAX_TOPOLOGY_LEAVES:
        raise ValueError(
            f"Ward topology enumeration requires 2 through {MAX_TOPOLOGY_LEAVES} leaves."
        )
    distances = pdist(x) if x.ndim == 2 else x
    if not 1 <= len(distances) <= MAX_TOPOLOGY_LEAVES * (MAX_TOPOLOGY_LEAVES - 1) // 2:
        raise ValueError(
            f"Ward topology enumeration requires 2 through {MAX_TOPOLOGY_LEAVES} leaves."
        )
    max_distance = distances.max()
    normalized = distances / max_distance if max_distance > 0 else distances
    squared = squareform(normalized**2)
    n = len(squared)
    if x.ndim == 1:
        centered = np.eye(n) - 1 / n
        if np.linalg.eigvalsh(-0.5 * centered @ squared @ centered)[0] < -1e-12:
            raise ValueError("Condensed distances must be Euclidean.")

    @cache
    def merge_cost(left: tuple[int, ...], right: tuple[int, ...]) -> float:
        a, b = len(left), len(right)
        cross = squared[np.ix_(left, right)].sum()
        within_a = sum(squared[i, j] for i, j in combinations(left, 2))
        within_b = sum(squared[i, j] for i, j in combinations(right, 2))
        # (ab/(a+b)) times the squared centroid distance, via pair distances.
        return max(0.0, (a * b * cross - b**2 * within_a - a**2 * within_b) / (a * b * (a + b)))

    @cache
    def expand(forest: tuple[tuple[tuple[int, ...], Tree], ...]) -> frozenset[Tree]:
        if len(forest) == 1:
            return frozenset((forest[0][1],))
        costs = [
            (merge_cost(forest[i][0], forest[j][0]), i, j)
            for i, j in combinations(range(len(forest)), 2)
        ]
        threshold = min(cost for cost, _, _ in costs) + rtol
        trees = set()
        for cost, i, j in costs:
            if cost <= threshold:
                left, right = forest[i], forest[j]
                merged = (tuple(sorted(left[0] + right[0])), (left[1], right[1]))
                trees.update(expand(forest[:i] + (merged,) + forest[i + 1 : j] + forest[j + 1 :]))
        return frozenset(trees)

    return expand(tuple(((i,), i) for i in range(n)))


def _clades(tree: Tree) -> list[frozenset[int]]:
    if isinstance(tree, int):
        return [frozenset((tree,))]
    left, right = map(_clades, tree)
    return [*left, *right, left[-1] | right[-1]]


def tree_clades(tree: Tree) -> frozenset[frozenset[int]]:
    """Nontrivial clades for rooted topology comparison."""
    return frozenset(clade for clade in _clades(tree)[:-1] if len(clade) > 1)


def tree_edge_splits(tree: Tree) -> frozenset[Cut]:
    """Unordered edge splits including pendant edges, deduplicating the root split."""
    clades = _clades(tree)
    leaves = clades.pop()
    first = min(leaves)
    splits = set()
    for clade in clades:
        left, right = (clade, leaves - clade) if first in clade else (leaves - clade, clade)
        splits.add(Cut(tuple(sorted(left)), tuple(sorted(right))))
    return frozenset(splits)


def rooted_robinson_foulds(left: Tree, right: Tree) -> int:
    """Number of nontrivial rooted clades present in only one of the trees."""
    if _clades(left)[-1] != _clades(right)[-1]:
        raise ValueError("Rooted RF requires the same leaf labels in both trees.")
    return len(tree_clades(left) ^ tree_clades(right))


def cophenetic_distances(tree_or_linkage: Tree | np.ndarray) -> np.ndarray:
    """Condensed cophenetic distances in row order.

    An abstract topology assigns a node the height of its longest path to a
    leaf in edge units. A linkage matrix preserves its supplied heights.
    """
    if isinstance(tree_or_linkage, np.ndarray):
        return cophenet(tree_or_linkage)

    leaves = _clades(tree_or_linkage)[-1]
    n = len(leaves)
    if n < 2 or leaves != frozenset(range(n)):
        raise ValueError("Cophenetic topology leaves must be numbered 0 through n-1.")
    merges = []

    def build(node: Tree) -> tuple[int, int, int]:
        if isinstance(node, int):
            return node, 0, 1
        a, ha, na = build(node[0])
        b, hb, nb = build(node[1])
        parent = n + len(merges)
        height, count = 1 + max(ha, hb), na + nb
        merges.append((a, b, height, count))
        return parent, height, count

    _, _, count = build(tree_or_linkage)
    if count != n:
        raise ValueError("Cophenetic topology leaves must be unique.")
    return cophenet(np.asarray(merges, dtype=float))

def sibling_cut_scores(values, cuts):
    """Existing full-space raw sibling Wald scores, zero branches, guarded covariance.

    This matches the earlier full-space sibling diagnostic, including parent
    covariance fallback. It excludes empirical inflation and adaptive rank.
    """
    data = _frame(values)
    x = data.to_numpy()
    space = continuous_feature_space_from_columns(data.columns)
    tree = build_tree(data, pdist(x), linkage_method="ward").tree
    tree.populate_node_distributions(data, feature_space=space)
    p = x.shape[1]
    scores = []
    for cut in cuts:
        if not cut.left or not cut.right or sorted((*cut.left, *cut.right)) != list(range(len(x))):
            raise ValueError("Each cut must partition the local row indices exactly once")
        covariance = pooled_within_child_covariance_by_block(
            tree, tree.root(), space, x, [list(cut.left), list(cut.right)]
        )
        statistic, _, _, _ = sibling_divergence_test(
            x[list(cut.left)].mean(0),
            x[list(cut.right)].mean(0),
            len(cut.left),
            len(cut.right),
            branch_length_sum=0.0,
            mean_branch_length=None,
            projection_dimension_from_edge_comparisons=p,
            parent_principal_component_projection=np.eye(p),
            parent_principal_component_eigenvalues=np.ones(p),
            feature_space=space,
            continuous_covariance_by_block=covariance,
        )
        if not np.isfinite(statistic):
            raise ValueError("Nonfinite sibling statistic")
        scores.append(statistic)
    return np.asarray(scores, dtype=float)


def _candidate_scores(points):
    spaces = spectral_subspaces(points)
    if not spaces:
        return (), np.array([], dtype=float)
    # Every smaller-span affine cut lies in the full cumulative span.
    coordinates = max(spaces, key=lambda space: space.rank).coordinates
    coordinates = coordinates / np.max(np.abs(coordinates), axis=0)
    cuts = tuple(
        cut
        for cut in unordered_cuts(len(points))
        if is_separable(coordinates, np.isin(np.arange(len(points)), cut.left))
    )
    return cuts, sibling_cut_scores(points, cuts)


def subtree_alternative_cut_test(

    result = subtree_gaussian_chain_test(

        selection_event=selection_event,
        steps=steps,
        seed=seed,

def ward_subtree_selection(values, *, path=(), keep_split=False):
    """Capture a prespecified Ward node path, ordered by the smallest leaf ID.

    Return (row_indices, event). The event preserves every ancestral child pair;
    keep_split also preserves the tested node's own pair. The path must be chosen
    before observing the data. It does not encode an additional search for a node
    or a TBS gate/traversal decision. All outside rows stay fixed in the sampler.
    """
    path = tuple(path)
    if any(
        isinstance(side, bool) or not isinstance(side, Integral) or side not in (0, 1)
        for side in path
    ):
        raise ValueError("path must contain only child positions 0 or 1")
    if not isinstance(keep_split, (bool, np.bool_)):
        raise ValueError("keep_split must be boolean")
    original = _frame(values)

    def describe(matrix):
        data = _frame(matrix)
        tree = build_tree(data, pdist(data.to_numpy()), linkage_method="ward").tree
        clades = tree.compute_descendant_sets()
        node, ancestors = tree.root(), []
        for depth in range(len(path) + 1):
            children = sorted(tree.successors(node), key=lambda child: min(clades[child]))
            if len(children) != 2:
                return None
            if depth < len(path) or keep_split:
                ancestors.append(tuple(tuple(sorted(clades[child])) for child in children))
            if depth == len(path):
                return tuple(sorted(clades[node])), tuple(ancestors)
            node = children[path[depth]]

    observed = describe(original.to_numpy())
    if observed is None:
        raise ValueError("The specified path must reach an internal node")
    rows = np.flatnonzero(original.index.isin(observed[0]))
    return rows, lambda matrix: describe(matrix) == observed

def topology_sibling_score(values, topology: Tree):
    """Prespecified topology score: sum of raw full-space sibling scores at its nodes."""
    x = _frame(values).to_numpy()

    def leaves(node):
        if isinstance(node, int):
            return [node]
        if not isinstance(node, tuple) or len(node) != 2:
            raise ValueError("Expected a binary nested-tuple topology")
        return leaves(node[0]) + leaves(node[1])

    if sorted(leaves(topology)) != list(range(len(x))):
        raise ValueError("Topology must contain each local row index exactly once")

    def score(node):
        if isinstance(node, int):
            return 0.0
        left, right = leaves(node[0]), leaves(node[1])
        cut = Cut(tuple(range(len(left))), tuple(range(len(left), len(left) + len(right))))
        local = sibling_cut_scores(x[left + right], (cut,))[0]
        return float(local + score(node[0]) + score(node[1]))

    return score(topology)


def topology_pair_reference(
    values,
    *,
    subtree_indices,
    first: Tree,
    second: Tree,
    selection_event=None,
    steps=999,
    seed=0,
    alpha=0.05,
):
    """Paired absolute score difference for two prespecified labeled topologies.

    A data-selected pair needs its pair-selection event as well. The null is
    homogeneous Gaussian rows, not equal expected accuracy of the two trees.
    """
    x, rows = _inputs(values, subtree_indices, alpha)

    def score(matrix):
        return abs(
            topology_sibling_score(matrix[rows], first)
            - topology_sibling_score(matrix[rows], second)
        )

    result = subtree_gaussian_chain_test(
        x,
        subtree_indices=rows,
        score=score,
        selection_event=selection_event,
        steps=steps,
        seed=seed,
        alpha=alpha,
    )
    return result | {"score": "absolute_difference_of_summed_full_space_sibling_scores"}


def subtree_topology_search_test(
    values, *, subtree_indices, selection_event=None, steps=999, seed=0, alpha=0.05
):
    """Replay the union of tied-Ward families over whole spectral spans (at most six rows).

    The geometry rule fixes the relative tie tolerance at 1e-10. Counts are
    descriptive; the maximum is recalculated over the new family on every draw.
    Scores sum raw full-space sibling scores, using the same feature space for
    every candidate. They are a declared diagnostic criterion, not a likelihood.
    """
    x, rows = _inputs(values, subtree_indices, alpha)
    if len(rows) > 6:
        raise ValueError("Tied-Ward topology search requires at most six subtree leaves")

    def candidates(matrix):
        points = matrix[rows]
        trees = set()
        for space in spectral_subspaces(points):
            trees.update(ward_topology_family(space.coordinates, rtol=1e-10))
        ordered = sorted(trees, key=repr)
        return ordered, [topology_sibling_score(points, tree) for tree in ordered]

    observed = {}

    def score(matrix):
        trees, scores = candidates(matrix)
        if not observed:
            observed.update(trees=trees, scores=scores)
        return max(scores, default=0.0)

    result = subtree_gaussian_chain_test(
        x,
        subtree_indices=rows,
        score=score,
        selection_event=selection_event,
        steps=steps,
        seed=seed,
        alpha=alpha,
    )
    result.update(candidate_family="union_of_spectral_tied_ward_topologies", tie_rtol=1e-10)
    result["topology_results"] = [
        dict(
            topology=tree,
            statistic=statistic,
            global_null_adjusted_p_value=(
                sum(value >= statistic for value in result["chain_scores"]) / (steps + 1)
                if result["status"] == "complete"
                else None
            ),
        )
        for tree, statistic in zip(
            observed.get("trees", ()), observed.get("scores", ()), strict=True
        )
    ]
    return result

"""Bounded geometry diagnostics on a fixed labeled subtree.

Leaf labels are row positions. Cut enumeration, Ward trees in spectral spans,
and the combinatorial topology family are separate objects.
"""

from __future__ import annotations

from dataclasses import dataclass
from functools import cache
from itertools import combinations, product

import numpy as np
from scipy.cluster.hierarchy import cophenet, linkage
from scipy.optimize import nnls
from scipy.spatial.distance import pdist, squareform

from tree_break_selection.eigenvalue_groups import tied_eigenvalue_neighbors
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import (
    is_separable,
)

MAX_CUT_LEAVES = 10
MAX_TOPOLOGY_LEAVES = 6

type Tree = int | tuple[Tree, Tree]


@dataclass(frozen=True)
class Cut:
    """Unordered split with the smallest leaf on the left."""

    left: tuple[int, ...]
    right: tuple[int, ...]


@dataclass(frozen=True, eq=False)
class SpectralSubspace:
    """Descending Gram eigenblock ranges and the resulting sample geometry."""

    eigenblocks: tuple[tuple[int, int], ...]
    rank: int
    projector: np.ndarray
    coordinates: np.ndarray


@dataclass(frozen=True, eq=False)
class SpectralGeometry:
    subspace: SpectralSubspace
    cuts: tuple[Cut, ...]
    distances: np.ndarray
    ward_linkage: np.ndarray
    ward_tree: Tree


@dataclass(frozen=True, eq=False)
class AdditiveDistanceFit:
    """Nonnegative lengths by unique edge split and fitted condensed distances."""

    edge_splits: tuple[Cut, ...]
    branch_lengths: np.ndarray
    fitted_distances: np.ndarray
    rmse: float
    max_abs_residual: float


def unordered_cuts(n: int) -> tuple[Cut, ...]:
    """All nonempty complementary splits, with row zero on the left."""
    if not 2 <= n <= MAX_CUT_LEAVES:
        raise ValueError(f"Cut enumeration requires 2 through {MAX_CUT_LEAVES} leaves.")
    return tuple(
        Cut(
            (0, *(j for j in range(1, n) if mask & (1 << (j - 1)))),
            tuple(j for j in range(1, n) if not mask & (1 << (j - 1))),
        )
        for mask in range((1 << (n - 1)) - 1)
    )


def spectral_subspaces(points: np.ndarray) -> tuple[SpectralSubspace, ...]:
    """Return each positive whole Gram block and each distinct leading block union.

    Each projector acts on row space. The coordinates retain Euclidean geometry
    within the projected feature space. SVD avoids explicitly squaring the
    condition number or allocating the feature Gram matrix.
    """
    x = np.asarray(points, dtype=np.float64)
    if x.ndim != 2 or min(x.shape) < 1 or not np.isfinite(x).all():
        raise ValueError("Expected a finite, nonempty rows-by-features matrix.")
    if not 2 <= len(x) <= MAX_CUT_LEAVES:
        raise ValueError(f"Subtree geometry requires 2 through {MAX_CUT_LEAVES} leaves.")

    # Scale before centering and SVD to keep the Gram spectrum finite.
    scale = float(np.max(np.abs(x)))
    if scale == 0:
        return ()
    normalized = x / scale
    normalized -= normalized[0].copy()
    u, singular, _ = np.linalg.svd(normalized - normalized.mean(axis=0), full_matrices=False)
    threshold = np.finfo(float).eps * max(x.shape) * singular[0]
    rank = min(len(x) - 1, int(np.count_nonzero(singular > threshold)))
    if rank == 0:
        return ()

    values = (singular[:rank] / singular[0]) ** 2
    ends = [*(np.flatnonzero(~tied_eigenvalue_neighbors(values)) + 1).tolist(), rank]
    blocks = tuple(zip([0, *ends[:-1]], ends, strict=True))
    groups = [(block,) for block in blocks] + [blocks[:k] for k in range(2, len(blocks) + 1)]
    subspaces = []
    for group in groups:
        start, stop = group[0][0], group[-1][1]
        row_basis = u[:, start:stop]
        coordinates = (row_basis * singular[start:stop]) * scale
        if not np.isfinite(coordinates).all():
            raise ValueError("Projected coordinates must fit in finite float64.")
        subspaces.append(
            SpectralSubspace(group, stop - start, row_basis @ row_basis.T, coordinates)
        )
    return tuple(subspaces)


def spectral_geometry(points: np.ndarray) -> tuple[SpectralGeometry, ...]:
    """Enumerate strict affine cuts and compute a fixed-row-order Ward tree per span."""
    subspaces = spectral_subspaces(points)
    records = []
    for subspace in subspaces:
        x = subspace.coordinates
        # Column rescaling preserves affine feasibility and avoids scaling the LP
        # by the Gram eigenvalues or measurement units.
        lp_points = x / np.max(np.abs(x), axis=0)
        cuts = tuple(
            cut
            for cut in unordered_cuts(len(x))
            if is_separable(lp_points, np.isin(np.arange(len(x)), cut.left))
        )
        scale = np.max(np.abs(x))
        scaled_distances = pdist(x / scale)
        distances = scaled_distances * scale
        if not np.isfinite(distances).all():
            raise ValueError("Projected distances must fit in finite float64.")
        # shortcut: round relative input distances at 1e-12 for deterministic numerical
        # ties in fixed row order, replace if exact near-tie Ward decisions are required.
        max_distance = scaled_distances.max()
        ward = linkage(np.round(scaled_distances / max_distance, 12), method="ward")
        ward[:, 2] *= max_distance * scale
        if not np.isfinite(ward).all():
            raise ValueError("Ward linkage heights must fit in finite float64.")
        nodes: list[Tree] = list(range(len(x)))
        first_leaf = list(range(len(x)))
        for a, b in ward[:, :2].astype(int):
            if first_leaf[b] < first_leaf[a]:
                a, b = b, a
            nodes.append((nodes[a], nodes[b]))
            first_leaf.append(first_leaf[a])
        records.append(SpectralGeometry(subspace, cuts, distances, ward, nodes[-1]))
    return tuple(records)


def rooted_binary_topologies(n: int) -> tuple[Tree, ...]:
    """Enumerate all unordered rooted full binary trees of at most six labeled leaves."""
    if not 2 <= n <= MAX_TOPOLOGY_LEAVES:
        raise ValueError(f"Topology enumeration requires 2 through {MAX_TOPOLOGY_LEAVES} leaves.")

    def recurse(labels: tuple[int, ...]) -> tuple[Tree, ...]:
        if len(labels) == 1:
            return (labels[0],)
        trees = []
        for cut in unordered_cuts(len(labels)):
            left = recurse(tuple(labels[i] for i in cut.left))
            right = recurse(tuple(labels[i] for i in cut.right))
            trees.extend(product(left, right))
        return tuple(trees)

    return recurse(tuple(range(n)))


def ward_topology_family(
    points_or_condensed: np.ndarray, *, rtol: float = 1e-10
) -> frozenset[Tree]:
    """All Ward trees reachable by tied merges on two through six leaves.

    Accepts Euclidean coordinates or condensed Euclidean distances. At each
    step include every increase in within-cluster SSE within `rtol` times the
    largest original squared pair distance of the minimum. Zero uses exact
    floating-point comparisons.
    """
    x = np.asarray(points_or_condensed, dtype=float)
    if not np.isfinite(rtol) or rtol < 0:
        raise ValueError("rtol must be finite and nonnegative.")
    if not x.size or not np.isfinite(x).all() or x.ndim not in (1, 2):
        raise ValueError("Expected finite coordinates or condensed Euclidean distances.")
    if x.ndim == 1 and (x < 0).any():
        raise ValueError("Euclidean distances must be nonnegative.")
    scale = np.max(np.abs(x))
    if scale > 0:
        x = x / scale
    if x.ndim == 2 and not 2 <= len(x) <= MAX_TOPOLOGY_LEAVES:
        raise ValueError(
            f"Ward topology enumeration requires 2 through {MAX_TOPOLOGY_LEAVES} leaves."
        )
    distances = pdist(x) if x.ndim == 2 else x
    if not 1 <= len(distances) <= MAX_TOPOLOGY_LEAVES * (MAX_TOPOLOGY_LEAVES - 1) // 2:
        raise ValueError(
            f"Ward topology enumeration requires 2 through {MAX_TOPOLOGY_LEAVES} leaves."
        )
    max_distance = distances.max()
    normalized = distances / max_distance if max_distance > 0 else distances
    squared = squareform(normalized**2)
    n = len(squared)
    if x.ndim == 1:
        centered = np.eye(n) - 1 / n
        if np.linalg.eigvalsh(-0.5 * centered @ squared @ centered)[0] < -1e-12:
            raise ValueError("Condensed distances must be Euclidean.")

    @cache
    def merge_cost(left: tuple[int, ...], right: tuple[int, ...]) -> float:
        a, b = len(left), len(right)
        cross = squared[np.ix_(left, right)].sum()
        within_a = sum(squared[i, j] for i, j in combinations(left, 2))
        within_b = sum(squared[i, j] for i, j in combinations(right, 2))
        # (ab/(a+b)) times the squared centroid distance, via pair distances.
        return max(0.0, (a * b * cross - b**2 * within_a - a**2 * within_b) / (a * b * (a + b)))

    @cache
    def expand(forest: tuple[tuple[tuple[int, ...], Tree], ...]) -> frozenset[Tree]:
        if len(forest) == 1:
            return frozenset((forest[0][1],))
        costs = [
            (merge_cost(forest[i][0], forest[j][0]), i, j)
            for i, j in combinations(range(len(forest)), 2)
        ]
        threshold = min(cost for cost, _, _ in costs) + rtol
        trees = set()
        for cost, i, j in costs:
            if cost <= threshold:
                left, right = forest[i], forest[j]
                merged = (tuple(sorted(left[0] + right[0])), (left[1], right[1]))
                trees.update(expand(forest[:i] + (merged,) + forest[i + 1 : j] + forest[j + 1 :]))
        return frozenset(trees)

    return expand(tuple(((i,), i) for i in range(n)))


def _clades(tree: Tree) -> list[frozenset[int]]:
    if isinstance(tree, int):
        return [frozenset((tree,))]
    left, right = map(_clades, tree)
    return [*left, *right, left[-1] | right[-1]]


def tree_clades(tree: Tree) -> frozenset[frozenset[int]]:
    """Nontrivial clades for rooted topology comparison."""
    return frozenset(clade for clade in _clades(tree)[:-1] if len(clade) > 1)


def tree_edge_splits(tree: Tree) -> frozenset[Cut]:
    """Unordered edge splits including pendant edges, deduplicating the root split."""
    clades = _clades(tree)
    leaves = clades.pop()
    first = min(leaves)
    splits = set()
    for clade in clades:
        left, right = (clade, leaves - clade) if first in clade else (leaves - clade, clade)
        splits.add(Cut(tuple(sorted(left)), tuple(sorted(right))))
    return frozenset(splits)


def fit_additive_distances(tree: Tree, condensed_distances: np.ndarray) -> AdditiveDistanceFit:
    """Fit all leaf-pair distances to unique edge splits on a small fixed tree.

    Targets use the supplied distance units. Complementary root edges have
    the same incidence column, so their lengths are represented by one sum.
    Zero fitted internal lengths give unresolved splits in the fitted metric.
    """
    clades = _clades(tree)
    n = len(clades[-1])
    if not 2 <= n <= MAX_TOPOLOGY_LEAVES:
        raise ValueError(
            f"Additive distance fitting requires 2 through {MAX_TOPOLOGY_LEAVES} leaves."
        )
    if clades[-1] != frozenset(range(n)) or sum(len(clade) == 1 for clade in clades) != n:
        raise ValueError("Expected unique leaf labels 0 through n-1.")
    observed = np.asarray(condensed_distances, dtype=float)
    if (
        observed.shape != (n * (n - 1) // 2,)
        or not np.isfinite(observed).all()
        or (observed < 0).any()
    ):
        raise ValueError("Expected finite nonnegative condensed distances for the supplied leaves.")

    splits = tuple(sorted(tree_edge_splits(tree), key=lambda split: split.left))
    membership = np.column_stack([np.isin(np.arange(n), split.left) for split in splits])
    left, right = np.triu_indices(n, k=1)
    design = (membership[left] != membership[right]).astype(float)
    scale = float(observed.max()) or 1.0
    normalized = observed / scale
    lengths, _ = nnls(design, normalized)
    fitted = design @ lengths
    residuals = fitted - normalized
    lengths, fitted = lengths * scale, fitted * scale
    if not np.isfinite(fitted).all():
        raise ValueError("Fitted distances must fit in finite float64.")
    return AdditiveDistanceFit(
        splits,
        lengths,
        fitted,
        float(np.sqrt(np.mean(residuals**2))) * scale,
        float(np.max(np.abs(residuals))) * scale,
    )


def rooted_robinson_foulds(left: Tree, right: Tree) -> int:
    """Number of nontrivial rooted clades present in only one of the trees."""
    if _clades(left)[-1] != _clades(right)[-1]:
        raise ValueError("Rooted RF requires the same leaf labels in both trees.")
    return len(tree_clades(left) ^ tree_clades(right))


def cophenetic_distances(tree_or_linkage: Tree | np.ndarray) -> np.ndarray:
    """Condensed cophenetic distances in row order.

    An abstract topology assigns a node the height of its longest path to a
    leaf in edge units. A linkage matrix preserves its supplied heights.
    """
    if isinstance(tree_or_linkage, np.ndarray):
        return cophenet(tree_or_linkage)

    leaves = _clades(tree_or_linkage)[-1]
    n = len(leaves)
    if n < 2 or leaves != frozenset(range(n)):
        raise ValueError("Cophenetic topology leaves must be numbered 0 through n-1.")
    merges = []

    def build(node: Tree) -> tuple[int, int, int]:
        if isinstance(node, int):
            return node, 0, 1
        a, ha, na = build(node[0])
        b, hb, nb = build(node[1])
        parent = n + len(merges)
        height, count = 1 + max(ha, hb), na + nb
        merges.append((a, b, height, count))
        return parent, height, count

    _, _, count = build(tree_or_linkage)
    if count != n:
        raise ValueError("Cophenetic topology leaves must be unique.")
    return cophenet(np.asarray(merges, dtype=float))

"""Gaussian common-mean references for a specified subtree."""

import math
from numbers import Integral, Real

import numpy as np
from scipy.linalg import helmert
from scipy.stats import beta


def _rows(indices, n, *, name, minimum):
    try:
        rows = tuple(indices)
    except TypeError as error:
        raise ValueError(f"{name} must be a sequence of row indices") from error
    if (
        len(rows) < minimum
        or any(isinstance(i, bool) or not isinstance(i, Integral) or not 0 <= i < n for i in rows)
        or len(set(rows)) != len(rows)
    ):
        raise ValueError(f"{name} must contain at least {minimum} distinct valid row indices")
    return np.asarray(rows, dtype=np.intp)


def _inputs(values, subtree_indices, alpha):
    if isinstance(alpha, bool) or not isinstance(alpha, Real) or not 0 < alpha < 1:
        raise ValueError("alpha must lie strictly between zero and one")
    x = np.asarray(values, dtype=float)
    if x.ndim != 2 or x.shape[1] < 1 or not np.isfinite(x).all():
        raise ValueError("values must be a finite matrix with at least one feature")
    rows = _rows(subtree_indices, len(x), name="subtree_indices", minimum=2)
    # Membership alone fixes the proposal basis, including for data-ordered input indices.
    return x, np.sort(rows)


def _rotated(state, rng):
    proposed = state.copy()
    if len(state) == 1:
        sign = 1 if rng.integers(2) else -1
        proposed[0] *= sign
        return proposed, (0, 0, 0.0, sign)
    i, j = sorted(map(int, rng.choice(len(state), size=2, replace=False)))
    angle = float(rng.uniform(-np.pi, np.pi))
    sign = 1 if rng.integers(2) else -1
    c, s = math.cos(angle), math.sin(angle)
    proposed[i] = c * state[i] - s * state[j]
    proposed[j] = sign * (s * state[i] + c * state[j])
    return proposed, (i, j, angle, sign)


def subtree_gaussian_chain_test(
    values, *, subtree_indices, score, selection_event=None, steps=999, seed=0, alpha=0.05
):
    """Rank a deterministic full-matrix score under the common-mean Gaussian null.

    Condition on the subtree mean, centered feature Gram matrix, outside rows
    and, if supplied, the caller's selection event. Subtree rows must be iid
    Gaussian with one common mean and covariance and independent of outside
    rows. Unknown covariance is allowed. The event must describe how the tested
    membership and any other conditioned selection were obtained. Omitting it
    requires prespecified membership. Callbacks receive read-only matrices.
    Any score-defining choices captured by callbacks must be prespecified or
    part of the event. A failed evaluation invalidates the rank. No descendant or traversal event
    is inferred here, and native floating-point event membership is uncertified.

    Chain moves store accepted transforms as (i, j, angle, sign), relative to
    the preceding state when moving outward from the observed anchor. An empty
    entry keeps the preceding state. Equal indices denote the one-axis case.
    """
    x, rows = _inputs(values, subtree_indices, alpha)
    for name, value, minimum in (("steps", steps, 1), ("seed", seed, 0)):
        if isinstance(value, bool) or not isinstance(value, Integral) or value < minimum:
            raise ValueError(f"{name} must be an integer >= {minimum}")
    if not callable(score) or (selection_event is not None and not callable(selection_event)):
        raise ValueError("score and any selection_event must be callable")

    def in_event(matrix):
        if selection_event is None:
            return True
        accepted = selection_event(matrix)
        if not isinstance(accepted, (bool, np.bool_)):
            raise ValueError("selection_event must return a boolean")
        return bool(accepted)

    def score_value(matrix):
        value = score(matrix)
        if isinstance(value, bool) or not isinstance(value, Real) or not math.isfinite(value):
            raise ValueError("score must return a finite real number")
        return float(value)

    rng = np.random.default_rng(seed)
    anchor = int(rng.integers(steps + 1))
    scores = [None] * (steps + 1)
    moves = [None] * (steps + 1)
    result = dict(
        status="unresolved",
        decision="unresolved",
        p_value=None,
        error=None,
        sampling_status="unresolved",
        observed_score=None,
        anchor_index=anchor,
        accepted_moves=0,
        completed_steps=0,
        event_evaluations=0,
        chain_scores=scores,
        chain_moves=moves,
        alpha=float(alpha),
        seed=int(seed),
        steps_requested=int(steps),
        resolution_sufficient=1 / (steps + 1) <= alpha,
        subtree_indices=rows.tolist(),
        numerical_event_certified=False,
        selection_event="fixed_membership" if selection_event is None else "caller_supplied",
        observation_model="independent_gaussian_common_mean_covariance",
    )
    observed = x.copy()
    observed.setflags(write=False)
    phase = "observation"
    try:
        if not in_event(observed):
            raise ValueError("observed matrix does not satisfy selection_event")
        scores[anchor] = result["observed_score"] = score_value(observed)
        with np.errstate(over="raise", invalid="raise"):
            basis = helmert(len(rows), full=False)
            mean = x[rows].mean(axis=0)
            original = basis @ (x[rows] - mean)
            phase = "sampling"
            for positions in (range(anchor - 1, -1, -1), range(anchor + 1, steps + 1)):
                current, current_score = original, scores[anchor]
                for index in positions:
                    proposed, move = _rotated(current, rng)
                    if not np.array_equal(proposed, current):
                        candidate = x.copy()
                        candidate[rows] = mean + basis.T @ proposed
                        candidate.setflags(write=False)
                        if selection_event is not None:
                            result["event_evaluations"] += 1
                        if in_event(candidate):
                            current_score = score_value(candidate)
                            current = proposed
                            moves[index] = move
                            result["accepted_moves"] += 1
                    scores[index] = current_score
                    result["completed_steps"] += 1
    except Exception as error:
        result.update(error_phase=phase, error=f"{type(error).__name__}: {error}")
        return result

    p_value = sum(value >= scores[anchor] for value in scores) / (steps + 1)
    moved = result["accepted_moves"] > 0
    result.update(
        status="complete", p_value=p_value, sampling_status="moved" if moved else "stalled"
    )
    if moved and result["resolution_sufficient"]:
        result["decision"] = "reject" if p_value <= alpha else "do_not_reject"
    return result


def fixed_subtree_contrast_test(
    values, *, subtree_indices, left_indices, rank=None, all_labels=False, alpha=0.05
):
    """Test a fixed binary row contrast on a prespecified common-mean subtree.

    The statistic is squared projection of the unit row contrast onto the
    leading rank-k left singular subspace of the centered subtree. The right
    directions and numerical rank are functions of its feature Gram matrix.
    For fixed membership and labels the conditional law is Beta(k/2,(m-1-k)/2).
    An optional Bonferroni factor counts all unordered nonempty binary splits.
    It permits label search on the specified subtree. Selected membership and
    contrast-selected rank are outside the reference.
    """
    x, rows = _inputs(values, subtree_indices, alpha)
    left = _rows(left_indices, len(x), name="left_indices", minimum=1)
    left_mask = np.isin(rows, left)
    if left_mask.sum() != len(left) or len(left) == len(rows):
        raise ValueError("left_indices must form a nonempty proper subset of subtree_indices")
    if rank is not None and (isinstance(rank, bool) or not isinstance(rank, Integral) or rank < 0):
        raise ValueError("rank must be a nonnegative integer or None")
    if not isinstance(all_labels, (bool, np.bool_)):
        raise ValueError("all_labels must be a boolean")

    m = len(rows)
    count = (1 << (m - 1)) - 1 if all_labels else 1
    result = dict(
        status="unresolved",
        decision="unresolved",
        error=None,
        p_value=None,
        unadjusted_p_value=None,
        statistic=None,
        rank=None,
        subtree_indices=rows.tolist(),
        left_indices=left.tolist(),
        number_of_labels=count,
        alpha=float(alpha),
        selection_event="fixed_membership",
        all_labels=bool(all_labels),
        observation_model="independent_gaussian_common_mean_covariance",
    )
    try:
        with np.errstate(over="raise", invalid="raise"):
            basis = helmert(m, full=False)
            centered = x[rows] - x[rows].mean(axis=0)
            u, singular, _ = np.linalg.svd(basis @ centered, full_matrices=False)
        tolerance = singular[0] * max(centered.shape) * np.finfo(float).eps
        full_rank = int(np.count_nonzero(singular > tolerance))
        k = full_rank if rank is None else int(rank)
        result["rank"] = k
        if not 0 < k < m - 1 or k > full_rank:
            raise ValueError(
                "rank is zero, exceeds the numerical rank, or spans the centered row space"
            )
        if k < full_rank and singular[k - 1] - singular[k] <= tolerance:
            raise ValueError("rank cuts a tied Gram eigenspace")
        contrast = np.where(left_mask, 1 / len(left), -1 / (m - len(left)))
        contrast /= np.linalg.norm(contrast)
        q = float(np.linalg.norm(u[:, :k].T @ (basis @ contrast)) ** 2)
        if not 0 <= q < 1:
            raise FloatingPointError("projection statistic is outside numerical Beta support")
        log_tail = float(beta.logsf(q, k / 2, (m - 1 - k) / 2))
        if not math.isfinite(log_tail):
            raise FloatingPointError("Beta tail underflowed or is nonfinite")
    except (ValueError, FloatingPointError, np.linalg.LinAlgError) as error:
        result["error"] = f"{type(error).__name__}: {error}"
        return result

    p_value = math.exp(min(0.0, log_tail + math.log(count)))
    return result | dict(
        status="complete",
        decision="reject" if p_value <= alpha else "do_not_reject",
        p_value=p_value,
        unadjusted_p_value=math.exp(log_tail),
        statistic=q,
    )

def test_chain_preserves_subtree_mean_gram_and_outside_rows_at_full_row_rank():

    rows = [8, 1, 3, 4, 7]
    outside = sorted(set(range(len(x))) - set(rows))
    centered = x[rows] - x[rows].mean(0)

    def score(values):
        np.testing.assert_array_equal(values[outside], x[outside])
        np.testing.assert_allclose(values[rows].mean(0), x[rows].mean(0), atol=1e-14)
        y = values[rows] - values[rows].mean(0)
        np.testing.assert_allclose(y.T @ y, centered.T @ centered, atol=1e-13)
        return values[rows[0], 0]

    assert r["status"] == "complete"
    assert r["accepted_moves"] == r["completed_steps"] == 39
    assert r["chain_scores"][r["anchor_index"]] == score(x)
    assert r["p_value"] == np.mean(np.asarray(r["chain_scores"]) >= score(x))
    assert r["anchor_index"] == np.random.default_rng(17).integers(40)
    assert len(r["chain_moves"]) == 40
    assert {move[3] for move in r["chain_moves"] if move is not None} == {-1, 1}
    sorted_rows = r["subtree_indices"]
    sorted_centered = x[sorted_rows] - x[sorted_rows].mean(0)
    basis = helmert(len(rows), full=False)
    anchor = r["anchor_index"]
    for positions in (range(anchor - 1, -1, -1), range(anchor + 1, 40)):

            a, b = state[[i, j]]
            c, sine = np.cos(angle), np.sin(angle)

            assert r["chain_scores"][position] == pytest.approx(

            )

    assert r == subtree_gaussian_chain_test(x, subtree_indices=rows, score=score, steps=39, seed=17)

def test_whole_tied_block_rotation_preserves_geometry_and_ward_tie_convention():
    square = np.array([[-1, -1], [-1, 1], [1, -1], [1, 1]], dtype=float)
    angle = 0.42
    rot = np.array([[np.cos(angle), -np.sin(angle)], [np.sin(angle), np.cos(angle)]])
    (first,) = spectral_geometry(square)
    (rotated,) = spectral_geometry(square @ rot)
    assert first.subspace.eigenblocks == rotated.subspace.eigenblocks == ((0, 2),)
    assert first.subspace.rank == rotated.subspace.rank == 2
    np.testing.assert_allclose(first.subspace.projector, rotated.subspace.projector, atol=1e-12)
    np.testing.assert_allclose(first.distances, rotated.distances)
    np.testing.assert_allclose(first.distances, pdist(square))
    assert first.cuts == rotated.cuts
    assert len(first.cuts) == 6 and Cut((0, 3), (1, 2)) not in first.cuts
    assert first.ward_tree == rotated.ward_tree
    np.testing.assert_allclose(
        cophenetic_distances(first.ward_linkage), cophenet(first.ward_linkage)
    )

def test_additive_distance_fit_recovers_lengths_and_is_invariant_to_root():
    observed = np.array([3, 8, 9, 9, 10, 7], dtype=float)
    balanced = fit_additive_distances(((0, 1), (2, 3)), observed)
    rerooted = fit_additive_distances((0, (1, (2, 3))), observed)
    wrong = fit_additive_distances(((0, 2), (1, 3)), observed)
    assert balanced.edge_splits == rerooted.edge_splits
    expected = {
        Cut((0,), (1, 2, 3)): 1,
        Cut((0, 2, 3), (1,)): 2,
        Cut((0, 1, 3), (2,)): 3,
        Cut((0, 1, 2), (3,)): 4,

    }
    np.testing.assert_allclose(balanced.branch_lengths, [expected[s] for s in balanced.edge_splits])
    np.testing.assert_allclose(balanced.branch_lengths, rerooted.branch_lengths)
    np.testing.assert_allclose(balanced.fitted_distances, observed)
    assert balanced.rmse < 1e-12 and balanced.max_abs_residual < 1e-12
    assert wrong.rmse > 0.1 and wrong.max_abs_residual > 0.1
    assert wrong.branch_lengths[wrong.edge_splits.index(Cut((0, 2), (1, 3)))] == 0

def test_sibling_score_matches_pooled_hotelling_and_parent_fallback():
    x = np.random.default_rng(12).normal(size=(20, 3))
    for count in (4, 10):
        left, right = x[:count], x[count:]
        if count < 8:
            covariance = np.cov(x, rowvar=False)
        else:
            residual = np.vstack([left - left.mean(0), right - right.mean(0)])
            covariance = residual.T @ residual / 18
        difference = left.mean(0) - right.mean(0)
        expected = (
            difference @ np.linalg.solve(covariance, difference) / (1 / count + 1 / (20 - count))
        )
        score = sibling_cut_scores(x, [Cut(tuple(range(count)), tuple(range(count, 20)))])
        assert score[0] == pytest.approx(expected, rel=1e-10)
