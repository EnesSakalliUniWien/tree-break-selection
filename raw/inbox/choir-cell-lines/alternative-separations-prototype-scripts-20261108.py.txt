# ===== alternative_separations.py =====
"""Prototype: report alternative separations where a subtree's top eigenvalues tie.

For a subtree whose top eigenvalues above the noise band form one eigenspace of multiplicity m >= 2
(block equality test, `subtree_spectrum_topology_20261108/run.py`), the tree is cut into m + 1 blocks by
following its own splits (largest merge height first; peels smaller than `min_block` are walked through and
their cells left out). Every two-sided split of the blocks is reported with its strength
n_S n_R / n * d' W^-1 d (W = pooled within-block covariance) and whether the tree holds it (one side is the
block set under a node of the tree). Splits the tree does not hold are the alternative separations.
"""

import importlib.util
import itertools
from pathlib import Path

import numpy as np

_spec = importlib.util.spec_from_file_location(
    "subtree_spectrum", Path(__file__).resolve().parent.parent / "subtree_spectrum_topology_20261108" / "run.py")
spectrum_module = importlib.util.module_from_spec(_spec)
_spec.loader.exec_module(spectrum_module)
spectrum = spectrum_module.spectrum


class Tree:
    """Children, sizes and merge heights of a scipy linkage (leaves 0..n-1)."""

    def __init__(self, z):
        self.n = len(z) + 1
        self.left, self.right = z[:, 0].astype(int), z[:, 1].astype(int)
        self.height = np.r_[np.zeros(self.n), z[:, 2]]
        self.size = np.r_[np.ones(self.n, dtype=int), z[:, 3].astype(int)]

    def children(self, u):
        return (self.left[u - self.n], self.right[u - self.n]) if u >= self.n else None

    def leaves(self, u):
        stack, out = [u], []
        while stack:
            w = stack.pop()
            if w < self.n:
                out.append(w)
            else:
                stack += list(self.children(w))
        return np.array(out)

    def effective_children(self, u, min_block):
        """First split below u whose two sides both hold >= min_block cells (walking down the larger side)."""
        while u >= self.n:
            a, b = self.children(u)
            if min(self.size[a], self.size[b]) >= min_block:
                return a, b
            u = a if self.size[a] >= self.size[b] else b
        return None


def blocks_of(tree, v, count, min_block):
    """Cut subtree v into `count` blocks along the tree; returns block nodes and the held block sets."""
    first = tree.effective_children(v, min_block)
    if first is None:
        return [], set(), frozenset()
    blocks, held, top = list(first), [frozenset([0]), frozenset([1])], {0}  # top: blocks of v's first child
    while len(blocks) < count:
        splittable = [(tree.height[b], i) for i, b in enumerate(blocks) if tree.effective_children(b, min_block)]
        if not splittable:
            break
        _, i = max(splittable)
        a, b = tree.effective_children(blocks[i], min_block)
        j = len(blocks)
        blocks[i] = a
        blocks.append(b)
        # every held set containing block i now contains the new block j as well; i and j alone are new clades
        held = [h | {j} if i in h else h for h in held] + [frozenset([i]), frozenset([j])]
        if i in top:
            top.add(j)
    return blocks, set(held), frozenset(top)


def report_node(points, tree, v, min_block):
    """Alternative-separation rows for subtree v, or [] when its top eigenvalues do not tie."""
    cells = tree.leaves(v)
    s = spectrum(points[cells])
    m = s["multiplicity"][0] if s["multiplicity"] else 0
    if m < 2:
        return [], s
    blocks, held, top = blocks_of(tree, v, m + 1, min_block)
    g = len(blocks)
    if g < 3:
        return [], s
    block_cells = [tree.leaves(b) for b in blocks]
    x = [points[c] for c in block_cells]
    centered = np.vstack([b - b.mean(0) for b in x])
    w_inv = np.linalg.pinv(centered.T @ centered / (len(centered) - g))
    sizes = np.array([len(c) for c in block_cells])
    means = np.array([b.mean(0) for b in x])
    rows = []
    for size in range(1, g // 2 + 1):
        for side in itertools.combinations(range(g), size):
            if 2 * size == g and 0 not in side:
                continue
            other = [i for i in range(g) if i not in side]
            n_s, n_r = sizes[list(side)].sum(), sizes[other].sum()
            d = sizes[list(side)] @ means[list(side)] / n_s - sizes[other] @ means[other] / n_r
            rows.append({"node": int(v), "n": int(len(cells)), "k": s["k"], "multiplicity": "+".join(map(str, s["multiplicity"])),
                         "m": int(m), "blocks": g, "block_sizes": sizes.tolist(), "left_out": int(len(cells) - sizes.sum()),
                         "side": list(side), "strength": float(n_s * n_r / (n_s + n_r) * d @ w_inv @ d),
                         "held": frozenset(side) in held or frozenset(other) in held,
                         "tree_split": top in (frozenset(side), frozenset(other)),
                         "block_cells": block_cells})
    first_strength = next(r["strength"] for r in rows if r["tree_split"])
    for r in rows:
        r["ratio_to_tree_split"] = r["strength"] / first_strength
    return rows, s


def report(points, z, *, min_n=60, min_block=None, min_small=0):
    """Rows for every subtree of the tree whose top eigenvalues tie."""
    tree = Tree(z)
    rows = []
    for v in range(tree.n, 2 * tree.n - 1):
        if tree.size[v] < min_n:
            continue
        a, b = tree.children(v)
        if min(tree.size[a], tree.size[b]) < min_small:
            continue
        block = min_block if min_block is not None else max(20, int(0.02 * tree.size[v]))
        node_rows, _ = report_node(points, tree, v, block)
        rows += node_rows
    return rows


def _self_check():
    """Tied case: 4 groups at the corners of a regular simplex; distinct case: groups at unequal spacing."""
    from scipy.cluster.hierarchy import linkage

    rng = np.random.default_rng(7)
    p, per = 30, 250
    corners = np.c_[np.eye(4) - 0.25, np.zeros((4, p - 4))] * 8.0  # equal pairwise distances
    tied = np.vstack([c + rng.standard_normal((per, p)) for c in corners])
    rows = report(tied, linkage(tied, "ward"), min_n=900)
    root = [r for r in rows if r["n"] == 4 * per]
    assert root and root[0]["m"] == 3 and root[0]["blocks"] == 4, "tied simplex root must be flagged with 4 blocks"
    assert len(root) == 7 and sum(r["held"] for r in root) == 5, "4 blocks: 7 splits, the tree holds 5"
    alternatives = [r for r in root if not r["held"]]
    assert all(r["ratio_to_tree_split"] > 0.9 for r in alternatives), "dropped 2+2 splits must be about as strong"
    for block in root[0]["block_cells"]:
        assert len(np.unique(block // per)) == 1, "each block must be one simulated group"
    spaced = np.vstack([c * f + rng.standard_normal((per, p)) for c, f in zip(corners, (1.0, 1.6, 2.4, 3.4))])
    rows = report(spaced, linkage(spaced, "ward"), min_n=900)
    assert not [r for r in rows if r["n"] == 4 * per], "unequal spacing must not be flagged at the root"
    print("self-check passed:", [round(r["ratio_to_tree_split"], 3) for r in alternatives])


if __name__ == "__main__":
    _self_check()

# ===== run.py =====
"""Run the alternative-separation prototype on simulated data (both trees) and the real CHOIR trees.

usage: run.py sim WORKERS | real
"""

import json
import sys
from multiprocessing import Pool
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parent / "depth_confound_20261107"))
sys.path.insert(1, str(HERE.parents[2]))
sys.path.insert(2, str(HERE))

import numpy as np  # noqa: E402
import pandas as pd  # noqa: E402
from scipy.cluster.hierarchy import linkage  # noqa: E402
from scipy.spatial.distance import pdist  # noqa: E402
from threadpoolctl import threadpool_limits  # noqa: E402

from alternative_separations import report  # noqa: E402


def describe(rows, labels):
    """Drop cell lists; add each block's main label and its share, and whether every block is one label."""
    out = []
    for r in rows:
        tops = [pd.Series(labels[c]).value_counts(normalize=True) for c in r["block_cells"]]
        r = {k: v for k, v in r.items() if k != "block_cells"}
        r["block_labels"] = [str(t.index[0]) for t in tops]
        r["block_purity"] = [round(float(t.iloc[0]), 3) for t in tops]
        r["side"] = [int(i) for i in r["side"]]
        out.append(r)
    return out


def sim_task(job):
    import depth  # noqa: PLC0415
    from validate import prepare  # noqa: PLC0415

    model, replicate = job
    counts, truth = depth.generate(model, replicate)
    labels = np.asarray(truth).astype(str)
    rows = []
    with threadpool_limits(limits=1):
        data, symmetric, markov = prepare(counts)
        points = data.to_numpy(dtype=np.float64)
        for name, coords, method in (("symmetric_average", symmetric, "average"), ("markov_ward", markov, "ward")):
            z = linkage(pdist(np.asarray(coords)), method=method)
            rows += [{"model": model, "replicate": replicate, "tree": name, **r}
                     for r in describe(report(points, z, min_n=60), labels)]
    return model, replicate, rows


def simulate(workers):
    import depth  # noqa: F401,PLC0415
    from validate import MODELS  # noqa: PLC0415

    jobs = [(m, r) for m in ("signal_balanced", "signal_imbalanced", "signal_singleton", "signal_density_variation")
            for r in range(MODELS[m]["replicates"])] + [(m, r) for m in ("low", "matched_median", "high") for r in range(50)]
    with Pool(workers) as pool, (HERE / "sim_rows.jsonl").open("w") as f, (HERE / "sim_done.jsonl").open("w") as d:
        for model, replicate, rows in pool.imap_unordered(sim_task, jobs):
            f.write("".join(json.dumps(r) + "\n" for r in rows))
            d.write(json.dumps({"model": model, "replicate": replicate}) + "\n")


def real():
    base = HERE.parent
    meta = pd.read_csv(base / "choir_cell_lines_tbs_merged_20261106/cell_metadata.csv", index_col=0, dtype=str)
    points = np.load(base / "choir_cell_lines_tbs_merged_20261106/pca.npy").astype(np.float64)
    labels = meta.Cell_line.to_numpy()
    trees = {"symmetric_average": base / "choir_cell_lines_tbs_merged_20261106/linkage.npy",
             "markov_ward": base / "choir_tbs_recovery_comparison_20261103_174622/markov_ward/linkage.npy"}
    with threadpool_limits(limits=4), (HERE / "real_rows.jsonl").open("w") as f:
        for name, path in trees.items():
            rows = describe(report(points, np.load(path), min_n=500, min_small=50), labels)
            f.write("".join(json.dumps({"tree": name, **r}) + "\n" for r in rows))
            print(name, len({r["node"] for r in rows}), "flagged nodes", flush=True)


if __name__ == "__main__":
    real() if sys.argv[1] == "real" else simulate(int(sys.argv[2]))

# ===== summarize.py =====
"""Summaries of the prototype's output (sim_rows.jsonl, real_rows.jsonl)."""
import json
from pathlib import Path

import numpy as np
import pandas as pd

HERE = Path(__file__).resolve().parent
pd.set_option("display.width", 250)
pd.set_option("display.max_colwidth", 120)


def load(name):
    return pd.DataFrame(map(json.loads, (HERE / f"{name}.jsonl").read_text().splitlines()))


def per_node(rows, keys):
    def one(x):
        alt = x[~x.held]
        return pd.Series({"n": x.n.iloc[0], "m": x.m.iloc[0], "blocks": x.blocks.iloc[0], "left_out": x.left_out.iloc[0],
                          "pure_blocks": bool(min(x.block_purity.iloc[0]) >= 0.9),
                          "between_groups": len(set(x.block_labels.iloc[0])) >= 2,
                          "alternatives": len(alt), "best_alternative": alt.ratio_to_tree_split.max() if len(alt) else np.nan})
    return rows.groupby(keys + ["node"]).apply(one, include_groups=False).reset_index()


def main():
    sim, done = load("sim_rows"), load("sim_done")
    nodes = per_node(sim, ["model", "replicate", "tree"])
    datasets = done.groupby("model").size().rename("datasets")
    table = nodes.groupby(["model", "tree", "between_groups"]).agg(
        flagged=("node", "size"), median_n=("n", "median"), median_m=("m", "median"),
        share_pure_blocks=("pure_blocks", "mean"), median_alternatives=("alternatives", "median"),
        median_best_alternative=("best_alternative", "median"),
        share_best_alt_ge_090=("best_alternative", lambda x: (x >= 0.9).mean())).join(datasets, on="model")
    table["flagged_per_dataset"] = table.flagged / table.datasets
    print("Simulated\n", table.round(3).to_string(), "\n")
    print("Flagged nodes by m\n", nodes.groupby(["between_groups", "m"]).agg(nodes=("node", "size"),
          median_alternatives=("alternatives", "median"), best_alt=("best_alternative", "median")).round(3).to_string(), "\n")
    roots = nodes[nodes.n >= nodes.groupby(["model", "replicate", "tree"]).n.transform("max")]
    print("Flagged roots\n", roots.groupby(["model", "tree"]).agg(flagged_roots=("node", "size"),
          m=("m", "median"), pure=("pure_blocks", "mean"), best_alt=("best_alternative", "median")).round(3).to_string(), "\n")
    real = load("real_rows")
    rn = per_node(real, ["tree"])
    print("Real flagged nodes\n", rn.round(3).to_string(index=False), "\n")
    alt = real[~real.held].sort_values("ratio_to_tree_split", ascending=False)
    show = alt.groupby(["tree", "node"]).head(2)[["tree", "node", "n", "m", "block_sizes", "block_labels", "block_purity",
                                                   "side", "ratio_to_tree_split"]]
    print("Real: two strongest alternatives per flagged node\n", show.round(3).to_string(index=False))
    table.round(4).to_csv(HERE / "sim_summary.csv")
    rn.to_csv(HERE / "real_nodes.csv", index=False)
    show.to_csv(HERE / "real_alternatives.csv", index=False)


if __name__ == "__main__":
    main()
