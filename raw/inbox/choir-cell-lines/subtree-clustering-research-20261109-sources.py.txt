def data_seed(*, phase: int, panel: int, cell: int, replicate: int) -> int:
    """Separate data phases/panels without selecting seeds from their results."""
    if phase not in (1, 2) or min(panel, cell, replicate) < 0:
        raise ValueError("Expected phase 1/2 and nonnegative panel, cell and replicate")
    return int(
        np.random.SeedSequence([20261109, phase, panel, cell, replicate]).generate_state(1)[0]
    )


def recovery_data(cell: int, replicate: int, *, phase: int = 1, n: int = 240):
    p, imbalanced, gap, anisotropic = RECOVERY_CELLS[cell]
    if n != 240 and imbalanced:
        raise ValueError("The prespecified imbalanced counts require 240 observations")
    if n % 4:
        raise ValueError("Balanced square panels require a multiple of four observations")
    seed = data_seed(phase=phase, panel=1, cell=cell, replicate=replicate)
    rng = np.random.default_rng(seed)
    sizes = (132, 60, 36, 12) if imbalanced else (n // 4,) * 4
    truth = np.repeat(np.arange(4), sizes)
    values = rng.standard_normal((n, p))
    if anisotropic:
        values[:, :2] = (values[:, :2] * [2, 0.5]) @ (np.array([[1, -1], [1, 1]]) / np.sqrt(2))
    means = gap * np.array([[-1, -1], [-1, 1], [1, -1], [1, 1]]) / 2
    values[:, :2] += means[truth]
    order = rng.permutation(n)
    return values[order], truth[order], seed


def null_data(cell: int, replicate: int, *, phase: int = 2):
    n, p, covariance, path = NULL_CELLS[cell]
    seed = data_seed(phase=phase, panel=2, cell=cell, replicate=replicate)
    if covariance == "ar08":
        sigma = 0.8 ** np.abs(np.arange(p)[:, None] - np.arange(p)[None, :])
    elif covariance == "diagonal100":
        sigma = np.diag(np.geomspace(1.0, 100.0, p))
    else:
        sigma = np.eye(p)
    values = np.random.default_rng(seed).standard_normal((n, p)) @ np.linalg.cholesky(sigma).T
    return values, path, sigma, seed


def _frame(values):
    x = np.asarray(values, dtype=float)
    if x.ndim != 2 or len(x) < 2 or x.shape[1] < 1 or not np.isfinite(x).all():
        raise ValueError("Expected a finite matrix with at least two rows")
    # These labels have the same order as build_tree's canonical string order.
    data = pd.DataFrame(x, index=[f"S{i:06d}" for i in range(len(x))])
    data.columns = [f"F{i}" for i in range(x.shape[1])]
    return data, continuous_feature_space_from_columns(data.columns)


def evaluate_tree(data, truth, space, z, *, branch_source="nnls", energy=True, trace=False):
    """Run native gates/traversal on a supplied local topology, retaining its heights."""
    if branch_source not in ("linkage", "nnls", "zero"):
        raise ValueError("Unknown branch source")
    tree = tree_from_linkage(z, data.index.tolist())
    fit = None
    if branch_source == "nnls":
        fit = fit_fixed_topology_nnls_branch_lengths(
            tree,
            data,
            pair_sample_size=None,
            random_state=0,
            solver_tolerance=1e-5,
            max_iterations=5000,
        )
        if not fit.applied_to_tree:
            return {"status": "error", "reason": f"NNLS not applied: {fit.status}"}, {}
    elif branch_source == "zero":
        for edge in tree.edges:
            tree.edges[edge]["branch_length"] = 0.0
    tree.populate_node_distributions(data, feature_space=space)
    bundle = run_gate_annotation_pipeline(
        tree,
        tree.annotations_df,
        leaf_data=data,
        feature_space=space,
        edge_alpha=0.001,
        sibling_alpha=0.01,
        spectral_minimum_dimension=2,
        adaptive_projection_dimension_energy_fraction=0.9 if energy else None,
        continuous_covariance_policy="guarded_within_child",
        continuous_covariance_min_child_leaf_count=8,
        edge_branch_length_variance_policy="none"
        if branch_source == "zero"
        else "normalized_branch_length",
        sibling_gate_method="projected_wald_inflation",
        enforce_internal_support_thresholds=False,
    )
    annotations = bundle.annotated_df
    tree.annotations_df = annotations
    reason = unsupported_empirical_null_reason(
        annotations, sibling_gate_method="projected_wald_inflation"
    )
    row = {"status": "unsupported" if reason else "ok", "reason": str(reason) if reason else ""}
    root = tree_root(tree)
    row.update(
        root_sibling_p=float(annotations.loc[root, "Sibling_Divergence_P_Value"]),
        root_sibling_reject=bool(annotations.loc[root, "Sibling_BH_Different"]),
        mean_positive_branch=compute_mean_branch_length(tree),
        branch_source=branch_source,
        energy=bool(energy),
    )
    if fit is not None:
        row.update(nnls_status=fit.status, nnls_relative_rmse=fit.residual_rmse_to_target_mean)
    details = {}
    if reason is None:
        decomposition = TreeDecomposition(
            tree=tree,
            gate_annotation_bundle=bundle,
            passthrough=True,
            trace_level="full" if trace else "compact",
        ).decompose_tree()
        labels, report = labels_and_report_from_decomposition(decomposition, data.index.tolist())
        metrics = calculate_clustering_metrics(report, data.index, np.asarray(truth))
        row.update(
            asdict(metrics),
            found_clusters=int(len(np.unique(labels))),
            labels=np.asarray(labels).tolist(),
        )
        if trace:
            details["traversal"] = decomposition
    if trace:
        spectral = bundle.edge_gate_result.spectral_context
        dimensions = audit.derive_sibling_projection_dimensions_from_child_edge_comparisons(
            tree, spectral_context=spectral
        )
        projections, eigenvalues = (
            audit.collect_parent_principal_component_inputs_for_sibling_tests(
                dimensions, spectral_context=spectral
            )
        )
        records, _ = audit.collect_sibling_pair_records(
            tree,
            bundle.edge_gate_result.annotated_df,
            feature_space=space,
            sibling_projection_dimensions_from_edge_comparisons=dimensions,
            parent_principal_component_projections=projections,
            parent_principal_component_eigenvalues=eigenvalues,
            adaptive_projection_dimension_energy_fraction=0.9 if energy else None,
        )
        details["raw_records"] = [asdict(record) for record in records]
        keep = [
            column for column in annotations if column.startswith(("Sibling_", "Child_Parent_"))
        ]
        details["annotations"] = annotations[keep].reset_index().to_dict("records")
        details["branches"] = [
            (str(a), str(b), float(d["branch_length"])) for a, b, d in tree.edges(data=True)
        ]
        clades = tree.compute_descendant_sets()
        position = {name: i for i, name in enumerate(data.index)}
        subtrees = {
            str(node): tuple(
                tuple(position[name] for name in sorted(clades[child]))
                for child in tree.successors(node)
            )
            for node in tree
            if tree.out_degree(node) == 2
        }
        details["subtree_profiles"] = build_statistical_profiles(
            data, subtrees=subtrees
        ).subtrees.to_dict("records")
    return row, details


def run_method(
    values,
    truth,
    *,
    operator="symmetric",
    linkage_method="average",
    branch_source="nnls",
    energy=True,
    candidate=False,
    trace=False,
):
    """Complete independent geometry-to-partition timing; truth is used only for metrics."""
    start = perf_counter()
    data, space = _frame(values)
    geometry = feature_family_diffusion_geometry(data, space, coordinate_operator=operator)
    z = linkage(geometry.distance_condensed, method=linkage_method)
    extra = {"candidate_count": 1, "candidate_search_sec": 0.0}
    if candidate:
        selected = select_subtree_topology(
            data.to_numpy(),
            current_topology=topology_from_linkage(z),
            current_linkage=z,
            leaf_labels=data.index.tolist(),
            target_metric="mean_squared_standardized_euclidean",
        )
        z = selected.selected_linkage
        extra.update(
            candidate_count=selected.candidate_count,
            candidate_search_sec=selected.elapsed_sec,
            candidate_coverage=selected.coverage,
            candidate_pair_count=selected.pair_count,
        )
    row, details = evaluate_tree(
        data, truth, space, z, branch_source=branch_source, energy=energy, trace=trace
    )
    row.update(
        extra,
        elapsed_sec=perf_counter() - start,
        process_lifetime_peak_rss_bytes=int(resource.getrusage(resource.RUSAGE_SELF).ru_maxrss)
        * (1 if sys.platform == "darwin" else 1024),
        operator=operator,
        linkage_method=linkage_method,
        candidate=bool(candidate),
    )
    if trace:
        details.update(
            linkage=z.tolist(),
            baseline_diffusion_distance=geometry.distance_condensed.tolist(),
            geometry_metadata=geometry.metadata,
        )
        if candidate:
            details["candidate_fits"] = [asdict(item) for item in selected.candidates]
    return row, details


def source_hashes():
    paths = set((ROOT / "tree_break_selection").rglob("*.py"))
    paths |= set((ROOT / "benchmarks/shared").rglob("*.py"))
    paths |= set((ROOT / "benchmarks/diagnostics/calibration").rglob("*.py"))
    paths |= {ROOT / "uv.lock"}
    return {
        str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(paths)
    }


def _json(path, value):
    path.write_text(json.dumps(value, default=diagnostic_json_default, indent=2) + "\n")


def _safe_run(values, truth, **params):
    start = perf_counter()
    try:
        return run_method(values, truth, **params)
    except (ValueError, RuntimeError, np.linalg.LinAlgError, FloatingPointError) as exc:
        return {
            "status": "error",
            "reason": f"{type(exc).__name__}: {exc}",
            "elapsed_sec": perf_counter() - start,
        }, {}


def runtime_gate(rows):
    """All cases require three successful complete paired timings and ratio <= 5."""
    frame = pd.DataFrame(rows)
    results = []
    for (n, p), group in frame.groupby(["n", "p"]):
        medians = group.groupby("method").elapsed_sec.median()
        complete = all(
            len(group[group.method == m]) == 3
            and group.loc[group.method == m, "status"].eq("ok").all()
            and set(group.loc[group.method == m, "repeat"]) == {0, 1, 2}
            for m in ("baseline", "candidate")
        )
        complete = complete and group.groupby("repeat").seed.nunique().eq(1).all()
        complete = complete and group.groupby("repeat").data_hash.nunique().eq(1).all()
        complete = (
            complete and group.elapsed_sec.gt(0).all() and np.isfinite(group.elapsed_sec).all()
        )
        ratio = float(medians.get("candidate", np.inf) / medians.get("baseline", np.nan))
        results.append(
            {
                "n": int(n),
                "p": int(p),
                "ratio": ratio,
                "complete": bool(complete),
                "passed": bool(complete and np.isfinite(ratio) and ratio <= 5),
            }
        )
    expected = {(n, p) for n in (32, 128, 240) for p in (12, 72)}
    covered = {(r["n"], r["p"]) for r in results} == expected
    return {
        "passed": covered and all(r["passed"] for r in results),
        "complete_panel": covered,
        "cases": results,
        "scope": "geometric candidate with existing empirical sibling reference; statistical validity not established",
    }


def run_panel(stage, output, *, cells=None, replicates=50):
    """Write a new, resumable-by-external-shard panel; never overwrite evidence."""
    limits = {"preflight": 6, "recovery": 16, "factorial": 3}
    if (
        stage not in limits
        or isinstance(replicates, bool)
        or not isinstance(replicates, int)
        or not 1 <= replicates <= 50
    ):
        raise ValueError("Expected a development stage and 1 through 50 replicates")
    if cells is not None and (
        not cells
        or len(set(cells)) != len(cells)
        or any(
            isinstance(c, bool) or not isinstance(c, int) or not 0 <= c < limits[stage]
            for c in cells
        )
    ):
        raise ValueError("Expected distinct cell IDs within the panel")
    output = Path(output)
    output.mkdir(parents=True, exist_ok=True)
    if (output / "manifest.json").exists():
        raise FileExistsError(f"Refusing to overwrite {output}")
    before = source_hashes()
    _json(
        output / "manifest.json",
        {
            "stage": stage,
            "cells": cells,
            "replicates": 3 if stage == "preflight" else replicates,
            "source_hashes": before,
            "seed_key": [20261109],
            "phase": "development",
        },
    )
    rows = []
    with (output / "runs.jsonl").open("x") as stream:

        def record(base, values, truth, params, trace=False):
            row, detail = _safe_run(values, truth, trace=trace, **params)
            row = {
                **base,
                **row,
                "data_hash": hashlib.sha256(
                    np.ascontiguousarray(values, dtype=np.float64).tobytes()
                ).hexdigest(),
                "truth_hash": hashlib.sha256(
                    np.ascontiguousarray(truth, dtype=np.int64).tobytes()
                ).hexdigest(),
            }
            rows.append(row)
            stream.write(json.dumps(row, default=diagnostic_json_default) + "\n")
            stream.flush()
            if trace and detail:
                _json(output / f"trace_{base['run_id']}.json", detail)
            return row

        if stage == "preflight":
            for cell, (n, p) in enumerate(itertools.product((32, 128, 240), (12, 72))):
                if cells is not None and cell not in cells:
                    continue
                recovery_cell = RECOVERY_CELLS.index((p, False, 5.0, False))
                x, truth, seed = recovery_data(recovery_cell, 0, n=n)
                for repeat in range(3):
                    for method in (
                        ("baseline", "candidate") if repeat % 2 == 0 else ("candidate", "baseline")
                    ):
                        record(
                            dict(
                                run_id=f"preflight_{cell}_{repeat}_{method}",
                                cell=cell,
                                n=n,
                                p=p,
                                seed=seed,
                                repeat=repeat,
                                method=method,
                            ),
                            x,
                            truth,
                            {"candidate": method == "candidate"},
                        )
                print(f"preflight cell {cell} complete", flush=True)
            _json(output / "runtime_gate.json", runtime_gate(rows))
        elif stage == "recovery":
            for cell in range(len(RECOVERY_CELLS)) if cells is None else cells:
                for rep in range(replicates):
                    x, truth, seed = recovery_data(cell, rep)
                    profiles = build_statistical_profiles(x, true_labels=truth)
                    if rep == 0:
                        profiles.groups.to_csv(output / f"profiles_{cell}.csv", index=False)
                    for method, operator, link in (
                        ("symmetric_average", "symmetric", "average"),
                        ("markov_ward", "markov", "ward"),
                    ):
                        record(
                            dict(
                                run_id=f"recovery_{cell}_{rep}_{method}",
                                cell=cell,
                                replicate=rep,
                                seed=seed,
                                n=len(x),
                                p=x.shape[1],
                                method=method,
                            ),
                            x,
                            truth,
                            {"operator": operator, "linkage_method": link},
                            trace=rep == 0,
                        )
                print(f"recovery cell {cell} complete", flush=True)
        elif stage == "factorial":
            factors = benchmark_grid(
                benchmark_class="subtree_research",
                grid_name="operator_linkage_branch_rank",
                base_params={},
                axes=FACTORIAL_AXES,
            )
            for cell in range(3) if cells is None else cells:
                for rep in range(replicates):
                    if cell == 0:
                        seed = data_seed(phase=1, panel=3, cell=cell, replicate=rep)
                        x = np.random.default_rng(seed).standard_normal((64, 12))
                        truth = np.zeros(64, dtype=int)
                    else:
                        c = RECOVERY_CELLS.index((12, cell == 2, 5.0, False))
                        x, truth, seed = recovery_data(c, rep)
                    for index, params in enumerate(factors):
                        selected = {key: params[key] for key in FACTORIAL_AXES}
                        record(
                            dict(
                                run_id=f"factorial_{cell}_{rep}_{index}",
                                cell=cell,
                                replicate=rep,
                                seed=seed,
                                n=len(x),
                                p=x.shape[1],
                                method=params["benchmark_run_id"],
                            ),
                            x,
                            truth,
                            selected,
                            trace=rep == 0,
                        )
                print(f"factorial cell {cell} complete", flush=True)
        else:
            raise ValueError("Unknown development stage")
    frame = pd.DataFrame(rows)
    frame.drop(columns=["labels"], errors="ignore").to_csv(output / "runs.csv", index=False)
    after = source_hashes()
    changed = sorted(p for p in set(before) | set(after) if before.get(p) != after.get(p))
    _json(
        output / "verification.json",
        {
            "attempted": len(rows),
            "source_files_changed_during_run": changed,
            "status_counts": frame.status.value_counts().to_dict(),
        },
    )
    return rows


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("stage", choices=("preflight", "recovery", "factorial"))
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--cells", type=int, nargs="+")
    parser.add_argument("--replicates", type=int, default=50)
    args = parser.parse_args()
    if not 1 <= args.replicates <= 50:
        parser.error(
            "Development replicates must be 1 through 50; validation has a separate seed phase"
        )
    run_panel(args.stage, args.output, cells=args.cells, replicates=args.replicates)


if __name__ == "__main__":
    main()

def topology_from_linkage(matrix: np.ndarray) -> Tree:
    """Read the unranked topology from a SciPy linkage over local row positions."""
    matrix = np.asarray(matrix, dtype=float)
    if matrix.ndim != 2 or matrix.shape[1] != 4 or not np.isfinite(matrix).all():
        raise ValueError("Linkage must be a finite matrix with four columns")
    integral = matrix[:, [0, 1, 3]]
    if np.any(integral != np.floor(integral)):
        raise ValueError("Linkage child IDs and leaf counts must be integers")
    is_valid_linkage(matrix, throw=True)
    nodes: list[Tree] = list(range(len(matrix) + 1))
    smallest = list(range(len(nodes)))
    counts = [1] * len(nodes)
    heights = [0.0] * len(nodes)
    for (a, b), (_, _, height, count) in zip(matrix[:, :2].astype(int), matrix, strict=True):
        if count != counts[a] + counts[b]:
            raise ValueError("Linkage leaf count must equal the sum of its child counts")
        if height < max(heights[a], heights[b]):
            raise ValueError("Linkage merge height cannot be below a child height")
        if smallest[b] < smallest[a]:
            a, b = b, a
        nodes.append((nodes[a], nodes[b]))
        smallest.append(smallest[a])
        counts.append(int(count))
        heights.append(height)
    return nodes[-1]


def _relabel(tree: Tree, labels: np.ndarray) -> Tree:
    if isinstance(tree, int):
        return int(labels[tree])
    return (_relabel(tree[0], labels), _relabel(tree[1], labels))


def _local_ward(coordinates: np.ndarray) -> np.ndarray:
    scale = float(np.max(np.abs(coordinates)))
    distances = pdist(coordinates / scale)
    maximum = float(distances.max())
    z = linkage(np.round(distances / maximum, decimals=12), method="ward")
    z[:, 2] *= maximum * scale
    if not np.isfinite(z).all():
        raise ValueError("Ward heights must fit in finite float64")
    return z


def _relabel_linkage(matrix: np.ndarray, order: np.ndarray) -> np.ndarray:
    z = matrix.copy()
    children = z[:, :2].astype(int)
    is_leaf = children < len(order)
    z[:, :2][is_leaf] = order[children[is_leaf]]
    return z


def _fit_from_split_gram(
    splits: frozenset[Cut], normalized_target: np.ndarray, scale: float
) -> AdditiveDistanceFit:
    """Fit the exact all-pair NNLS objective without a pair-by-edge matrix."""
    ordered = tuple(sorted(splits, key=lambda split: split.left))
    n = len(normalized_target)
    member = np.column_stack([np.isin(np.arange(n), split.left) for split in ordered]).astype(float)
    inside = member.sum(axis=0)
    both = member.T @ member
    neither = n - inside[:, None] - inside[None, :] + both
    only_left = inside[:, None] - both
    gram = both * neither + only_left * only_left.T
    rhs = np.sum(member * (normalized_target @ (1 - member)), axis=0)
    # Unique splits eliminate the duplicate root columns. Distinct splits of a
    # binary tree have linearly independent pair-incidence columns.
    factor = cholesky(gram, lower=False)
    transformed = solve_triangular(factor, rhs, trans="T")
    lengths, _ = nnls(factor, transformed)
    weighted_member = member * lengths
    totals = weighted_member.sum(axis=1)
    fitted = totals[:, None] + totals[None, :] - 2 * (weighted_member @ member.T)
    left, right = np.triu_indices(n, k=1)
    fitted = np.maximum(0.0, fitted[left, right])
    residual = fitted - normalized_target[left, right]
    with np.errstate(over="raise", invalid="raise"):
        return AdditiveDistanceFit(
            ordered,
            lengths * scale,
            fitted * scale,
            float(np.sqrt(np.mean(residual**2)) * scale),
            float(np.max(np.abs(residual)) * scale),
        )


def select_subtree_topology(
    points: np.ndarray,
    *,
    current_topology: Tree,
    target_metric: str = "euclidean",
    leaf_labels=None,
    exhaustive: bool = False,
    current_linkage: np.ndarray | None = None,
) -> SubtreeTopologySelection:
    """Rank current and spectral Ward topologies on one shared full-subtree target.

    Search is bounded to 256 leaves and uses every pair. `exhaustive=True` adds
    all rooted binary trees only through six leaves. Supply unique sortable
    leaf labels to keep Ward ties and residual tie-breaking stable on row
    reorder. Returned topologies and linkages use the caller's local row order.
    Original linkage heights are retained for the current tree if supplied.
    Exhaustive-only candidates have topology heights, not fitted branch times.
    """
    start = perf_counter()
    x = np.asarray(points, dtype=float)
    if (
        x.ndim != 2
        or not 2 <= len(x) <= MAX_CANDIDATE_LEAVES
        or x.shape[1] < 1
        or not np.isfinite(x).all()
    ):
        raise ValueError(f"Expected finite coordinates for 2 through {MAX_CANDIDATE_LEAVES} leaves")
    n = len(x)
    if exhaustive and n > MAX_TOPOLOGY_LEAVES:
        raise ValueError(
            f"Exhaustive topology enumeration requires at most {MAX_TOPOLOGY_LEAVES} leaves"
        )
    pending = [current_topology]
    seen = set()
    while pending:
        node = pending.pop()
        if isinstance(node, tuple) and len(node) == 2:
            pending.extend(node)
        elif type(node) is int and 0 <= node < n and node not in seen:
            seen.add(node)
        else:
            raise ValueError("Current topology must contain each local leaf exactly once")
    if len(seen) != n:
        raise ValueError("Current topology must contain each local leaf exactly once")
    topology_matrix = topology_linkage(current_topology)
    current = topology_from_linkage(topology_matrix)
    z_current = (
        topology_matrix if current_linkage is None else np.asarray(current_linkage, dtype=float)
    )
    if topology_from_linkage(z_current) != current:
        raise ValueError("Current linkage must encode current_topology")

    labels = tuple(range(n)) if leaf_labels is None else tuple(leaf_labels)
    if len(labels) != n or len(set(labels)) != n:
        raise ValueError("leaf_labels must contain one unique sortable label per row")
    order = np.asarray(sorted(range(n), key=labels.__getitem__), dtype=np.intp)
    inverse = np.argsort(order)
    canonical_current = topology_from_linkage(topology_linkage(_relabel(current, inverse)))

    if target_metric == "euclidean":
        target = pdist(x)
    elif target_metric == BRANCH_LENGTH_TARGET_MEAN_SQUARED_STANDARDIZED_EUCLIDEAN:
        left, right = np.triu_indices(n, k=1)
        target = _mean_squared_standardized_targets(pd.DataFrame(x), left, right)
    else:
        raise ValueError(f"Unsupported subtree target metric: {target_metric!r}")
    if not np.isfinite(target).all():
        raise ValueError("Full-subtree distance target must be finite")
    scale = float(target.max()) or 1.0
    normalized_target = squareform(target / scale)

    sources = {canonical_current: ["current"]}
    linkages = {canonical_current: z_current}
    span_count = 0
    for group, _, coordinates in _spectral_coordinate_spans(x[order]):
        z = _local_ward(coordinates)
        tree = topology_from_linkage(z)
        sources.setdefault(tree, []).append(f"spectral:{group}")
        if tree not in linkages:
            linkages[tree] = _relabel_linkage(z, order)
        span_count += 1
    if exhaustive:
        for tree in rooted_binary_topologies(n):
            sources.setdefault(tree, []).append("exhaustive")

    trees = {key: topology_from_linkage(topology_linkage(_relabel(key, order))) for key in sources}
    by_splits = {}
    fits = {}
    for key, tree in trees.items():
        splits = tree_edge_splits(tree)
        if splits not in by_splits:
            by_splits[splits] = _fit_from_split_gram(splits, normalized_target, scale)
        fits[key] = by_splits[splits]

    tolerance = FIT_TIE_RTOL * float(target.max())
    best = min(fit.rmse for fit in fits.values())
    tied = [key for key, fit in fits.items() if fit.rmse <= best + tolerance]
    selected = canonical_current if canonical_current in tied else min(tied, key=repr)
    selected_fit = fits[selected]
    candidates = []
    for key in sorted(fits, key=repr):
        fit = fits[key]
        zero_internal = sum(
            len(split.left) > 1 and len(split.right) > 1 and length <= tolerance
            for split, length in zip(fit.edge_splits, fit.branch_lengths, strict=True)
        )
        same_metric = bool(
            np.max(np.abs(fit.fitted_distances - selected_fit.fitted_distances)) <= tolerance
        )
        candidates.append(
            CandidateFit(trees[key], tuple(sources[key]), fit, zero_internal, same_metric)
        )

    selected_linkage = linkages.get(selected)
    if selected_linkage is None:
        selected_linkage = topology_linkage(trees[selected])
    return SubtreeTopologySelection(
        selected_topology=trees[selected],
        selected_linkage=selected_linkage,
        current_topology=current,
        candidates=tuple(candidates),
        candidate_count=len(candidates),
        spectral_span_count=span_count,
        coverage="exhaustive_rooted" if exhaustive else "bounded_spectral_ward",
        pair_count=len(target),
        target_metric=target_metric,
        elapsed_sec=perf_counter() - start,
    )

def native_sibling_attribution(values, subtree_indices, *, oracle_covariance=None, gram_rank=None):
    """Refit native components, then vary one stated part of the raw score.

    Unit branch holds the fitted tree, covariance and projection context fixed.
    Oracle covariance uses the supplied externally fixed positive-definite
    matrix in the same native context. Fixed Gram rank replaces the native
    projection by the leading raw subtree Gram directions at a prespecified
    rank while retaining the guarded covariance and branch factor. These are
    score controls; neither projected variant implies an analytic null law.
    """
    data, space, fitted, dimensions, spectral, record = _native_fit_record(values, subtree_indices)
    tree = fitted.extra["tree"]
    projection = resolve_parent_projection_inputs_for_sibling_test(
        record.parent,
        sibling_projection_dimensions_from_edge_comparisons=dimensions,
        parent_principal_component_projections=spectral.principal_component_projections_by_node,
        parent_principal_component_eigenvalues=spectral.principal_component_eigenvalues_by_node,
    )
    contrast = extract_sibling_pair_observations(tree, record.parent, record.left, record.right)[:4]
    guarded = resolve_node_continuous_covariance_by_block(tree, record.parent, space)
    mean = compute_mean_branch_length(tree)
    g = 1.0 if mean is None else 1.0 + record.branch_length_sum / (2 * mean)
    annotations = fitted.extra["annotations"]
    annotated = annotations.loc[record.parent]
    inflation = float(annotated.get("Sibling_Inflation_Factor", np.nan))
    has_display = np.isfinite(inflation) and inflation > 0

    def raw(*, branch=record.branch_length_sum, covariance=guarded, context=projection, adapt=0.90):
        k, basis, eig = context
        return sibling_divergence_test(
            *contrast,
            branch_length_sum=branch,
            mean_branch_length=mean,
            feature_space=space,
            projection_dimension_from_edge_comparisons=k,
            parent_principal_component_projection=basis,
            parent_principal_component_eigenvalues=eig,
            continuous_covariance_by_block=covariance,
            adaptive_projection_dimension_energy_fraction=adapt,
        )

    unit, _, unit_df, _ = raw(branch=0.0)
    result = dict(
        native_raw=float(record.stat),
        native_df=float(record.degrees_of_freedom),
        native_display=float(record.stat / (record.reference_scale * inflation))
        if has_display
        else None,
        annotated_display=float(annotated["Sibling_Test_Statistic"]) if has_display else None,
        scale=float(record.reference_scale),
        inflation=inflation if has_display else None,
        unit_branch_raw=float(unit),
        unit_branch_df=float(unit_df),
        branch_factor=float(g),
        sibling_branch_sum=float(record.branch_length_sum),
        positive_branch_mean=mean,
        child_sizes=[int(contrast[2]), int(contrast[3])],
    )
    if oracle_covariance is not None:
        oracle = np.asarray(oracle_covariance, dtype=float)
        p = data.shape[1]
        if (
            oracle.shape != (p, p)
            or not np.isfinite(oracle).all()
            or not np.allclose(oracle, oracle.T)
        ):
            raise ValueError("oracle_covariance must be a symmetric feature covariance matrix")
        np.linalg.cholesky(oracle)
        name = space.continuous_blocks[0].name
        oracle_raw, _, oracle_df, _ = raw(covariance={name: oracle})
        result.update(
            oracle_covariance_raw=float(oracle_raw), oracle_covariance_df=float(oracle_df)
        )
    if gram_rank is not None:
        if isinstance(gram_rank, bool) or not isinstance(gram_rank, Integral) or gram_rank < 1:
            raise ValueError("gram_rank must be a positive integer")
        clades = tree.compute_descendant_sets()
        x = data.loc[sorted(clades[record.parent])].to_numpy()
        _, singular, basis = np.linalg.svd(x - x.mean(axis=0), full_matrices=False)
        eigenvalues = singular**2
        available = int(
            np.count_nonzero(singular > singular[0] * max(x.shape) * np.finfo(float).eps)
        )
        if gram_rank > available or (
            gram_rank < available and tied_eigenvalue_neighbors(eigenvalues)[gram_rank - 1]
        ):
            raise ValueError("gram_rank exceeds the positive rank or cuts a tied eigenspace")
        gram_raw, _, gram_df, _ = raw(
            context=(gram_rank, basis[:gram_rank], eigenvalues[:gram_rank]), adapt=None
        )
        result.update(gram_rank_raw=float(gram_raw), gram_rank_df=float(gram_df))
    return result

def _fixed_split_score(matrix, rows, left, right):
    # Under two-row Gram conditioning the only orientations swap the rows.
    if len(rows) == 2:
        return 0.0
    count = len(left)
    cut = Cut(tuple(range(count)), tuple(range(count, len(rows))))
    return float(sibling_cut_scores(matrix[np.r_[left, right]], (cut,))[0])


def ward_subtree_family_test(values, *, path=(), score=None, steps=999, seed=0, alpha=0.05):
    """Test every internal node under one shared Ward selection event.

    The predetermined path selects the subtree. The common event preserves
    its ancestral child pairs and the whole labeled topology inside it. For
    each node, freeze all outside rows and rotate only that node. The true
    null at a tested node requires independent rows with one common Gaussian
    mean and covariance, independent of outside rows. A custom deterministic
    score receives (matrix, node rows, left rows, right rows); all row arrays
    use input IDs. It must be prespecified or determined by the shared event.
    Holm controls false homogeneity rejections under that
    model and event, including configurations with false nulls.
    """
    path = _validated_ward_path(path)
    if score is not None and not callable(score):
        raise ValueError("score must be callable")
    score = _fixed_split_score if score is None else score
    if isinstance(seed, bool) or not isinstance(seed, Integral) or seed < 0:
        raise ValueError("seed must be a nonnegative integer")
    try:
        observed = _ward_description(values, path, keep_subtree=True)
    except (ValueError, RuntimeError, FloatingPointError, np.linalg.LinAlgError) as error:
        return dict(
            status="unresolved",
            decision="unresolved",
            nodes=[],
            error_phase="observation",
            error=f"{type(error).__name__}: {error}",
        )
    if observed is None:
        return dict(
            status="unresolved",
            decision="unresolved",
            nodes=[],
            error="The specified path must reach an internal node",
        )
    subtree, _, family = observed

    def event(matrix):
        return _ward_description(matrix, path, keep_subtree=True) == observed

    nodes = []
    for i, (node, left, right) in enumerate(family):
        rows, a, b = map(np.asarray, (node, left, right))
        for indices in (rows, a, b):
            indices.setflags(write=False)
        result = subtree_gaussian_chain_test(
            values,
            subtree_indices=rows,
            score=lambda x: score(x, rows, a, b),
            selection_event=event,
            steps=steps,
            seed=seed + i,
            alpha=alpha,
        )
        result.update(left=list(left), right=list(right))
        nodes.append(result)

    m = len(nodes)
    resolution = 1 / (steps + 1) <= alpha / m
    order = sorted(
        range(m), key=lambda j: nodes[j]["p_value"] if nodes[j]["decision"] != "unresolved" else 1
    )
    running = 0.0
    for i, j in enumerate(order):
        node = nodes[j]
        p_value = node["p_value"] if node["decision"] != "unresolved" else 1.0
        running = max(running, min(1.0, (m - i) * p_value))
        node["holm_p_value"] = running
        node["family_decision"] = (
            "unresolved"
            if not resolution or node["decision"] == "unresolved"
            else "reject"
            if running <= alpha
            else "do_not_reject"
        )
    decisions = [node["family_decision"] for node in nodes]
    return dict(
        status="complete",
        decision="reject"
        if "reject" in decisions
        else "unresolved"
        if "unresolved" in decisions
        else "do_not_reject",
        unresolved_nodes=decisions.count("unresolved"),
        subtree_indices=list(subtree),
        hypotheses=m,
        nodes=nodes,
        path=list(path),
        selection_event="common_ward_ancestry_and_subtree_topology",
        alpha=float(alpha),
        resolution_sufficient=resolution,
        holm_first_cutoff=alpha / m,
        minimum_p_value=1 / (steps + 1),
        minimum_steps_for_holm=math.ceil(m / alpha) - 1,
    )

def _frame(values):

def sibling_cut_scores(values, cuts):
    """Existing full-space raw sibling Wald scores, zero branches, guarded covariance.

    This matches the earlier full-space sibling diagnostic, including parent
    covariance fallback. It excludes empirical inflation and adaptive rank.
    """
    data = _frame(values)
    x = data.to_numpy()
    space = continuous_feature_space_from_columns(data.columns)
    tree = build_tree(data, pdist(x), linkage_method="ward").tree
    tree.populate_node_distributions(data, feature_space=space)
    p = x.shape[1]
    scores = []
    for cut in cuts:
        if not cut.left or not cut.right or sorted((*cut.left, *cut.right)) != list(range(len(x))):
            raise ValueError("Each cut must partition the local row indices exactly once")
        covariance = pooled_within_child_covariance_by_block(
            tree, tree.root(), space, x, [list(cut.left), list(cut.right)]
        )
        statistic, _, _, _ = sibling_divergence_test(
            x[list(cut.left)].mean(0),
            x[list(cut.right)].mean(0),
            len(cut.left),
            len(cut.right),
            branch_length_sum=0.0,
            mean_branch_length=None,
            projection_dimension_from_edge_comparisons=p,
            parent_principal_component_projection=np.eye(p),
            parent_principal_component_eigenvalues=np.ones(p),
            feature_space=space,
            continuous_covariance_by_block=covariance,
        )
        if not np.isfinite(statistic):
            raise ValueError("Nonfinite sibling statistic")
        scores.append(statistic)
    return np.asarray(scores, dtype=float)


def _candidate_scores(points):
    spaces = spectral_subspaces(points)
    if not spaces:
        return (), np.array([], dtype=float)
    # Every smaller-span affine cut lies in the full cumulative span.
    coordinates = max(spaces, key=lambda space: space.rank).coordinates
    coordinates = coordinates / np.max(np.abs(coordinates), axis=0)
    cuts = tuple(
        cut
        for cut in unordered_cuts(len(points))
        if is_separable(coordinates, np.isin(np.arange(len(points)), cut.left))
    )
    return cuts, sibling_cut_scores(points, cuts)


def subtree_alternative_cut_test(
    values, *, subtree_indices, selection_event=None, steps=999, seed=0, alpha=0.05
):
    """Replay all full-span affine candidates and their maximum sibling score.

    Returned cut ranks share the maximum reference and give simultaneous control
    only under the complete common-mean subtree null. Membership is prespecified
    unless the supplied event reproduces its selection. No observed candidate
    count is substituted into a fixed-label Beta tail.
    """
    x, rows = _inputs(values, subtree_indices, alpha)
    observed = {}

    def score(matrix):
        cuts, scores = _candidate_scores(matrix[rows])
        if not observed:
            observed.update(cuts=cuts, scores=scores)
        return float(np.max(scores, initial=0.0))

    result = subtree_gaussian_chain_test(
        x,
        subtree_indices=rows,
        score=score,
        selection_event=selection_event,
        steps=steps,
        seed=seed,
        alpha=alpha,
    )
    result["candidate_family"] = "full_span_strict_affine_cuts"
    result["score"] = "full_space_raw_sibling_zero_branches"
    result["cut_results"] = [
        dict(
            left=rows[list(cut.left)].tolist(),
            right=rows[list(cut.right)].tolist(),
            statistic=float(statistic),
            global_null_adjusted_p_value=(
                sum(value >= statistic for value in result["chain_scores"]) / (steps + 1)
                if result["status"] == "complete"
                else None
            ),
        )
        for cut, statistic in zip(observed.get("cuts", ()), observed.get("scores", ()), strict=True)
    ]
    return result


def ward_subtree_selection(values, *, path=(), keep_split=False):
    """Capture a prespecified Ward node path, ordered by the smallest leaf ID.

    Return (row_indices, event). The event preserves every ancestral child pair;
    keep_split also preserves the tested node's own pair. The path must be chosen
    before observing the data. It does not encode an additional search for a node
    or a TBS gate/traversal decision. All outside rows stay fixed in the sampler.
    """
    path = _validated_ward_path(path)
    if not isinstance(keep_split, (bool, np.bool_)):
        raise ValueError("keep_split must be boolean")
    observed = _ward_description(values, path, keep_split=keep_split)
    if observed is None:
        raise ValueError("The specified path must reach an internal node")
    rows, _, _ = observed
    return np.asarray(
        rows
    ), lambda matrix: _ward_description(matrix, path, keep_split=keep_split) == observed


def _validated_ward_path(path):
    path = tuple(path)
    if any(
        isinstance(side, bool) or not isinstance(side, Integral) or side not in (0, 1)
        for side in path
    ):
        raise ValueError("path must contain only child positions 0 or 1")
    return path


def _ward_description(values, path, *, keep_split=False, keep_subtree=False):
    x = np.asarray(values, dtype=float)
    if x.ndim != 2 or len(x) < 2 or x.shape[1] < 1 or not np.isfinite(x).all():
        raise ValueError("Expected at least two finite rows and one feature")
    # Native build_tree calls this linkage on the same canonical positional IDs.
    rows = [(i,) for i in range(len(x))]
    children_by_node = [()] * len(x)
    for i, j in linkage(pdist(x), method="ward")[:, :2].astype(int):
        pair = tuple(sorted((i, j), key=rows.__getitem__))
        rows.append(tuple(sorted(rows[i] + rows[j])))
        children_by_node.append(pair)
    node, ancestors = len(rows) - 1, []
    for depth in range(len(path) + 1):
        pair = children_by_node[node]
        if len(pair) != 2:
            return None
        if depth < len(path) or keep_split:
            ancestors.append(tuple(rows[child] for child in pair))
        if depth < len(path):
            node = pair[path[depth]]

    family = []
    if keep_subtree:
        pending = [node]
        while pending:
            parent = pending.pop()
            pair = children_by_node[parent]
            if pair:
                if len(pair) != 2:
                    raise ValueError("Expected a binary Ward subtree")
                family.append((rows[parent], *(rows[child] for child in pair)))
                pending.extend(pair)
    return rows[node], tuple(ancestors), tuple(family)


def native_sibling_statistic(values, subtree_indices):
    """Refit Ward, NNLS, edge spectral context and native adaptive raw sibling score.

    This retains the native raw statistic, including branch time and .90
    contrast-energy rank. It does not reuse the native chi-square/inflated p-value,
    impose its eligibility gate, or claim control of the final partition.
    """
    _, _, _, _, _, record = _native_fit_record(values, subtree_indices)
    return float(record.stat)


def _native_fit_record(values, subtree_indices):
    data = _frame(values)
    _, rows = _inputs(values, subtree_indices, 0.05)
    space = continuous_feature_space_from_columns(data.columns)
    result = run_tbs_on_distance(
        data,
        pdist(data.to_numpy()),
        0.05,
        tree_linkage_method="ward",
        feature_space=space,
        adaptive_projection_dimension_energy_fraction=0.90,
        branch_length_optimization_method="fixed_topology_nnls",
        branch_length_data_df=data,
    )
    if result.status != "ok":
        raise ValueError(f"Native TBS replay failed: {result.status}")
    tree = result.extra["tree"]
    edge = result.extra["gate_bundle"].edge_gate_result
    spectral = edge.spectral_context
    dimensions = derive_sibling_projection_dimensions_from_child_edge_comparisons(
        tree, spectral_context=spectral
    )
    records, _ = collect_sibling_pair_records(
        tree,
        edge.annotated_df,
        sibling_projection_dimensions_from_edge_comparisons=dimensions,
        parent_principal_component_projections=spectral.principal_component_projections_by_node,
        parent_principal_component_eigenvalues=spectral.principal_component_eigenvalues_by_node,
        feature_space=space,
        adaptive_projection_dimension_energy_fraction=0.90,
    )
    target = frozenset(data.index[rows])
    clades = tree.compute_descendant_sets()
    record = next((record for record in records if clades[record.parent] == target), None)
    if record is None or not np.isfinite(record.stat):
        raise ValueError("Requested subtree has no finite native raw sibling score")
    return data, space, result, dimensions, spectral, record


def topology_sibling_score(values, topology: Tree):
    """Prespecified topology score: sum of raw full-space sibling scores at its nodes."""
    x = _frame(values).to_numpy()

    def leaves(node):
        if isinstance(node, int):
            return [node]
        if not isinstance(node, tuple) or len(node) != 2:
            raise ValueError("Expected a binary nested-tuple topology")
        return leaves(node[0]) + leaves(node[1])

    if sorted(leaves(topology)) != list(range(len(x))):
        raise ValueError("Topology must contain each local row index exactly once")

    def score(node):
        if isinstance(node, int):
            return 0.0
        left, right = leaves(node[0]), leaves(node[1])
        cut = Cut(tuple(range(len(left))), tuple(range(len(left), len(left) + len(right))))
        local = sibling_cut_scores(x[left + right], (cut,))[0]
        return float(local + score(node[0]) + score(node[1]))

    return score(topology)


def topology_pair_reference(
    values,
    *,
    subtree_indices,
    first: Tree,
    second: Tree,
    selection_event=None,
    steps=999,
    seed=0,
    alpha=0.05,
):
    """Paired absolute score difference for two prespecified labeled topologies.

    A data-selected pair needs its pair-selection event as well. The null is
    homogeneous Gaussian rows, not equal expected accuracy of the two trees.
    """
    x, rows = _inputs(values, subtree_indices, alpha)

    def score(matrix):
        return abs(
            topology_sibling_score(matrix[rows], first)
            - topology_sibling_score(matrix[rows], second)
        )

    result = subtree_gaussian_chain_test(
        x,
        subtree_indices=rows,
        score=score,
        selection_event=selection_event,
        steps=steps,
        seed=seed,
        alpha=alpha,
    )
    return result | {"score": "absolute_difference_of_summed_full_space_sibling_scores"}


def subtree_topology_search_test(
    values, *, subtree_indices, selection_event=None, steps=999, seed=0, alpha=0.05
):
    """Replay the union of tied-Ward families over whole spectral spans (at most six rows).

    The geometry rule fixes the relative tie tolerance at 1e-10. Counts are
    descriptive; the maximum is recalculated over the new family on every draw.
    Scores sum raw full-space sibling scores, using the same feature space for
    every candidate. They are a declared diagnostic criterion, not a likelihood.
    """
    x, rows = _inputs(values, subtree_indices, alpha)
    if len(rows) > 6:
        raise ValueError("Tied-Ward topology search requires at most six subtree leaves")

    def candidates(matrix):
        points = matrix[rows]
        trees = set()
        for space in spectral_subspaces(points):
            trees.update(ward_topology_family(space.coordinates, rtol=1e-10))
        ordered = sorted(trees, key=repr)
        return ordered, [topology_sibling_score(points, tree) for tree in ordered]

    observed = {}

    def score(matrix):
        trees, scores = candidates(matrix)
        if not observed:
            observed.update(trees=trees, scores=scores)
        return max(scores, default=0.0)

    result = subtree_gaussian_chain_test(
        x,
        subtree_indices=rows,
        score=score,
        selection_event=selection_event,
        steps=steps,
        seed=seed,
        alpha=alpha,
    )
    result.update(candidate_family="union_of_spectral_tied_ward_topologies", tie_rtol=1e-10)
    result["topology_results"] = [
        dict(
            topology=tree,
            statistic=statistic,
            global_null_adjusted_p_value=(
                sum(value >= statistic for value in result["chain_scores"]) / (steps + 1)
                if result["status"] == "complete"
                else None
            ),
        )
        for tree, statistic in zip(
            observed.get("trees", ()), observed.get("scores", ()), strict=True
        )
    ]
    return result

def nan_clustering_metrics() -> ClusteringMetrics:
    """Return the all-NaN metrics recorded for runs without usable labels."""
    return ClusteringMetrics(**{field.name: np.nan for field in fields(ClusteringMetrics)})


def _unit_interval_score(value: float) -> float:
    """Normalize bounded scores against tiny floating-point overshoots."""
    score = float(value)
    if not np.isfinite(score):
        return score
    return float(np.clip(score, 0.0, 1.0))


def _is_noise_label(value: object) -> bool:
    if pd.isna(value):
        return False
    try:
        return bool(float(value) == -1.0)
    except (TypeError, ValueError):
        return str(value).strip() == "-1"


def _effective_cluster_count_from_sizes(sizes: pd.Series) -> tuple[float, float]:
    if sizes.empty or float(sizes.sum()) <= 0.0:
        return np.nan, np.nan
    size_entropy = float(entropy(sizes.to_numpy(dtype=float)))
    return float(np.exp(size_entropy)), size_entropy


def _cluster_size_gini(sizes: pd.Series) -> float:
    values = np.sort(sizes.to_numpy(dtype=float))
    if values.size == 0 or np.isclose(values.sum(), 0.0):
        return np.nan
    index = np.arange(1, values.size + 1)
    numerator = ((2.0 * index - values.size - 1.0) * values).sum()
    denominator = values.size * values.sum()
    return float(numerator / denominator)


def _cluster_purity(y_true: np.ndarray, y_pred: np.ndarray) -> float:
    """Return majority-label purity across predicted clusters."""
    matrix = contingency_matrix(y_true, y_pred)
    if matrix.size == 0 or matrix.sum() <= 0:
        return np.nan
    return float(np.max(matrix, axis=0).sum() / matrix.sum())


def _calculate_fragmentation_metrics(
    pred_cluster: pd.Series,
) -> tuple[float, float, float, float, float, float, float, float]:
    """Return cluster-size diagnostics excluding DBSCAN-style noise label -1."""
    if pred_cluster.empty or pred_cluster.isna().any():
        return (np.nan,) * 8

    noise_mask = pred_cluster.map(_is_noise_label).to_numpy(dtype=bool)
    noise_label_fraction = float(noise_mask.mean()) if noise_mask.size else np.nan
    non_noise = pred_cluster.loc[~noise_mask]
    if non_noise.empty:
        return (
            0.0,
            0.0,
            np.nan,
            np.nan,
            np.nan,
            np.nan,
            np.nan,
            noise_label_fraction,
        )

    sizes = non_noise.value_counts()
    n_samples = float(len(pred_cluster))
    effective_count, size_entropy = _effective_cluster_count_from_sizes(sizes)
    return (
        float((sizes == 1).sum()),
        float(sizes[sizes == 1].sum() / n_samples),
        float(sizes.median()),
        float(sizes.max() / n_samples),
        effective_count,
        size_entropy,
        _cluster_size_gini(sizes),
        noise_label_fraction,
    )


def _calculate_internal_geometry_metrics(
    feature_matrix: np.ndarray | pd.DataFrame | None,
    y_pred: np.ndarray,
) -> tuple[float, float, float]:
    """Return silhouette, Davies-Bouldin, and Calinski-Harabasz when defined."""
    if feature_matrix is None:
        return np.nan, np.nan, np.nan
    values = np.asarray(feature_matrix, dtype=float)
    labels = np.asarray(y_pred)
    if values.ndim != 2 or values.shape[0] != labels.shape[0] or values.shape[0] == 0:
        return np.nan, np.nan, np.nan
    if not np.isfinite(values).all():
        return np.nan, np.nan, np.nan

    non_noise = np.array([not _is_noise_label(label) for label in labels], dtype=bool)
    values = values[non_noise]
    labels = labels[non_noise]
    n_samples = labels.shape[0]
    n_labels = np.unique(labels).size
    if not 1 < n_labels < n_samples:
        return np.nan, np.nan, np.nan

    silhouette = np.nan
    davies_bouldin = np.nan
    calinski_harabasz = np.nan
    try:
        sample_size = min(2000, n_samples)
        silhouette = float(
            silhouette_score(
                values,
                labels,
                sample_size=sample_size if sample_size < n_samples else None,
                random_state=0,
            )
        )
    except ValueError:
        silhouette = np.nan
    try:
        davies_bouldin = float(davies_bouldin_score(values, labels))
    except ValueError:
        davies_bouldin = np.nan
    try:
        calinski_harabasz = float(calinski_harabasz_score(values, labels))
    except ValueError:
        calinski_harabasz = np.nan
    return silhouette, davies_bouldin, calinski_harabasz


def _calculate_outlier_isolation_metrics(
    report_df: pd.DataFrame,
    sample_names: pd.Index,
    outlier_indices: list[int] | tuple[int, ...] | np.ndarray,
) -> tuple[float, float, float, float, float]:
    """Compute outlier-isolation metrics from predicted clusters.

    A predicted cluster counts as an outlier cluster when a strict majority of its
    members are true outliers. This makes the metric meaningful even when methods
    do not emit a dedicated noise label.
    """

    if len(outlier_indices) == 0:
        return np.nan, np.nan, np.nan, np.nan, np.nan

    sample_index = pd.Index(sample_names)
    if sample_index.empty:
        return np.nan, np.nan, np.nan, np.nan, np.nan

    outlier_mask_by_position = np.zeros(len(sample_index), dtype=bool)
    outlier_positions = np.asarray(outlier_indices, dtype=int)
    if np.any(outlier_positions < 0) or np.any(outlier_positions >= len(sample_index)):
        raise ValueError("outlier_indices contains positions outside sample_names")
    outlier_mask_by_position[outlier_positions] = True

    outlier_lookup = pd.Series(outlier_mask_by_position, index=sample_index)
    true_outlier = report_df.index.to_series().map(outlier_lookup)
    pred_cluster = report_df["cluster_id"]

    if true_outlier.isna().any() or pred_cluster.isna().any():
        return np.nan, np.nan, np.nan, np.nan, np.nan

    cluster_outlier_fraction = true_outlier.groupby(pred_cluster).mean()
    predicted_outlier_clusters = cluster_outlier_fraction[cluster_outlier_fraction > 0.5].index
    predicted_outlier = pred_cluster.isin(predicted_outlier_clusters).to_numpy(dtype=bool)
    true_outlier_arr = true_outlier.to_numpy(dtype=bool)

    outlier_precision = float(precision_score(true_outlier_arr, predicted_outlier, zero_division=0))
    outlier_recall = float(recall_score(true_outlier_arr, predicted_outlier, zero_division=0))
    outlier_f1 = float(f1_score(true_outlier_arr, predicted_outlier, zero_division=0))

    singleton_outlier_isolated = np.nan
    if len(outlier_positions) == 1:
        outlier_sample_name = sample_index[int(outlier_positions[0])]
        if outlier_sample_name in pred_cluster.index:
            outlier_cluster = pred_cluster.loc[outlier_sample_name]
            singleton_outlier_isolated = float((pred_cluster == outlier_cluster).sum() == 1)

    grouped_outlier_cluster_recovered = np.nan
    if len(outlier_positions) > 1:
        outlier_sample_names = sample_index[outlier_positions]
        outlier_cluster_ids = pred_cluster.loc[outlier_sample_names]
        if outlier_cluster_ids.nunique(dropna=False) == 1:
            grouped_cluster_id = outlier_cluster_ids.iloc[0]
            grouped_cluster_mask = pred_cluster == grouped_cluster_id
            grouped_outlier_cluster_recovered = float(
                true_outlier[grouped_cluster_mask].all()
                and int(grouped_cluster_mask.sum()) == len(outlier_positions)
            )
        else:
            grouped_outlier_cluster_recovered = 0.0

    return (
        outlier_precision,
        outlier_recall,
        outlier_f1,
        singleton_outlier_isolated,
        grouped_outlier_cluster_recovered,
    )


def _remap_predicted_labels_to_truth(
    y_true: np.ndarray,
    y_pred: np.ndarray,
) -> tuple[np.ndarray, np.ndarray]:
    """Map predicted cluster ids to true-label ids for per-class metrics."""
    le_true = LabelEncoder()
    le_pred = LabelEncoder()
    y_true_enc = le_true.fit_transform(y_true)
    y_pred_enc = le_pred.fit_transform(y_pred)

    cm = contingency_matrix(y_true_enc, y_pred_enc)
    if cm.size == 0:
        return y_true_enc, y_pred_enc

    row_ind, col_ind = linear_sum_assignment(-cm)  # maximize agreement
    mapping_enc = {pred: true for true, pred in zip(row_ind, col_ind)}

    # Map any extra predicted clusters to their majority true class.
    for pred in range(cm.shape[1]):
        if pred not in mapping_enc:
            mapping_enc[pred] = int(np.argmax(cm[:, pred]))

    y_pred_remapped_enc = np.asarray([mapping_enc[p] for p in y_pred_enc], dtype=int)
    return y_true_enc, y_pred_remapped_enc


def _matched_recovery_from_table(
    table: pd.DataFrame, rare_labels
) -> tuple[pd.DataFrame, float, float]:
    counts = table.to_numpy(dtype=np.int64)
    truth_n, pred_n = counts.sum(axis=1), counts.sum(axis=0)
    f1 = 2 * counts / (truth_n[:, None] + pred_n)
    rows, cols = linear_sum_assignment(f1, maximize=True)
    shared = counts[rows, cols]
    keep = shared > 0
    rows, cols, shared = rows[keep], cols[keep], shared[keep]
    matched = pd.DataFrame({"group": table.index, "truth_n": truth_n})
    matched["predicted"] = pd.Series(index=matched.index, dtype=object)
    matched[["precision", "recall", "f1"]] = 0.0
    matched.loc[rows, "predicted"] = table.columns[cols].to_numpy(dtype=object)
    matched.loc[rows, "precision"] = shared / pred_n[cols]
    matched.loc[rows, "recall"] = shared / truth_n[rows]
    matched.loc[rows, "f1"] = f1[rows, cols]
    if rare_labels is None:
        rare_labels = matched.sort_values(["truth_n", "group"]).head((len(matched) + 3) // 4).group
    matched["rare"] = matched.group.isin(rare_labels)
    matched["matched_90_90"] = (matched.precision >= 0.9) & (matched.recall >= 0.9)
    biggest = counts.argmax(axis=1)
    matched["legacy_90_90"] = (counts.max(axis=1) / truth_n >= 0.9) & (
        counts.max(axis=0)[biggest] / pred_n[biggest] >= 0.9
    )
    shared_pairs = (counts * (counts - 1)).sum()
    truth_pairs = (truth_n * (truth_n - 1)).sum()
    predicted_pairs = (pred_n * (pred_n - 1)).sum()
    split = 1 - shared_pairs / truth_pairs if truth_pairs else np.nan
    merge = 1 - shared_pairs / predicted_pairs if predicted_pairs else np.nan
    return matched, float(split), float(merge)


def matched_group_recovery(true_labels, predicted_labels, *, rare_labels=None) -> pd.DataFrame:
    """One-to-one maximum-total-F1 recovery with unmatched truth groups scored zero.

    Two Series are aligned by their unique indexes. Other inputs are aligned
    by position. Rare defaults to the smallest ceil(K/4) truth groups, with
    label order breaking size ties. Pass an empty collection for no rare groups.
    """
    if isinstance(true_labels, pd.Series) and isinstance(predicted_labels, pd.Series):
        if (
            not true_labels.index.is_unique
            or not predicted_labels.index.is_unique
            or len(true_labels) != len(predicted_labels)
            or not true_labels.index.isin(predicted_labels.index).all()
        ):
            raise ValueError("Truth and prediction Series must share a unique sample index.")
        predicted_labels = predicted_labels.reindex(true_labels.index)
    truth, predicted = np.asarray(true_labels), np.asarray(predicted_labels)
    if (
        truth.ndim != 1
        or predicted.shape != truth.shape
        or not len(truth)
        or pd.isna(truth).any()
        or pd.isna(predicted).any()
    ):
        raise ValueError(
            "Expected aligned nonempty truth and predicted labels without missing values."
        )
    return _matched_recovery_from_table(pd.crosstab(truth, predicted), rare_labels)[0]


def calculate_clustering_metrics(
    report_df: pd.DataFrame | None,
    sample_names: pd.Index,
    true_labels: np.ndarray,
    metadata: dict[str, object] | None = None,
    feature_matrix: np.ndarray | pd.DataFrame | None = None,
) -> ClusteringMetrics:
    """Calculate external, fragmentation, internal, and outlier metrics."""

    if report_df is None or report_df.empty:
        return nan_clustering_metrics()
    if "cluster_id" not in report_df.columns:
        return nan_clustering_metrics()

    sample_index = pd.Index(sample_names)
    true_arr = np.asarray(true_labels)

    # Metrics are only valid when predicted labels and truth align by sample.
    if len(sample_index) != len(true_arr):
        return nan_clustering_metrics()
    if len(report_df.index) != len(sample_index):
        return nan_clustering_metrics()
    if sample_index.has_duplicates or report_df.index.has_duplicates:
        return nan_clustering_metrics()

    true_lookup = pd.Series(true_arr, index=sample_index)
    true_cluster = report_df.index.to_series().map(true_lookup)
    pred_cluster = report_df["cluster_id"]

    if true_cluster.isna().any() or pred_cluster.isna().any():
        return nan_clustering_metrics()

    y_true = true_cluster.to_numpy()
    y_pred = pred_cluster.to_numpy()
    (
        n_singleton_clusters,
        singleton_fraction,
        median_cluster_size,
        largest_cluster_fraction,
        effective_cluster_count,
        cluster_size_entropy,
        cluster_size_gini,
        noise_label_fraction,
    ) = _calculate_fragmentation_metrics(pred_cluster)
    (
        internal_silhouette,
        davies_bouldin,
        calinski_harabasz,
    ) = _calculate_internal_geometry_metrics(feature_matrix, y_pred)
    outlier_precision = np.nan
    outlier_recall = np.nan
    outlier_f1 = np.nan
    singleton_outlier_isolated = np.nan
    grouped_outlier_cluster_recovered = np.nan
    try:
        ari = adjusted_rand_score(y_true, y_pred)
        nmi = normalized_mutual_info_score(y_true, y_pred)
        ami = adjusted_mutual_info_score(y_true, y_pred)
        purity = _cluster_purity(y_true, y_pred)
        homogeneity, completeness, v_measure = homogeneity_completeness_v_measure(
            y_true,
            y_pred,
        )
        fowlkes_mallows = fowlkes_mallows_score(y_true, y_pred)
        matched, pair_split_share, pair_merge_share = _matched_recovery_from_table(
            pd.crosstab(y_true, y_pred), None if metadata is None else metadata.get("rare_labels")
        )
        y_true_enc, y_pred_remapped_enc = _remap_predicted_labels_to_truth(y_true, y_pred)
        labels = np.arange(len(np.unique(y_true_enc)), dtype=int)
        if labels.size == 0:
            macro_recall = np.nan
            macro_f1 = np.nan
            worst_cluster_recall = np.nan
        else:
            per_class_recall = recall_score(
                y_true_enc,
                y_pred_remapped_enc,
                labels=labels,
                average=None,
                zero_division=0,
            )
            macro_recall = float(np.mean(per_class_recall))
            macro_f1 = float(
                f1_score(
                    y_true_enc,
                    y_pred_remapped_enc,
                    labels=labels,
                    average="macro",
                    zero_division=0,
                )
            )
            worst_cluster_recall = float(np.min(per_class_recall))
        outlier_indices_raw = None if metadata is None else metadata.get("outlier_indices")
        if (
            isinstance(outlier_indices_raw, (list, tuple, np.ndarray))
            and len(outlier_indices_raw) > 0
        ):
            (
                outlier_precision,
                outlier_recall,
                outlier_f1,
                singleton_outlier_isolated,
                grouped_outlier_cluster_recovered,
            ) = _calculate_outlier_isolation_metrics(
                report_df,
                sample_names,
                outlier_indices_raw,
            )
    except ValueError:
        return nan_clustering_metrics()

    return ClusteringMetrics(
        ari=float(ari),
        nmi=_unit_interval_score(nmi),
        ami=float(ami),
        purity=_unit_interval_score(purity),
        homogeneity=_unit_interval_score(homogeneity),
        completeness=_unit_interval_score(completeness),
        v_measure=_unit_interval_score(v_measure),
        fowlkes_mallows=_unit_interval_score(fowlkes_mallows),
        macro_recall=_unit_interval_score(macro_recall),
        macro_f1=_unit_interval_score(macro_f1),
        worst_cluster_recall=_unit_interval_score(worst_cluster_recall),
        n_singleton_clusters=float(n_singleton_clusters),
        singleton_fraction=_unit_interval_score(singleton_fraction),
        median_cluster_size=float(median_cluster_size),
        largest_cluster_fraction=_unit_interval_score(largest_cluster_fraction),
        effective_cluster_count=float(effective_cluster_count),
        cluster_size_entropy=float(cluster_size_entropy),
        cluster_size_gini=_unit_interval_score(cluster_size_gini),
        noise_label_fraction=_unit_interval_score(noise_label_fraction),
        silhouette_score=float(internal_silhouette),
        davies_bouldin_index=float(davies_bouldin),
        calinski_harabasz_index=float(calinski_harabasz),
        outlier_precision=float(outlier_precision),
        outlier_recall=float(outlier_recall),
        outlier_f1=float(outlier_f1),
        singleton_outlier_isolated=float(singleton_outlier_isolated),

# SOURCE: benchmarks/shared/statistical_profiles.py
"""Descriptive geometry of truth groups and explicitly supplied subtree splits."""

from __future__ import annotations

from collections.abc import Mapping, Sequence
from dataclasses import dataclass
from itertools import combinations

import numpy as np
import pandas as pd


@dataclass(frozen=True)
class StatisticalProfiles:
    """Group-level, pairwise group, and explicit subtree profiles."""

    groups: pd.DataFrame
    pairs: pd.DataFrame
    subtrees: pd.DataFrame


@dataclass(frozen=True)
class _Sample:
    points: np.ndarray
    centroid: np.ndarray
    covariance: np.ndarray


def _sample(points: np.ndarray) -> _Sample:
    centroid = points.mean(axis=0)
    residual = points - centroid
    covariance = residual.T @ residual
    if not np.isfinite(covariance).all() or not np.isfinite(centroid).all():
        raise ValueError("Centroids and covariance must be finite.")
    return _Sample(points, centroid, covariance)


def _spectrum(eigenvalues: np.ndarray) -> tuple[tuple[float, ...], int, float]:
    values = np.maximum(eigenvalues[::-1], 0)
    if values[0] == 0:
        return tuple(values), 0, np.nan
    scaled = values / values[0]
    rank = int((scaled > 1e-10).sum())
    weights = scaled[scaled > 0] / scaled.sum()
    return tuple(values), rank, float(np.exp(-np.sum(weights * np.log(weights))))


def _group_profile(sample: _Sample, total_n: int) -> dict[str, object]:
    n = len(sample.points)
    covariance = sample.covariance / (n - 1) if n > 1 else None
    spectrum, rank, effective = (
        _spectrum(np.linalg.eigvalsh(covariance))
        if covariance is not None
        else ((np.nan,) * len(sample.centroid), np.nan, np.nan)
    )
    return {
        "size": n,
        "share": n / total_n,
        "effective_n": n,
        "covariance_eigenvalues": spectrum,
        "covariance_rank": rank,
        "effective_rank": effective,
    }


def _contrast_profile(left: _Sample, right: _Sample) -> dict[str, object]:
    a, b = len(left.points), len(right.points)
    delta = right.centroid - left.centroid
    distance = float(np.linalg.norm(delta))
    if not np.isfinite(distance):
        raise ValueError("Centroid distance must be finite.")
    mahalanobis, overlap, covariance_difference = np.nan, np.nan, np.nan
    if distance > 0:
        direction = delta / distance
        scores_left = np.einsum("ij,j->i", left.points - left.centroid, direction) - distance / 2
        scores_right = np.einsum("ij,j->i", right.points - right.centroid, direction) + distance / 2
        tolerance = 1e-12 * max(distance, np.abs(scores_left).max(), np.abs(scores_right).max())
        overlap = float(((scores_left > tolerance).mean() + (scores_right < -tolerance).mean()) / 2)
        overlap += float(
            ((np.abs(scores_left) <= tolerance).mean() + (np.abs(scores_right) <= tolerance).mean())
            / 4
        )
    if min(a, b) > 1:
        covariance_a = left.covariance / (a - 1)
        covariance_b = right.covariance / (b - 1)
        denominator = np.linalg.norm(covariance_a) + np.linalg.norm(covariance_b)
        if denominator > 0:
            covariance_difference = float(np.linalg.norm(covariance_a - covariance_b) / denominator)
    if a + b > 2:
        pooled = left.covariance / (a + b - 2) + right.covariance / (a + b - 2)
        eigenvalues, vectors = np.linalg.eigh(pooled)
        spectrum, rank, effective = _spectrum(eigenvalues)
        status = "rank_deficient"
        if rank == len(delta):
            mahalanobis = float(np.linalg.norm((delta @ vectors) / np.sqrt(eigenvalues)))
            status = "full_rank"
    else:
        spectrum, rank, effective = (np.nan,) * len(delta), np.nan, np.nan
        status = "no_covariance_degrees_of_freedom"
    return {
        "size": a + b,
        "left_size": a,
        "right_size": b,
        "smaller_share": min(a, b) / (a + b),
        "effective_n": a * b / (a + b),
        "centroid_distance": distance,
        "mahalanobis_distance": mahalanobis,
        "mahalanobis_status": status,
        "centroid_overlap_share": overlap,
        "pooled_covariance_eigenvalues": spectrum,
        "pooled_covariance_rank": rank,
        "pooled_effective_rank": effective,
        "relative_covariance_difference": covariance_difference,
    }


def _positions(values: Sequence[int], n: int) -> np.ndarray:
    rows = np.asarray(values)
    if (
        rows.ndim != 1
        or not rows.size
        or rows.dtype.kind not in "iu"
        or (rows < 0).any()
        or (rows >= n).any()
        or len(np.unique(rows)) != len(rows)
    ):
        raise ValueError("Subtree positions must be nonempty, unique, in-range integers.")
    return rows


def build_statistical_profiles(
    features: np.ndarray | pd.DataFrame,
    *,
    true_labels: np.ndarray | pd.Series | None = None,
    subtrees: Mapping[str, tuple[Sequence[int], Sequence[int]]] | None = None,
    geometry_disagreement: Mapping[str, float] | None = None,
) -> StatisticalProfiles:
    """Profile truth groups and explicit row-position subtree splits separately.

    Effective rank is the exponential entropy of the covariance spectrum.
    Contrast effective n is 1/(1/n_left+1/n_right). Overlap is the average
    in-sample opposite-centroid share across sides, with half weight for
    equidistant points (relative projection tolerance 1e-12). It is
    undefined at coincident centroids. Mahalanobis
    distance is unavailable if pooled covariance is numerically rank-deficient
    (eigenvalues at most 1e-10 times the largest are excluded from rank).
    """
    x = np.asarray(features, dtype=float)
    if x.ndim != 2 or min(x.shape) < 1 or not np.isfinite(x).all():
        raise ValueError("Expected a nonempty finite rows-by-features matrix.")
    if isinstance(features, pd.DataFrame) and not features.index.is_unique:
        raise ValueError("Feature index must be unique.")
    subtrees = {} if subtrees is None else subtrees
    geometry = {} if geometry_disagreement is None else geometry_disagreement
    if not geometry.keys() <= subtrees.keys() or not all(np.isfinite(v) for v in geometry.values()):
        raise ValueError("Supply finite geometry disagreement only for explicit subtrees.")

    group_rows, pair_rows, subtree_rows = [], [], []
    if true_labels is not None:
        if isinstance(true_labels, pd.Series) and isinstance(features, pd.DataFrame):
            if (
                not true_labels.index.is_unique
                or len(true_labels) != len(x)
                or not features.index.isin(true_labels.index).all()
            ):
                raise ValueError("Feature and truth indexes must match uniquely.")
            true_labels = true_labels.reindex(features.index)
        labels = np.asarray(true_labels)
        if labels.shape != (len(x),) or pd.isna(labels).any():
            raise ValueError("Truth labels must align to all feature rows without missing values.")
        codes, unique = pd.factorize(labels, sort=True)
        samples = {group: _sample(x[codes == index]) for index, group in enumerate(unique)}
        nearest = {}
        for left, right in combinations(samples, 2):
            row = _contrast_profile(samples[left], samples[right])
            pair_rows.append({"left_group": left, "right_group": right, **row})
            for own, other in ((left, right), (right, left)):
                if (
                    own not in nearest
                    or row["centroid_distance"] < nearest[own][1]["centroid_distance"]
                ):
                    nearest[own] = (other, row)
        for group, sample in samples.items():
            other, neighbor = nearest.get(group, (None, {}))
            group_rows.append(
                {
                    "group": group,
                    **_group_profile(sample, len(x)),
                    "nearest_group": other,
                    **{
                        f"nearest_{field}": neighbor.get(field, np.nan)
                        for field in (
                            "centroid_distance",
                            "mahalanobis_distance",
                            "centroid_overlap_share",
                            "relative_covariance_difference",
                        )
                    },
                }
            )

    for name, (left, right) in subtrees.items():
        a, b = _positions(left, len(x)), _positions(right, len(x))
        if np.intersect1d(a, b).size:
            raise ValueError("Explicit subtree sides must be disjoint.")
        row = _contrast_profile(_sample(x[a]), _sample(x[b]))
        subtree_rows.append(
            {"subtree": name, **row, "geometry_disagreement": geometry.get(name, np.nan)}
        )

    return StatisticalProfiles(
        pd.DataFrame(group_rows) if group_rows else pd.DataFrame(columns=["group"]),
        pd.DataFrame(pair_rows)
        if pair_rows
        else pd.DataFrame(columns=["left_group", "right_group"]),
        pd.DataFrame(subtree_rows) if subtree_rows else pd.DataFrame(columns=["subtree"]),
    )


# SOURCE: benchmarks/shared/performance_grid.py
"""Write clear benchmark performance grids from canonical result rows."""

from __future__ import annotations

from collections.abc import Sequence
from dataclasses import dataclass
from pathlib import Path

import numpy as np
import pandas as pd

CASE_COLUMNS = [
    "test_case",
    "case_id",
    "case_category",
    "source_family",
    "feature_representation",
    "samples",
    "features",
    "true_clusters",
]

METRIC_COLUMNS = ["ari", "nmi", "purity"]
OPTIONAL_METRIC_COLUMNS = [
    "ami",
    "matched_macro_f1",
    "matched_rare_f1",
    "pair_split_share",
    "pair_merge_share",
]


@dataclass(frozen=True)
class BenchmarkPerformanceGridArtifacts:
    """Paths emitted by the performance-grid writer."""

    report_md: Path
    summary_csv: Path
    ari_grid_csv: Path
    nmi_grid_csv: Path
    purity_grid_csv: Path
    status_grid_csv: Path
    support_coverage_csv: Path
    ami_grid_csv: Path | None = None


def _require_columns(results: pd.DataFrame, required: set[str]) -> None:
    missing = sorted(required.difference(results.columns))
    if missing:
        raise ValueError(f"Benchmark performance grid is missing result columns: {missing}")


def _normalize_grid_frame(results: pd.DataFrame) -> pd.DataFrame:
    required = set(CASE_COLUMNS) | {
        "method",
        "run_id",
        "benchmark_class",
        "benchmark_grid",
        "benchmark_repeat",
        "params",
        "found_clusters",
        "status",
        "skip_reason",
        "cluster_count_abs_error",
        *METRIC_COLUMNS,
    }
    _require_columns(results, required)

    frame = results.copy()
    for column in OPTIONAL_METRIC_COLUMNS:
        if column not in frame:
            frame[column] = np.nan
    for column in CASE_COLUMNS + [
        "method",
        "run_id",
        "benchmark_class",
        "benchmark_grid",
        "params",
        "status",
        "skip_reason",
    ]:
        frame[column] = frame[column].fillna("").astype(str)

    for column in [
        "benchmark_repeat",
        "found_clusters",
        "true_clusters",
        "cluster_count_abs_error",
        *METRIC_COLUMNS,
        *OPTIONAL_METRIC_COLUMNS,
    ]:
        frame[column] = pd.to_numeric(frame[column], errors="coerce")

    quality_columns = [*METRIC_COLUMNS, *OPTIONAL_METRIC_COLUMNS, "cluster_count_abs_error"]
    frame.loc[~frame["status"].eq("ok"), quality_columns] = np.nan

    duplicated = frame.duplicated(["case_id", "run_id"], keep=False)
    if duplicated.any():
        duplicate_rows = frame.loc[duplicated, ["case_id", "run_id"]].drop_duplicates()
        examples = duplicate_rows.head(5).to_dict(orient="records")
        raise ValueError(
            "Benchmark performance grid requires unique case_id/run_id rows; "
            f"duplicates include {examples}."
        )
    return frame


def _case_metadata(frame: pd.DataFrame) -> pd.DataFrame:
    return (
        frame[CASE_COLUMNS]
        .drop_duplicates()
        .sort_values(["test_case", "case_id"])
        .reset_index(drop=True)
    )


def _metric_grid(frame: pd.DataFrame, metric: str) -> pd.DataFrame:
    metadata = _case_metadata(frame)
    values = frame.pivot(index="case_id", columns="run_id", values=metric)
    values = values.reindex(metadata["case_id"])
    values = values.reindex(sorted(values.columns), axis=1)
    values = values.reset_index(drop=True)
    return pd.concat([metadata, values], axis=1)


def _first_nonempty(values: pd.Series) -> str:
    nonempty = values.dropna().astype(str)
    nonempty = nonempty[nonempty.str.len() > 0]
    return "" if nonempty.empty else str(nonempty.iloc[0])


def _summary(frame: pd.DataFrame) -> pd.DataFrame:
    working = frame.copy()
    working["ok"] = working["status"].eq("ok")
    working["unsupported"] = working["status"].eq("unsupported")
    working["skip"] = working["status"].eq("skip")
    working["error"] = working["status"].eq("error")
    working["exact_k"] = np.where(
        working["ok"] & working["true_clusters"].notna() & working["found_clusters"].notna(),
        (working["true_clusters"] == working["found_clusters"]).astype(float),
        np.nan,
    )

    grouped = working.groupby("run_id", sort=False, dropna=False)
    summary = grouped.agg(
        method=("method", _first_nonempty),
        benchmark_class=("benchmark_class", _first_nonempty),
        benchmark_grid=("benchmark_grid", _first_nonempty),
        benchmark_repeat=("benchmark_repeat", "first"),
        params=("params", _first_nonempty),
        total_rows=("case_id", "size"),
        successful_count=("ok", "sum"),
        unsupported_count=("unsupported", "sum"),
        skip_count=("skip", "sum"),
        error_count=("error", "sum"),
        mean_ari=("ari", "mean"),
        median_ari=("ari", "median"),
        mean_nmi=("nmi", "mean"),
        mean_purity=("purity", "mean"),
        **{f"mean_{column}": (column, "mean") for column in OPTIONAL_METRIC_COLUMNS},
        exact_k_rate=("exact_k", "mean"),
        mean_cluster_count_abs_error=("cluster_count_abs_error", "mean"),
    )
    summary = summary.reset_index()
    summary["attempted_count"] = summary["successful_count"] + summary["unsupported_count"]
    summary["unsupported_rate"] = np.where(
        summary["attempted_count"] > 0,
        summary["unsupported_count"] / summary["attempted_count"],
        np.nan,
    )
    summary = summary.sort_values(
        # Errors sort first: an erroring method's mean ARI skips the cases it failed on.
        ["error_count", "unsupported_rate", "mean_ari", "mean_nmi", "successful_count", "run_id"],
        ascending=[True, True, False, False, False, True],
        na_position="last",
    ).reset_index(drop=True)
    summary.insert(0, "rank_support_then_mean_ari", np.arange(1, len(summary) + 1))
    return summary


def _support_coverage(frame: pd.DataFrame) -> pd.DataFrame:
    working = frame.assign(
        successful=frame["status"].eq("ok"),
        unsupported=frame["status"].eq("unsupported"),
        skipped=frame["status"].eq("skip"),
        errored=frame["status"].eq("error"),
    )
    coverage = (
        working.groupby(["method", "case_category"], sort=True, dropna=False)
        .agg(
            total_rows=("case_id", "size"),
            successful_count=("successful", "sum"),
            unsupported_count=("unsupported", "sum"),
            skip_count=("skipped", "sum"),
            error_count=("errored", "sum"),
        )
        .reset_index()
    )
    coverage["attempted_count"] = coverage["successful_count"] + coverage["unsupported_count"]
    coverage["unsupported_rate"] = np.where(
        coverage["attempted_count"] > 0,
        coverage["unsupported_count"] / coverage["attempted_count"],
        np.nan,
    )
    return coverage


def markdown_table(
    frame: pd.DataFrame, columns: Sequence[str] | None = None, *, max_rows: int | None = None
) -> str:
    """Render the first ``max_rows`` rows of ``columns`` as markdown; floats use four decimals."""
    if columns is not None:
        frame = frame[[column for column in columns if column in frame.columns]]
    visible = frame.iloc[:max_rows].copy()
    if visible.empty:
        return "_No rows._"
    for column in visible.columns:
        if pd.api.types.is_float_dtype(visible[column]):
            visible[column] = visible[column].map(
                lambda value: "" if pd.isna(value) else f"{float(value):.4f}"
            )
    lines = [
        "| " + " | ".join(visible.columns) + " |",
        "| " + " | ".join("---" for _ in visible.columns) + " |",
    ]
    lines.extend(
        "| " + " | ".join(str(value).replace("|", "\\|") for value in row) + " |"
        for row in visible.itertuples(index=False)
    )
    return "\n".join(lines)

    status_grid_csv = output / "benchmark_performance_grid_status.csv"
    support_coverage_csv = output / "benchmark_support_coverage_by_method_case_family.csv"
    report_md = output / "benchmark_performance_grid.md"

    summary.to_csv(summary_csv, index=False)
    ari_grid.to_csv(ari_grid_csv, index=False)
    ami_grid.to_csv(ami_grid_csv, index=False)
    nmi_grid.to_csv(nmi_grid_csv, index=False)
    purity_grid.to_csv(purity_grid_csv, index=False)
    status_grid.to_csv(status_grid_csv, index=False)
    support_coverage.to_csv(support_coverage_csv, index=False)

    source_line = f"- Source rows: `{source_path}`\n" if source_path is not None else ""
    report_md.write_text(
        "\n".join(
            [
                "# Benchmark Performance Grid",
                "",
                source_line.rstrip(),
                f"- Result rows: `{len(frame)}`",
                f"- Benchmark cases: `{frame['case_id'].nunique()}`",
                f"- Run cells: `{frame['run_id'].nunique()}`",

                "## Artifacts",

                f"- ARI case grid: `{ari_grid_csv.name}`",
                f"- AMI case grid: `{ami_grid_csv.name}`",
                f"- NMI case grid: `{nmi_grid_csv.name}`",
                f"- Purity case grid: `{purity_grid_csv.name}`",
                f"- Status case grid: `{status_grid_csv.name}`",
                f"- Support coverage by method/case family: `{support_coverage_csv.name}`",
                "",
                "## Ranked Run Cells",
                "",
                "Sorted by unsupported rate first, then mean ARI on successful rows.",
                "",
                markdown_table(summary, max_rows=20),
                "",
            ]
        ),
        encoding="utf-8",
    )

    return BenchmarkPerformanceGridArtifacts(
        report_md=report_md,
        summary_csv=summary_csv,
        ari_grid_csv=ari_grid_csv,
        ami_grid_csv=ami_grid_csv,
        nmi_grid_csv=nmi_grid_csv,
        purity_grid_csv=purity_grid_csv,
        status_grid_csv=status_grid_csv,
        support_coverage_csv=support_coverage_csv,
    )

# SOURCE: benchmarks/validation/sweeps/subtree_research_validation.py
"""Completeness and statistical summaries for the frozen subtree research panels."""

from __future__ import annotations

import numpy as np

_STATUSES = ("ok", "unresolved", "unsupported", "error", "failed", "skip")
_AUXILIARY = ("ari", "matched_macro_f1", "matched_rare_f1", "pair_split_share", "pair_merge_share")


def _panel(frame, expected_cells, expected_replicates, columns, keys):
    for value in (expected_cells, expected_replicates):
        if isinstance(value, bool) or not isinstance(value, (int, np.integer)) or value < 1:
            raise ValueError("Expected positive integer cell and replicate counts.")
    required = {"cell", "replicate", "status", *columns}
    if not required <= set(frame) or not frame.columns.is_unique:
        raise ValueError(f"Missing or duplicate panel columns: {sorted(required - set(frame))}.")
    frame = frame.reset_index(drop=True).copy()
    for column, stop in (("cell", expected_cells), ("replicate", expected_replicates)):
        values = pd.to_numeric(frame[column], errors="coerce")
        if (
            frame[column].map(lambda v: isinstance(v, (bool, np.bool_))).any()
            or not (values.ge(0) & values.lt(stop) & values.mod(1).eq(0)).fillna(False).all()
        ):
            raise ValueError(f"Unknown {column} ID.")
        frame[column] = values.astype(int)
    if not frame.status.isin(_STATUSES).all():
        raise ValueError("Unknown panel status.")
    if frame.duplicated(keys).any():
        raise ValueError("Panel contains duplicate attempt keys.")
    if "phase" in frame and (not frame.phase.isin([1, 2]).all() or frame.phase.nunique() > 1):
        raise ValueError("Expected one known data phase per panel.")
    return frame


def _metric(series, low, high, *, required=None):
    values = pd.to_numeric(series, errors="coerce").astype(float)
    valid = values.between(low, high)
    boolean = series.map(lambda value: isinstance(value, (bool, np.bool_))).astype(bool)
    present = series.notna() if required is None else (series.notna() | required)
    if (present & (~valid | boolean)).any():
        raise ValueError(f"Malformed {series.name} value.")
    return values


def _holm(values):
    """Keep absent hypotheses in the declared family size."""
    order = sorted((value, i) for i, value in enumerate(values) if value is not None)
    adjusted = [None] * len(values)
    running = 0.0
    for rank, (value, index) in enumerate(order):
        running = max(running, min(1.0, (len(values) - rank) * value))
        adjusted[index] = float(running)
    return adjusted


def _status_counts(status, expected):
    counts = status.value_counts()
    return {key: int(counts.get(key, 0)) for key in _STATUSES} | {
        "missing": int(expected - status.notna().sum())
    }


def _frozen_shape(frame, actual, prescribed):
    return (
        actual == prescribed
        and not frame.empty
        and "phase" in frame
        and bool(frame.phase.eq(2).all())
    )


def calibration_summary(
    frame: pd.DataFrame,
    expected_cells: int = 24,
    expected_replicates: int = 5000,
    alphas: tuple[float, ...] = (0.01, 0.05),
) -> dict:
    """Summarize per-dataset calibration with exact intervals and Holm excess tests.

    Attempted unresolved/failed rows remain in the denominator, and only
    completed p-values contribute rejections. Missing attempts are reported
    separately. The source decision is retained as a count, while threshold
    comparisons use the completed p-values. An unresolved decision overrides
    an apparently successful attempt status. A pass is local to the
    prescribed phase-2 panel and requires resolved, nondegenerate outputs.
    """
    alphas = tuple(float(value) for value in alphas)
    if not alphas or len(set(alphas)) != len(alphas) or not all(0 < a < 1 for a in alphas):
        raise ValueError("Alphas must be distinct probabilities strictly between zero and one.")
    frame = _panel(
        frame,
        expected_cells,
        expected_replicates,
        ("p_value", "decision"),
        ["cell", "replicate"],
    )
    expected_cells, expected_replicates = int(expected_cells), int(expected_replicates)
    decisions = frame.decision.astype("string").str.strip().str.lower()
    resolved = decisions.isin(("reject", "do_not_reject"))
    unresolved = decisions.eq("unresolved").fillna(False)
    if (decisions.notna() & ~(resolved | unresolved)).any():
        raise ValueError("Unknown calibration decision.")
    if (frame.status.eq("ok") & decisions.isna()).any() or (
        ~frame.status.eq("ok") & resolved
    ).any():
        raise ValueError("Calibration decision conflicts with attempt status.")
    frame.loc[frame.status.eq("ok") & unresolved, "status"] = "unresolved"
    frame["p_value"] = _metric(frame.p_value, 0, 1, required=frame.status.eq("ok"))
    expected = expected_cells * expected_replicates
    counts = _status_counts(frame.status, expected)
    frozen = _frozen_shape(
        frame, (expected_cells, expected_replicates, alphas), (24, 5000, (0.01, 0.05))
    )
    comparisons, degenerate = [], []
    for cell in range(expected_cells):
        group = frame.loc[frame.cell == cell]
        completed = group.loc[group.status.eq("ok"), "p_value"]
        n = len(group)
        status = _status_counts(group.status, expected_replicates)
        zeros, ones = int(completed.eq(0).sum()), int(completed.eq(1).sum())
        if len(completed) > 1 and completed.nunique() == 1:
            degenerate.append(cell)
        for alpha in alphas:
            rejections = int(completed.le(alpha).sum())
            interval, excess = None, None
            if n:
                test = binomtest(rejections, n, p=alpha, alternative="greater")
                exact = binomtest(rejections, n).proportion_ci(method="exact")
                interval, excess = [float(exact.low), float(exact.high)], float(test.pvalue)
            comparisons.append(
                {
                    "cell": cell,
                    "alpha": alpha,
                    "attempts": n,
                    "rejections": rejections,
                    "status_counts": status,
                    "zero_p_values": zeros,
                    "one_p_values": ones,
                    "rate": rejections / n if n else None,
                    "interval": interval,
                    "excess_p": excess,
                }
            )
    adjusted = _holm([row["excess_p"] for row in comparisons])
    for row, value in zip(comparisons, adjusted, strict=True):
        row["excess_p_holm"] = value
    excess_detected = any(value is not None and value <= 0.05 for value in adjusted)
    all_successful = counts["ok"] == expected
    return {
        "design": "frozen_shape" if frozen else "development_or_custom",
        "complete": counts["missing"] == 0,
        "all_successful": all_successful,
        "passed": frozen and all_successful and not degenerate and not excess_detected,
        "expected_attempts": expected,
        "attempts": len(frame),
        "counts": counts,
        "decision_counts": {
            str(key): int(v) for key, v in frame.decision.value_counts(dropna=False).items()
        },
        "degenerate_cells": degenerate,
        "excess_detected": excess_detected,
        "comparisons": comparisons,
    }


def _difference_stats(differences):
    values = np.asarray(differences, dtype=float)
    n = len(values)
    if not n:
        return None, None, None
    mean = float(values.mean())
    if n < 2:
        return mean, None, None
    if np.ptp(values) == 0:
        return mean, [mean, mean], float(mean >= 0)
    stderr = float(values.std(ddof=1) / np.sqrt(n))
    margin = float(t.ppf(0.975, n - 1)) * stderr
    return mean, [mean - margin, mean + margin], float(t.cdf(mean / stderr, n - 1))


def paired_recovery_summary(
    frame: pd.DataFrame,
    baseline: str,
    candidate: str,
    expected_cells: int = 16,
    expected_replicates: int = 200,
) -> dict:
    """Compare matched datasets, weighting all cells equally within a replicate block.

    The primary interval is Student t across blocks with a successful pair
    in every prescribed cell. Per-cell loss tests use paired differences and
    Holm over every prescribed cell. Incomplete pairs and blocks are counted
    explicitly and preclude a frozen pass.
    """
    if not baseline or not candidate or baseline == candidate:
        raise ValueError("Supply two distinct methods.")
    frame = _panel(
        frame,
        expected_cells,
        expected_replicates,
        ("method", "ami", "data_hash", "seed"),
        ["cell", "replicate", "method"],
    )
    expected_cells, expected_replicates = int(expected_cells), int(expected_replicates)
    if not frame.method.isin([baseline, candidate]).all():
        raise ValueError("Unknown method ID in paired frame.")
    if (
        not frame.data_hash.map(lambda value: isinstance(value, str)).all()
        or not frame.data_hash.astype(str).str.fullmatch(r"[0-9a-fA-F]{64}").all()
    ):
        raise ValueError("Expected a SHA-256 data_hash for each attempt.")
    frame["data_hash"] = frame.data_hash.astype(str).str.lower()
    seeds = pd.to_numeric(frame.seed, errors="coerce")
    if (
        frame.seed.map(lambda v: isinstance(v, (bool, np.bool_))).any()
        or not (np.isfinite(seeds) & seeds.ge(0) & seeds.mod(1).eq(0)).fillna(False).all()
    ):
        raise ValueError("Expected a nonnegative integer seed for each attempt.")
    frame["seed"] = seeds
    frame["ami"] = _metric(frame.ami, -1, 1, required=frame.status.eq("ok"))
    auxiliary = [metric for metric in _AUXILIARY if metric in frame]
    for metric in auxiliary:
        frame[metric] = _metric(frame[metric], -1 if metric == "ari" else 0, 1)
    expected = expected_cells * expected_replicates
    frozen = _frozen_shape(frame, (expected_cells, expected_replicates), (16, 200))
    index = pd.MultiIndex.from_product(
        [range(expected_cells), range(expected_replicates)], names=["cell", "replicate"]
    )
    base, trial = (
        frame.loc[frame.method == method].set_index(["cell", "replicate"]).reindex(index)
        for method in (baseline, candidate)
    )
    both_attempted = base.status.notna() & trial.status.notna()
    dataset_rows = frame.drop_duplicates(["cell", "replicate"])
    for key in ("seed", "data_hash"):
        if (both_attempted & base[key].ne(trial[key])).any():
            raise ValueError(f"Paired dataset {key} mismatch.")
        if dataset_rows[key].duplicated().any():
            raise ValueError(f"Dataset {key} is reused across distinct attempt keys.")
    successful = base.status.eq("ok") & trial.status.eq("ok")
    ami_delta = (trial.ami - base.ami).where(successful)
    cells = []
    for cell in range(expected_cells):
        valid = successful.loc[cell]
        mean, interval, loss = _difference_stats(ami_delta.loc[cell].dropna())
        other = {}
        for metric in auxiliary:
            delta = (trial[metric].loc[cell] - base[metric].loc[cell]).where(valid).dropna()
            other[metric] = {
                "pairs": len(delta),
                "mean_difference": float(delta.mean()) if len(delta) else None,
            }
        cells.append(
            {
                "cell": cell,
                "paired_attempted": int(both_attempted.loc[cell].sum()),
                "paired_successful": int(valid.sum()),
                "mean_ami_difference": mean,
                "interval": interval,
                "loss_p": loss,
                "auxiliary": other,
                "baseline_status_counts": _status_counts(
                    base.status.loc[cell], expected_replicates
                ),
                "candidate_status_counts": _status_counts(
                    trial.status.loc[cell], expected_replicates
                ),
            }
        )
    adjusted = _holm([row["loss_p"] for row in cells])
    for row, value in zip(cells, adjusted, strict=True):
        row["loss_p_holm"] = value
    loss_detected = any(value is not None and value <= 0.05 for value in adjusted)
    blocks = ami_delta.unstack("cell").dropna().mean(axis=1)
    mean, interval, _ = _difference_stats(blocks)
    all_successful = bool(successful.all())
    return {
        "design": "frozen_shape" if frozen else "development_or_custom",
        "baseline": baseline,
        "candidate": candidate,
        "complete": bool(both_attempted.all()),
        "all_successful": all_successful,
        "passed": frozen
        and all_successful
        and interval is not None
        and interval[0] > 0
        and not loss_detected,
        "loss_detected": loss_detected,
        "coverage": {
            "expected_pairs": expected,
            "paired_attempted": int(both_attempted.sum()),
            "paired_successful": int(successful.sum()),
            "paired_share": float(successful.sum() / expected),
            "baseline_status_counts": _status_counts(base.status, expected),
            "candidate_status_counts": _status_counts(trial.status, expected),
        },
        "aggregate": {
            "blocks": len(blocks),
            "expected_blocks": expected_replicates,
            "mean_ami_difference": mean,
            "interval": interval,
        },
        "cells": cells,
    }

def interval(values):
    x = np.asarray(values, dtype=float)
    if len(x) < 2:
        return [None, None]
    mean = x.mean()
    half = t.ppf(0.975, len(x) - 1) * x.std(ddof=1) / np.sqrt(len(x))
    return [float(mean - half), float(mean + half)]


def main():
    factorial = pd.concat(
        [pd.read_csv(HERE / f"factorial_cell_{cell}/runs.csv") for cell in range(3)],
        ignore_index=True,
    )
    recovery = pd.read_csv(HERE / "recovery/runs.csv")
    assert len(factorial) == 2400 and len(recovery) == 1600
    assert not factorial.duplicated(["cell", "replicate", "method"]).any()
    assert not recovery.duplicated(["cell", "replicate", "method"]).any()
    for frame in (factorial, recovery):
        assert frame.groupby(["cell", "replicate"]).data_hash.nunique().eq(1).all()
        assert frame.groupby(["cell", "replicate"]).truth_hash.nunique().eq(1).all()

    summary = (
        factorial.groupby(["cell", *FACTORIAL_AXES])
        .agg(
            attempted=("status", "size"),
            successful=("status", lambda x: int(x.eq("ok").sum())),
            root_rejections=("root_sibling_reject", "sum"),
            mean_ami=("ami", "mean"),
            mean_ari=("ari", "mean"),
            mean_matched_f1=("matched_macro_f1", "mean"),
            mean_groups=("found_clusters", "mean"),
            mean_split_share=("pair_split_share", "mean"),
            mean_merge_share=("pair_merge_share", "mean"),
        )
        .reset_index()
    )
    summary.to_csv(HERE / "factorial_summary.csv", index=False)
    effects = []
    for cell, group in factorial.groupby("cell"):
        for factor, levels in FACTORIAL_AXES.items():
            for metric in ("ami", "matched_macro_f1", "found_clusters", "root_sibling_reject"):
                pivot = group.assign(
                    value=pd.to_numeric(group[metric], errors="coerce")
                ).pivot_table(index="replicate", columns=factor, values="value", aggfunc="mean")
                supported = group.groupby("replicate").apply(
                    lambda g: len(g) == 16 and g.status.eq("ok").all(), include_groups=False
                )
                delta = (pivot[levels[1]] - pivot[levels[0]]).loc[supported].dropna()
                effects.append(

                        cell=int(cell),
                        factor=factor,
                        level0=str(levels[0]),

            matched_macro_f1=("matched_macro_f1", "mean"),

            pair_split_share=("pair_split_share", "mean"),
            pair_merge_share=("pair_merge_share", "mean"),
            clusters=("found_clusters", "mean"),
        )
        .reset_index()
    )

        grouped[name] = grouped.cell.map(lambda c: RECOVERY_CELLS[c][index])
    grouped.to_csv(HERE / "recovery_summary.csv", index=False)

    initial = json.loads((HERE / "environment.json").read_text())["source_hashes"]
    production_changed = [
        p
        for p, h in initial.items()
        if p.startswith("tree_break_selection/")
        and hashlib.sha256((ROOT / p).read_bytes()).hexdigest() != h
    ]
    runtime = json.loads((HERE / "preflight_isolated/runtime_gate.json").read_text())
    decision = {
        "status": "not_promoted",

def digest(value, dtype):
    return hashlib.sha256(np.ascontiguousarray(value, dtype=dtype).tobytes()).hexdigest()


def main():
    checked = {}
    distinct = set()
    for panel in (
        "recovery",
        "factorial_cell_0",

                    x, truth, seed = recovery_data(c, rep)
                inputs[key] = (seed, digest(x, np.float64), digest(truth, np.int64))
            assert inputs[key] == (row["seed"], row["data_hash"], row["truth_hash"]), row["run_id"]
            assert row["status"] == "ok", row["run_id"]

                distinct.add(row["data_hash"])
        before_after = json.loads((HERE / panel / "verification.json").read_text())
        assert not before_after["source_files_changed_during_run"], panel
        checked[panel] = {

# SOURCE: benchmarks/diagnostics/calibration/selected/hierarchy/subtree_gaussian_reference.py
"""Gaussian common-mean references for a specified subtree."""

import math
from numbers import Integral, Real

import numpy as np
from scipy.linalg import helmert
from scipy.stats import beta


def _rows(indices, n, *, name, minimum):
    try:
        rows = tuple(indices)
    except TypeError as error:
        raise ValueError(f"{name} must be a sequence of row indices") from error
    if (
        len(rows) < minimum
        or any(isinstance(i, bool) or not isinstance(i, Integral) or not 0 <= i < n for i in rows)
        or len(set(rows)) != len(rows)
    ):
        raise ValueError(f"{name} must contain at least {minimum} distinct valid row indices")
    return np.asarray(rows, dtype=np.intp)


def _inputs(values, subtree_indices, alpha):
    if isinstance(alpha, bool) or not isinstance(alpha, Real) or not 0 < alpha < 1:
        raise ValueError("alpha must lie strictly between zero and one")
    x = np.asarray(values, dtype=float)
    if x.ndim != 2 or x.shape[1] < 1 or not np.isfinite(x).all():
        raise ValueError("values must be a finite matrix with at least one feature")
    rows = _rows(subtree_indices, len(x), name="subtree_indices", minimum=2)
    # Membership alone fixes the proposal basis, including for data-ordered input indices.
    return x, np.sort(rows)


def _rotated(state, rng):
    proposed = state.copy()
    if len(state) == 1:
        sign = 1 if rng.integers(2) else -1
        proposed[0] *= sign
        return proposed, (0, 0, 0.0, sign)
    i, j = sorted(map(int, rng.choice(len(state), size=2, replace=False)))
    angle = float(rng.uniform(-np.pi, np.pi))
    sign = 1 if rng.integers(2) else -1
    c, s = math.cos(angle), math.sin(angle)
    proposed[i] = c * state[i] - s * state[j]
    proposed[j] = sign * (s * state[i] + c * state[j])
    return proposed, (i, j, angle, sign)


def subtree_gaussian_chain_test(
    values, *, subtree_indices, score, selection_event=None, steps=999, seed=0, alpha=0.05
):
    """Rank a deterministic full-matrix score under the common-mean Gaussian null.

    Condition on the subtree mean, centered feature Gram matrix, outside rows
    and, if supplied, the caller's selection event. Subtree rows must be iid
    Gaussian with one common mean and covariance and independent of outside
    rows. Unknown covariance is allowed. The event must describe how the tested
    membership and any other conditioned selection were obtained. Omitting it
    requires prespecified membership. Callbacks receive read-only matrices.
    Any score-defining choices captured by callbacks must be prespecified or
    part of the event. A failed evaluation invalidates the rank. No descendant or traversal event
    is inferred here, and native floating-point event membership is uncertified.

    Chain moves store accepted transforms as (i, j, angle, sign), relative to
    the preceding state when moving outward from the observed anchor. An empty
    entry keeps the preceding state. Equal indices denote the one-axis case.
    """
    x, rows = _inputs(values, subtree_indices, alpha)
    for name, value, minimum in (("steps", steps, 1), ("seed", seed, 0)):
        if isinstance(value, bool) or not isinstance(value, Integral) or value < minimum:
            raise ValueError(f"{name} must be an integer >= {minimum}")
    if not callable(score) or (selection_event is not None and not callable(selection_event)):
        raise ValueError("score and any selection_event must be callable")

    def in_event(matrix):
        if selection_event is None:
            return True
        accepted = selection_event(matrix)
        if not isinstance(accepted, (bool, np.bool_)):
            raise ValueError("selection_event must return a boolean")
        return bool(accepted)

    def score_value(matrix):
        value = score(matrix)
        if isinstance(value, bool) or not isinstance(value, Real) or not math.isfinite(value):
            raise ValueError("score must return a finite real number")
        return float(value)

    rng = np.random.default_rng(seed)
    anchor = int(rng.integers(steps + 1))
    scores = [None] * (steps + 1)
    moves = [None] * (steps + 1)
    result = dict(
        status="unresolved",
        decision="unresolved",
        p_value=None,
        error=None,
        sampling_status="unresolved",
        observed_score=None,
        anchor_index=anchor,
        accepted_moves=0,
        completed_steps=0,
        event_evaluations=0,
        chain_scores=scores,
        chain_moves=moves,
        alpha=float(alpha),
        seed=int(seed),
        steps_requested=int(steps),
        resolution_sufficient=1 / (steps + 1) <= alpha,
        subtree_indices=rows.tolist(),
        numerical_event_certified=False,
        selection_event="fixed_membership" if selection_event is None else "caller_supplied",
        observation_model="independent_gaussian_common_mean_covariance",
    )
    observed = x.copy()
    observed.setflags(write=False)
    phase = "observation"
    try:
        if not in_event(observed):
            raise ValueError("observed matrix does not satisfy selection_event")
        scores[anchor] = result["observed_score"] = score_value(observed)
        with np.errstate(over="raise", invalid="raise"):
            basis = helmert(len(rows), full=False)
            mean = x[rows].mean(axis=0)
            original = basis @ (x[rows] - mean)
            phase = "sampling"
            for positions in (range(anchor - 1, -1, -1), range(anchor + 1, steps + 1)):
                current, current_score = original, scores[anchor]
                for index in positions:
                    proposed, move = _rotated(current, rng)
                    if not np.array_equal(proposed, current):
                        candidate = x.copy()
                        candidate[rows] = mean + basis.T @ proposed
                        candidate.setflags(write=False)
                        if selection_event is not None:
                            result["event_evaluations"] += 1
                        if in_event(candidate):
                            current_score = score_value(candidate)
                            current = proposed
                            moves[index] = move
                            result["accepted_moves"] += 1
                    scores[index] = current_score
                    result["completed_steps"] += 1
    except Exception as error:
        result.update(error_phase=phase, error=f"{type(error).__name__}: {error}")
        return result

    p_value = sum(value >= scores[anchor] for value in scores) / (steps + 1)
    moved = result["accepted_moves"] > 0
    result.update(
        status="complete", p_value=p_value, sampling_status="moved" if moved else "stalled"
    )
    if moved and result["resolution_sufficient"]:
        result["decision"] = "reject" if p_value <= alpha else "do_not_reject"
    return result


def fixed_subtree_contrast_test(
    values, *, subtree_indices, left_indices, rank=None, all_labels=False, alpha=0.05
):
    """Test a fixed binary row contrast on a prespecified common-mean subtree.

    The statistic is squared projection of the unit row contrast onto the
    leading rank-k left singular subspace of the centered subtree. The right
    directions and numerical rank are functions of its feature Gram matrix.
    For fixed membership and labels the conditional law is Beta(k/2,(m-1-k)/2).
    An optional Bonferroni factor counts all unordered nonempty binary splits.
    It permits label search on the specified subtree. Selected membership and
    contrast-selected rank are outside the reference.
    """
    x, rows = _inputs(values, subtree_indices, alpha)
    left = _rows(left_indices, len(x), name="left_indices", minimum=1)
    left_mask = np.isin(rows, left)
    if left_mask.sum() != len(left) or len(left) == len(rows):
        raise ValueError("left_indices must form a nonempty proper subset of subtree_indices")
    if rank is not None and (isinstance(rank, bool) or not isinstance(rank, Integral) or rank < 0):
        raise ValueError("rank must be a nonnegative integer or None")
    if not isinstance(all_labels, (bool, np.bool_)):
        raise ValueError("all_labels must be a boolean")

    m = len(rows)
    count = (1 << (m - 1)) - 1 if all_labels else 1
    result = dict(
        status="unresolved",
        decision="unresolved",
        error=None,
        p_value=None,
        unadjusted_p_value=None,
        statistic=None,
        rank=None,
        subtree_indices=rows.tolist(),
        left_indices=left.tolist(),
        number_of_labels=count,
        alpha=float(alpha),
        selection_event="fixed_membership",
        all_labels=bool(all_labels),
        observation_model="independent_gaussian_common_mean_covariance",
    )
    try:
        with np.errstate(over="raise", invalid="raise"):
            basis = helmert(m, full=False)
            centered = x[rows] - x[rows].mean(axis=0)
            u, singular, _ = np.linalg.svd(basis @ centered, full_matrices=False)
        tolerance = singular[0] * max(centered.shape) * np.finfo(float).eps
        full_rank = int(np.count_nonzero(singular > tolerance))
        k = full_rank if rank is None else int(rank)
        result["rank"] = k
        if not 0 < k < m - 1 or k > full_rank:
            raise ValueError(
                "rank is zero, exceeds the numerical rank, or spans the centered row space"
            )
        if k < full_rank and singular[k - 1] - singular[k] <= tolerance:
            raise ValueError("rank cuts a tied Gram eigenspace")
        contrast = np.where(left_mask, 1 / len(left), -1 / (m - len(left)))
        contrast /= np.linalg.norm(contrast)
        q = float(np.linalg.norm(u[:, :k].T @ (basis @ contrast)) ** 2)
        if not 0 <= q < 1:
            raise FloatingPointError("projection statistic is outside numerical Beta support")
        log_tail = float(beta.logsf(q, k / 2, (m - 1 - k) / 2))
        if not math.isfinite(log_tail):
            raise FloatingPointError("Beta tail underflowed or is nonfinite")
    except (ValueError, FloatingPointError, np.linalg.LinAlgError) as error:
        result["error"] = f"{type(error).__name__}: {error}"
        return result

    p_value = math.exp(min(0.0, log_tail + math.log(count)))
    return result | dict(
        status="complete",
        decision="reject" if p_value <= alpha else "do_not_reject",
        p_value=p_value,
        unadjusted_p_value=math.exp(log_tail),
        statistic=q,
    )


# SOURCE: benchmarks/diagnostics/calibration/selected/hierarchy/subtree_topology_geometry.py
"""Bounded geometry diagnostics on a fixed labeled subtree.

Leaf labels are row positions. Cut enumeration, Ward trees in spectral spans,
and the combinatorial topology family are separate objects.
"""

from __future__ import annotations

from collections.abc import Iterator
from dataclasses import dataclass
from functools import cache
from itertools import combinations, product

import numpy as np
from scipy.cluster.hierarchy import cophenet, linkage
from scipy.optimize import nnls
from scipy.spatial.distance import pdist, squareform

from tree_break_selection.eigenvalue_groups import tied_eigenvalue_neighbors
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import (
    is_separable,
)

MAX_CUT_LEAVES = 10
MAX_TOPOLOGY_LEAVES = 6

type Tree = int | tuple[Tree, Tree]


@dataclass(frozen=True)
class Cut:
    """Unordered split with the smallest leaf on the left."""

    left: tuple[int, ...]
    right: tuple[int, ...]


@dataclass(frozen=True, eq=False)
class SpectralSubspace:
    """Descending Gram eigenblock ranges and the resulting sample geometry."""

    eigenblocks: tuple[tuple[int, int], ...]
    rank: int
    projector: np.ndarray
    coordinates: np.ndarray


@dataclass(frozen=True, eq=False)
class SpectralGeometry:
    subspace: SpectralSubspace
    cuts: tuple[Cut, ...]
    distances: np.ndarray
    ward_linkage: np.ndarray
    ward_tree: Tree


@dataclass(frozen=True, eq=False)
class AdditiveDistanceFit:
    """Nonnegative lengths by unique edge split and fitted condensed distances."""

    edge_splits: tuple[Cut, ...]
    branch_lengths: np.ndarray
    fitted_distances: np.ndarray
    rmse: float
    max_abs_residual: float


def unordered_cuts(n: int) -> tuple[Cut, ...]:
    """All nonempty complementary splits, with row zero on the left."""
    if not 2 <= n <= MAX_CUT_LEAVES:
        raise ValueError(f"Cut enumeration requires 2 through {MAX_CUT_LEAVES} leaves.")
    return tuple(
        Cut(
            (0, *(j for j in range(1, n) if mask & (1 << (j - 1)))),
            tuple(j for j in range(1, n) if not mask & (1 << (j - 1))),
        )
        for mask in range((1 << (n - 1)) - 1)
    )


def spectral_subspaces(points: np.ndarray) -> tuple[SpectralSubspace, ...]:
    """Return each positive whole Gram block and each distinct leading block union.

    Each projector acts on row space. The coordinates retain Euclidean geometry
    within the projected feature space. SVD avoids explicitly squaring the
    condition number or allocating the feature Gram matrix.
    """
    x = np.asarray(points, dtype=np.float64)
    if x.ndim == 2 and not 2 <= len(x) <= MAX_CUT_LEAVES:
        raise ValueError(f"Subtree geometry requires 2 through {MAX_CUT_LEAVES} leaves.")
    return tuple(
        SpectralSubspace(group, basis.shape[1], basis @ basis.T, coordinates)
        for group, basis, coordinates in _spectral_coordinate_spans(x)
    )


def _spectral_coordinate_spans(
    points: np.ndarray,
) -> Iterator[tuple[tuple[tuple[int, int], ...], np.ndarray, np.ndarray]]:
    """Generate whole spectral spans without retaining dense row projectors."""
    x = np.asarray(points, dtype=np.float64)
    if x.ndim != 2 or min(x.shape) < 1 or not np.isfinite(x).all():
        raise ValueError("Expected a finite, nonempty rows-by-features matrix.")
    if len(x) < 2:
        raise ValueError("Subtree geometry requires at least two leaves.")

    # Scale before centering and SVD to keep the Gram spectrum finite.
    scale = float(np.max(np.abs(x)))
    if scale == 0:
        return
    normalized = x / scale
    normalized -= normalized[0].copy()
    u, singular, _ = np.linalg.svd(normalized - normalized.mean(axis=0), full_matrices=False)
    threshold = np.finfo(float).eps * max(x.shape) * singular[0]
    rank = min(len(x) - 1, int(np.count_nonzero(singular > threshold)))
    if rank == 0:
        return

    values = (singular[:rank] / singular[0]) ** 2
    ends = [*(np.flatnonzero(~tied_eigenvalue_neighbors(values)) + 1).tolist(), rank]
    blocks = tuple(zip([0, *ends[:-1]], ends, strict=True))
    groups = [(block,) for block in blocks] + [blocks[:k] for k in range(2, len(blocks) + 1)]
    for group in groups:
        start, stop = group[0][0], group[-1][1]
        row_basis = u[:, start:stop]
        coordinates = (row_basis * singular[start:stop]) * scale
        if not np.isfinite(coordinates).all():
            raise ValueError("Projected coordinates must fit in finite float64.")
        yield group, row_basis, coordinates


def spectral_geometry(points: np.ndarray) -> tuple[SpectralGeometry, ...]:
    """Enumerate strict affine cuts and compute a fixed-row-order Ward tree per span."""
    subspaces = spectral_subspaces(points)
    records = []
    for subspace in subspaces:
        x = subspace.coordinates
        # Column rescaling preserves affine feasibility and avoids scaling the LP
        # by the Gram eigenvalues or measurement units.
        lp_points = x / np.max(np.abs(x), axis=0)
        cuts = tuple(
            cut
            for cut in unordered_cuts(len(x))
            if is_separable(lp_points, np.isin(np.arange(len(x)), cut.left))
        )
        scale = np.max(np.abs(x))
        scaled_distances = pdist(x / scale)
        distances = scaled_distances * scale
        if not np.isfinite(distances).all():
            raise ValueError("Projected distances must fit in finite float64.")
        # shortcut: round relative input distances at 1e-12 for deterministic numerical
        # ties in fixed row order, replace if exact near-tie Ward decisions are required.
        max_distance = scaled_distances.max()
        ward = linkage(np.round(scaled_distances / max_distance, 12), method="ward")
        ward[:, 2] *= max_distance * scale
        if not np.isfinite(ward).all():
            raise ValueError("Ward linkage heights must fit in finite float64.")
        nodes: list[Tree] = list(range(len(x)))
        first_leaf = list(range(len(x)))
        for a, b in ward[:, :2].astype(int):
            if first_leaf[b] < first_leaf[a]:
                a, b = b, a
            nodes.append((nodes[a], nodes[b]))
            first_leaf.append(first_leaf[a])
        records.append(SpectralGeometry(subspace, cuts, distances, ward, nodes[-1]))
    return tuple(records)


def rooted_binary_topologies(n: int) -> tuple[Tree, ...]:
    """Enumerate all unordered rooted full binary trees of at most six labeled leaves."""
    if not 2 <= n <= MAX_TOPOLOGY_LEAVES:
        raise ValueError(f"Topology enumeration requires 2 through {MAX_TOPOLOGY_LEAVES} leaves.")

    def recurse(labels: tuple[int, ...]) -> tuple[Tree, ...]:
        if len(labels) == 1:
            return (labels[0],)
        trees = []
        for cut in unordered_cuts(len(labels)):
            left = recurse(tuple(labels[i] for i in cut.left))
            right = recurse(tuple(labels[i] for i in cut.right))
            trees.extend(product(left, right))
        return tuple(trees)

    return recurse(tuple(range(n)))


def ward_topology_family(
    points_or_condensed: np.ndarray, *, rtol: float = 1e-10
) -> frozenset[Tree]:
    """All Ward trees reachable by tied merges on two through six leaves.

    Accepts Euclidean coordinates or condensed Euclidean distances. At each
    step include every increase in within-cluster SSE within `rtol` times the
    largest original squared pair distance of the minimum. Zero uses exact
    floating-point comparisons.
    """
    x = np.asarray(points_or_condensed, dtype=float)
    if not np.isfinite(rtol) or rtol < 0:
        raise ValueError("rtol must be finite and nonnegative.")
    if not x.size or not np.isfinite(x).all() or x.ndim not in (1, 2):
        raise ValueError("Expected finite coordinates or condensed Euclidean distances.")
    if x.ndim == 1 and (x < 0).any():
        raise ValueError("Euclidean distances must be nonnegative.")
    scale = np.max(np.abs(x))
    if scale > 0:
        x = x / scale
    if x.ndim == 2 and not 2 <= len(x) <= MAX_TOPOLOGY_LEAVES:
        raise ValueError(
            f"Ward topology enumeration requires 2 through {MAX_TOPOLOGY_LEAVES} leaves."
        )
    distances = pdist(x) if x.ndim == 2 else x
    if not 1 <= len(distances) <= MAX_TOPOLOGY_LEAVES * (MAX_TOPOLOGY_LEAVES - 1) // 2:
        raise ValueError(
            f"Ward topology enumeration requires 2 through {MAX_TOPOLOGY_LEAVES} leaves."
        )
    max_distance = distances.max()
    normalized = distances / max_distance if max_distance > 0 else distances
    squared = squareform(normalized**2)
    n = len(squared)
    if x.ndim == 1:
        centered = np.eye(n) - 1 / n
        if np.linalg.eigvalsh(-0.5 * centered @ squared @ centered)[0] < -1e-12:
            raise ValueError("Condensed distances must be Euclidean.")

    @cache
    def merge_cost(left: tuple[int, ...], right: tuple[int, ...]) -> float:
        a, b = len(left), len(right)
        cross = squared[np.ix_(left, right)].sum()
        within_a = sum(squared[i, j] for i, j in combinations(left, 2))
        within_b = sum(squared[i, j] for i, j in combinations(right, 2))
        # (ab/(a+b)) times the squared centroid distance, via pair distances.
        return max(0.0, (a * b * cross - b**2 * within_a - a**2 * within_b) / (a * b * (a + b)))

    @cache
    def expand(forest: tuple[tuple[tuple[int, ...], Tree], ...]) -> frozenset[Tree]:
        if len(forest) == 1:
            return frozenset((forest[0][1],))
        costs = [
            (merge_cost(forest[i][0], forest[j][0]), i, j)
            for i, j in combinations(range(len(forest)), 2)
        ]
        threshold = min(cost for cost, _, _ in costs) + rtol
        trees = set()
        for cost, i, j in costs:
            if cost <= threshold:
                left, right = forest[i], forest[j]
                merged = (tuple(sorted(left[0] + right[0])), (left[1], right[1]))
                trees.update(expand(forest[:i] + (merged,) + forest[i + 1 : j] + forest[j + 1 :]))
        return frozenset(trees)

    return expand(tuple(((i,), i) for i in range(n)))


def _clades(tree: Tree) -> list[frozenset[int]]:
    if isinstance(tree, int):
        return [frozenset((tree,))]
    left, right = map(_clades, tree)
    return [*left, *right, left[-1] | right[-1]]


def tree_clades(tree: Tree) -> frozenset[frozenset[int]]:
    """Nontrivial clades for rooted topology comparison."""
    return frozenset(clade for clade in _clades(tree)[:-1] if len(clade) > 1)


def tree_edge_splits(tree: Tree) -> frozenset[Cut]:
    """Unordered edge splits including pendant edges, deduplicating the root split."""
    clades = _clades(tree)
    leaves = clades.pop()
    first = min(leaves)
    splits = set()
    for clade in clades:
        left, right = (clade, leaves - clade) if first in clade else (leaves - clade, clade)
        splits.add(Cut(tuple(sorted(left)), tuple(sorted(right))))
    return frozenset(splits)


def fit_additive_distances(tree: Tree, condensed_distances: np.ndarray) -> AdditiveDistanceFit:
    """Fit all leaf-pair distances to unique edge splits on a small fixed tree.

    Targets use the supplied distance units. Complementary root edges have
    the same incidence column, so their lengths are represented by one sum.
    Zero fitted internal lengths give unresolved splits in the fitted metric.
    """
    clades = _clades(tree)
    n = len(clades[-1])
    if not 2 <= n <= MAX_TOPOLOGY_LEAVES:
        raise ValueError(
            f"Additive distance fitting requires 2 through {MAX_TOPOLOGY_LEAVES} leaves."
        )
    if clades[-1] != frozenset(range(n)) or sum(len(clade) == 1 for clade in clades) != n:
        raise ValueError("Expected unique leaf labels 0 through n-1.")
    observed = np.asarray(condensed_distances, dtype=float)
    if (
        observed.shape != (n * (n - 1) // 2,)
        or not np.isfinite(observed).all()
        or (observed < 0).any()
    ):
        raise ValueError("Expected finite nonnegative condensed distances for the supplied leaves.")

    splits = tuple(sorted(tree_edge_splits(tree), key=lambda split: split.left))
    membership = np.column_stack([np.isin(np.arange(n), split.left) for split in splits])
    left, right = np.triu_indices(n, k=1)
    design = (membership[left] != membership[right]).astype(float)
    scale = float(observed.max()) or 1.0
    normalized = observed / scale
    lengths, _ = nnls(design, normalized)
    fitted = design @ lengths
    residuals = fitted - normalized
    lengths, fitted = lengths * scale, fitted * scale
    if not np.isfinite(fitted).all():
        raise ValueError("Fitted distances must fit in finite float64.")
    return AdditiveDistanceFit(
        splits,
        lengths,
        fitted,
        float(np.sqrt(np.mean(residuals**2))) * scale,
        float(np.max(np.abs(residuals))) * scale,
    )


def rooted_robinson_foulds(left: Tree, right: Tree) -> int:
    """Number of nontrivial rooted clades present in only one of the trees."""
    if _clades(left)[-1] != _clades(right)[-1]:
        raise ValueError("Rooted RF requires the same leaf labels in both trees.")
    return len(tree_clades(left) ^ tree_clades(right))


def cophenetic_distances(tree_or_linkage: Tree | np.ndarray) -> np.ndarray:
    """Condensed cophenetic distances in row order.

    An abstract topology assigns a node the height of its longest path to a
    leaf in edge units. A linkage matrix preserves its supplied heights.
    """
    if isinstance(tree_or_linkage, np.ndarray):
        return cophenet(tree_or_linkage)

    return cophenet(topology_linkage(tree_or_linkage))


def topology_linkage(tree: Tree) -> np.ndarray:
    """Encode a topology using monotone heights in edge units."""
    leaves = _clades(tree)[-1]
    n = len(leaves)
    if n < 2 or leaves != frozenset(range(n)):
        raise ValueError("Cophenetic topology leaves must be numbered 0 through n-1.")
    merges = []

    def build(node: Tree) -> tuple[int, int, int]:
        if isinstance(node, int):
            return node, 0, 1
        a, ha, na = build(node[0])
        b, hb, nb = build(node[1])
        parent = n + len(merges)
        height, count = 1 + max(ha, hb), na + nb
        merges.append((a, b, height, count))
        return parent, height, count

    _, _, count = build(tree)
    if count != n:
        raise ValueError("Cophenetic topology leaves must be unique.")
    return np.asarray(merges, dtype=float)

def children_leaves(linkage, node):
    n = len(linkage) + 1

    def leaves(start):
        pending, out = [start], []
        while pending:
            i = pending.pop()
            if i < n:
                out.append(i)
            else:
                pending.extend(linkage[i - n, :2].astype(int))
        return out

    a, b = linkage[node - n, :2].astype(int)
    return leaves(a), leaves(b)

# SOURCE: reports/subtree_clustering_research_20261109/calibration/run.py
"""Small development panel, separate from any final validation gate."""

import hashlib
import json
import sys
from pathlib import Path
from time import perf_counter

import numpy as np
from scipy.stats import binomtest
from threadpoolctl import threadpool_limits

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[2]
sys.path.insert(0, str(ROOT))
from benchmarks.diagnostics.calibration.selected.hierarchy import (
    subtree_family_reference as family,
    subtree_gaussian_reference as gaussian,
    subtree_statistic_attribution as attribution,
    subtree_topology_tests as topology,
)

FACTOR = np.array([[1, 0.55, 0.1], [0, 0.8, 0.25], [0, 0, 1.3]])
CELLS = (
    ("attribution_null", 16, 3, 0, ()),
    ("attribution_signal", 16, 3, 6, ()),
    ("fixed_beta", 6, 2, 0, ()),
    ("family_null", 4, 2, 0, ()),
    ("family_selected_child", 8, 2, 14, (0,)),
    ("family_mixed_root", 8, 2, 4, ()),
)


def normalized_score(matrix, rows, left, right):
    if len(rows) == 2:
        return 0.0
    local = matrix[rows, 0]
    denom = ((local - local.mean()) ** 2).sum()
    delta = matrix[left, 0].mean() - matrix[right, 0].mean()
    return float(delta**2 / ((1 / len(left) + 1 / len(right)) * denom))


def beta_score(matrix):
    center = matrix - matrix.mean(axis=0)
    u, _, _ = np.linalg.svd(center, full_matrices=False)
    e = np.repeat([1.0, -1.0], 3) / np.sqrt(6)
    return float((u[:, 0] @ e) ** 2)


def trial(i, replicate):
    name, n, p, shift, path = CELLS[i]
    data_seed, chain_seed = np.random.SeedSequence([7811260911, i, replicate]).spawn(2)
    seed = int(chain_seed.generate_state(1, dtype=np.uint64)[0])
    factor = FACTOR[:p, :p]
    x = np.random.default_rng(data_seed).normal(size=(n, p)) @ factor
    means = np.zeros(n)
    if shift:
        x[n // 2 :, 0] += shift
        means[n // 2 :] = shift
    out = dict(
        cell=name,
        replicate=replicate,
        seed=seed,
        data_sha256=hashlib.sha256(x.tobytes()).hexdigest(),
    )
    if i < 2:
        try:
            result = attribution.native_sibling_attribution(
                x, range(n), oracle_covariance=factor.T @ factor, gram_rank=1
            )
        except (ValueError, RuntimeError, FloatingPointError, np.linalg.LinAlgError) as error:
            return out | {"attribution": None, "error": f"{type(error).__name__}: {error}"}
        result["ridge_scaling_residual"] = (
            result["unit_branch_raw"] - result["branch_factor"] * result["native_raw"]
        )
        result["unit_branch_rank_changed"] = result["unit_branch_df"] != result["native_df"]
        if result["native_display"] is not None:
            np.testing.assert_allclose(
                result["native_display"], result["annotated_display"], rtol=1e-12
            )
        return out | {"attribution": result}
    if i == 2:
        exact = gaussian.fixed_subtree_contrast_test(
            x, subtree_indices=range(n), left_indices=range(3), rank=1
        )
        np.testing.assert_allclose(exact["statistic"], beta_score(x), atol=1e-13)
        out["beta_p"] = exact["p_value"]
    comparisons = []
    for steps in (99, 999):
        start = perf_counter()
        if i == 2:
            r = gaussian.subtree_gaussian_chain_test(
                x, subtree_indices=range(n), score=beta_score, steps=steps, seed=seed
            )
            record = dict(p_value=r["p_value"], decision=r["decision"], moves=r["accepted_moves"])
        else:
            r = family.ward_subtree_family_test(
                x, path=path, score=normalized_score, steps=steps, seed=seed
            )
            nodes = []
            for node in r["nodes"]:
                indices = node["subtree_indices"]
                is_null = np.ptp(means[indices]) == 0
                equal_contrast = means[node["left"]].mean() == means[node["right"]].mean()
                nodes.append(
                    dict(
                        rows=indices,
                        null=bool(is_null),
                        equal_contrast=bool(equal_contrast),
                        decision=node["family_decision"],
                        p_value=node["p_value"],
                        holm=node["holm_p_value"],
                        moves=node["accepted_moves"],
                    )
                )
            record = dict(
                decision=r["decision"],
                nodes=nodes,
                hypotheses=r.get("hypotheses"),
                minimum_steps=r.get("minimum_steps_for_holm"),
                resolution=r.get("resolution_sufficient"),
            )
        comparisons.append(record | dict(steps=steps, seconds=perf_counter() - start))
    return out | {"comparisons": comparisons}


def interval(events):
    n = len(events)
    hits = sum(events)
    ci = binomtest(hits, n).proportion_ci()
    return dict(hits=hits, n=n, ci=[ci.low, ci.high])


def summarize(rows):
    summary = {}
    for name, *_ in CELLS:
        panel = [r for r in rows if r["cell"] == name]
        assert [r["replicate"] for r in panel] == list(range(50))
        if name.startswith("attribution"):
            a = [r["attribution"] for r in panel if r["attribution"] is not None]
            summary[name] = dict(
                n=50,
                unresolved=50 - len(a),
                unit_branch_rank_changes=sum(r["unit_branch_rank_changed"] for r in a),
                max_abs_ridge_scaling_residual=max(abs(r["ridge_scaling_residual"]) for r in a),
                display_supported=sum(r["native_display"] is not None for r in a),
                medians={
                    k: float(np.median([r[k] for r in a if r[k] is not None]))
                    for k in a[0]
                    if k not in ("child_sizes", "annotated_display")
                    and any(r[k] is not None for r in a)
                },
            )
            continue
        if name == "fixed_beta":
            stats = dict(beta=interval([r["beta_p"] <= 0.05 for r in panel]))
        else:
            stats = {}
        for j, steps in enumerate((99, 999)):
            runs = [r["comparisons"][j] for r in panel]
            st = dict(
                rejected=interval([r["decision"] == "reject" for r in runs]),
                unresolved=sum(r["decision"] == "unresolved" for r in runs),
                seconds_median=float(np.median([r["seconds"] for r in runs])),
            )
            if name == "fixed_beta":
                st["median_absolute_p_difference"] = float(
                    np.median([abs(a["beta_p"] - b["p_value"]) for a, b in zip(panel, runs)])
                )
            else:
                for null, endpoint in ((True, "false_family"), (False, "true_family")):
                    st[endpoint] = interval(
                        [
                            any(n["null"] == null and n["decision"] == "reject" for n in r["nodes"])
                            for r in runs
                        ]
                    )
                st.update(
                    resolution_failures=sum(not r["resolution"] for r in runs),
                    median_moves=float(np.median([n["moves"] for r in runs for n in r["nodes"]])),
                    unresolved_nodes=sum(
                        n["decision"] == "unresolved" for r in runs for n in r["nodes"]
                    ),
                    heterogeneous_zero_contrast=sum(
                        not n["null"] and n["equal_contrast"] for r in runs for n in r["nodes"]
                    ),
                )
            stats[str(steps)] = st
        summary[name] = stats
    return summary


def main():
    files = [Path(mod.__file__) for mod in (family, gaussian, attribution, topology)] + [
        Path(__file__),
        HERE / "protocol.md",
    ]
    hashes = {str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest() for p in files}
    rows = []
    with threadpool_limits(limits=1), (HERE / "results.jsonl").open("x") as stream:
        for i, (name, *_) in enumerate(CELLS):
            for j in range(50):
                result = trial(i, j)
                rows.append(result)
                stream.write(json.dumps(result, allow_nan=False) + "\n")
                stream.flush()
            print(name, flush=True)
    assert all(hashlib.sha256((ROOT / p).read_bytes()).hexdigest() == h for p, h in hashes.items())
    (HERE / "summary.json").write_text(json.dumps(summarize(rows), indent=2) + "\n")
    (HERE / "manifest.json").write_text(
        json.dumps(dict(sha256=hashes, attempts=len(rows)), indent=2) + "\n"
    )


if __name__ == "__main__":
    main()


# SOURCE: reports/subtree_clustering_research_20261109/calibration/variant.py
"""Use the same conditional trajectory for each declared score control."""

import json
from time import perf_counter

import numpy as np
from threadpoolctl import threadpool_limits

from run import FACTOR, HERE, attribution, gaussian, topology

KEYS = ("native_raw", "unit_branch_raw", "oracle_covariance_raw", "gram_rank_raw", "native_display")


def run(i, steps):
    data, seed = np.random.SeedSequence([7811260911, i, 0]).spawn(2)
    seed = int(seed.generate_state(1, dtype=np.uint64)[0])
    x = np.random.default_rng(data).normal(size=(16, 3)) @ FACTOR
    if i:
        x[8:, 0] += 6
    rows, event = topology.ward_subtree_selection(x, keep_split=True)
    scored = []

    def score(matrix):
        evidence = attribution.native_sibling_attribution(
            matrix, rows, oracle_covariance=FACTOR.T @ FACTOR, gram_rank=1
        )
        scored.append({k: evidence[k] for k in KEYS})
        return evidence["native_raw"]

    start = perf_counter()
    r = gaussian.subtree_gaussian_chain_test(
        x, subtree_indices=rows, score=score, selection_event=event, seed=seed, steps=steps
    )
    out = dict(cell=i, steps=steps, seed=seed, seconds=perf_counter() - start, result=r)
    if r["status"] != "complete":
        return out
    anchor = r["anchor_index"]
    cursor = iter(scored)
    observed = next(cursor)
    states = [None] * (steps + 1)
    states[anchor] = observed
    for positions in (range(anchor - 1, -1, -1), range(anchor + 1, steps + 1)):
        current = observed
        for j in positions:
            if r["chain_moves"][j] is not None:
                current = next(cursor)
            states[j] = current
    assert next(cursor, None) is None
    assert [s["native_raw"] for s in states] == r["chain_scores"]
    ranks = {}
    for k in KEYS:
        if any(s[k] is None for s in states):
            ranks[k] = None
        else:
            ranks[k] = sum(s[k] >= observed[k] for s in states) / len(states)
    assert ranks["native_raw"] == r["p_value"]
    return out | dict(score_states=states, rank_p_values=ranks)


if __name__ == "__main__":
    with threadpool_limits(limits=1), (HERE / "variant.jsonl").open("x") as stream:
        for i in range(2):
            for steps in (99, 999):
                row = run(i, steps)
                stream.write(json.dumps(row, allow_nan=False) + "\n")
                stream.flush()
                print(i, steps, row.get("rank_p_values"), row["seconds"], flush=True)

"""Run the frozen bounded coverage and six-leaf perturbation checks."""

import json
import sys
from collections import Counter
from pathlib import Path

import numpy as np
from scipy.cluster.hierarchy import linkage
from scipy.linalg import helmert
from scipy.spatial.distance import pdist
from threadpoolctl import threadpool_limits

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parents[2]))

from benchmarks.diagnostics.calibration.selected.hierarchy.subtree_topology_candidate import (
    select_subtree_topology,
    topology_from_linkage,
)
from benchmarks.diagnostics.calibration.selected.hierarchy.subtree_topology_geometry import (
    spectral_subspaces,
    ward_topology_family,
)
from tree_break_selection.tree.optimized_branch_lengths import (
    BRANCH_LENGTH_TARGET_MEAN_SQUARED_STANDARDIZED_EUCLIDEAN,
)


def bounded():
    records = []
    for n, p in ((128, 12), (240, 12), (240, 72), (256, 4)):
        x = np.random.default_rng([2026110911, n, p]).normal(size=(n, p))
        current_z = linkage(x, method="ward")
        current = topology_from_linkage(current_z)
        for metric in ("euclidean", BRANCH_LENGTH_TARGET_MEAN_SQUARED_STANDARDIZED_EUCLIDEAN):
            row = {"n": n, "p": p, "metric": metric}
            try:
                result = select_subtree_topology(
                    x, current_topology=current, current_linkage=current_z, target_metric=metric
                )
                current_fit = next(c.fit for c in result.candidates if c.topology == current)
                selected_fit = next(
                    c.fit for c in result.candidates if c.topology == result.selected_topology
                )
                row.update(
                    status="ok",
                    count=result.candidate_count,
                    spans=result.spectral_span_count,
                    pairs=result.pair_count,
                    coverage=result.coverage,
                    elapsed_sec=result.elapsed_sec,
                    changed=result.selected_topology != current,
                    current_rmse=current_fit.rmse,
                    selected_rmse=selected_fit.rmse,
                )
            except (ValueError, RuntimeError, FloatingPointError, np.linalg.LinAlgError) as error:
                row.update(status="failed", error=f"{type(error).__name__}: {error}")
            records.append(row)
            print(json.dumps(row), flush=True)
    return records


def canonical_ward(values):
    distances = pdist(values)
    maximum = distances.max()
    scaled = np.round(distances / maximum, 12) if maximum > 0 else distances
    return topology_from_linkage(linkage(scaled, method="ward"))


def resampling():
    row = np.repeat([-3.0, 3.0], 3)
    column = np.tile([-2.0, 0.0, 2.0], 2)
    alternating = 0.15 * np.tile([-1.0, 1.0], 3)
    x = np.column_stack([row, row, column, column, alternating])
    family = set().union(*(ward_topology_family(s.coordinates) for s in spectral_subspaces(x)))
    mean = x.mean(axis=0)
    basis = helmert(len(x), full=False)
    centered = basis @ (x - mean)
    rms = float(np.sqrt(np.mean((x - mean) ** 2)))
    records = []
    for index, regime in enumerate(
        ("feature_bootstrap", "small_noise", "large_noise", "row_control", "gaussian_orbit")
    ):
        rng = np.random.default_rng([2026110912, index])
        counts = Counter()
        for _ in range(2000):
            if regime == "feature_bootstrap":
                data = x[:, rng.integers(x.shape[1], size=x.shape[1])]
            elif regime in {"small_noise", "large_noise"}:
                sigma = rms * (0.05 if regime == "small_noise" else 0.5)
                data = x + rng.normal(scale=sigma, size=x.shape)
            elif regime == "row_control":
                order = rng.permutation(len(x))
                data = x[order][np.argsort(order)]
            else:
                rotation, upper = np.linalg.qr(rng.normal(size=(len(basis), len(basis))))
                rotation *= np.sign(np.diag(upper))
                data = mean + basis.T @ rotation @ centered
            counts[canonical_ward(data)] += 1
        record = {
            "regime": regime,
            "replicates": sum(counts.values()),
            "distinct": len(counts),
            "outside_replicates": sum(v for k, v in counts.items() if k not in family),
            "outside_distinct": sum(k not in family for k in counts),
            "counts": [(k, v) for k, v in sorted(counts.items(), key=lambda kv: repr(kv[0]))],
        }
        records.append(record)
        print(json.dumps({k: v for k, v in record.items() if k != "counts"}), flush=True)
    return {
        "baseline_family_count": len(family),
        "baseline_family": sorted(family, key=repr),
        "baseline_full_ward": canonical_ward(x),
        "regimes": records,
    }


if __name__ == "__main__":

        result = {"coverage": bounded, "resampling": resampling}[command]()
    (HERE / f"{command}.json").write_text(json.dumps(result, indent=2, allow_nan=False) + "\n")
