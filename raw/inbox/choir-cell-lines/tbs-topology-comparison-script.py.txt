"""Peel structure and cell-line subtrees of the four saved cancer topologies.

Run from results/analyses/ on 2026-11-06 (inline; captured verbatim apart from this
docstring). Reads the saved linkages; no tree is rebuilt.
"""

import numpy as np, pandas as pd
from scipy.cluster.hierarchy import fcluster
from sklearn.metrics import adjusted_rand_score
meta=pd.read_csv("choir_cell_lines_tbs_merged_20261106/cell_metadata.csv",index_col=0,dtype=str)
lines=meta.Cell_line.to_numpy(); codes,uniq=pd.factorize(lines); L=len(uniq); n=len(lines)
nline=np.bincount(codes,minlength=L)
trees={"symmetric_average (TBS run)":"choir_cell_lines_tbs_merged_20261106/linkage.npy",
 "symmetric_ward":"choir_tbs_recovery_comparison_20261103_174622/symmetric_ward/linkage.npy",
 "markov_average":"choir_tbs_recovery_comparison_20261103_174622/markov_average/linkage.npy",
 "markov_ward":"choir_tbs_recovery_comparison_20261103_174622/markov_ward/linkage.npy"}
rows=[]
for name,path in trees.items():
    z=np.load(path); assert len(z)==n-1
    kids=z[:,:2].astype(int); size=np.r_[np.ones(n,int),z[:,3].astype(int)]
    cnt=np.zeros((2*n-1,L),np.int32); cnt[np.arange(n),codes]=1
    for i,(a,b) in enumerate(kids): cnt[n+i]=cnt[a]+cnt[b]
    small=np.minimum(size[kids[:,0]],size[kids[:,1]])
    big=size[n:]>=1000
    # spine: follow the larger child from the root while the smaller child has <=10 leaves
    node=2*n-2; spine=0
    while node>=n:
        a,b=kids[node-n]; s=min(size[a],size[b])
        if s>10: break
        spine+=1; node=a if size[a]>size[b] else b
    recall=cnt/nline; purity=cnt/size[:,None]
    clean=((recall>=0.9)&(purity>=0.9)).any(axis=0)
    f1=(2*recall*purity/(recall+purity+1e-300)).max(axis=0)
    ari190=adjusted_rand_score(lines,fcluster(z,190,criterion="maxclust"))
    rows.append(dict(topology=name,root_children=f"{small[-1]}/{size[-1]-small[-1]}",
        nodes_ge_1000=int(big.sum()),frac_peel_le10=round(float((small[big]<=10).mean()),3),
        top_spine_peels=spine,spine_end_size=int(size[node]),
        lines_with_clean_subtree=int(clean.sum()),median_best_f1=round(float(np.median(f1)),3),
        ari_cut_190=round(ari190,3)))
    if name.startswith("symmetric_average"):
        sa_clean=clean; sa_cnt=cnt; sa_size=size; sa_kids=kids
print(pd.DataFrame(rows).to_string(index=False))
# where do the clean subtrees of the symmetric/average tree sit relative to the giant cluster?
lab=pd.read_csv("choir_cell_lines_tbs_merged_20261106/cluster_assignments.csv").TBS_cluster.to_numpy()
giant=pd.Series(lab).value_counts().index[0]
dom=pd.crosstab(lines,lab).idxmax(axis=1)
in_giant=dom.reindex(uniq).to_numpy()==giant
print("\nsymmetric/average: lines dominated by the giant cluster:",int(in_giant.sum()),"; of these with a clean subtree in the tree:",int((in_giant&sa_clean).sum()))
