def digest(value):
    return hashlib.sha256(np.ascontiguousarray(value).tobytes()).hexdigest()


def sibling_hook(*args):
    result = harness.swapped(*args)
    if harness.STATE["arm"] == "C":
        splits = iter(harness.STATE["splits"])
        for i, decision in enumerate(result[2]):
            if decision is None:
                continue
            split = next(splits)
            p = audit.sup_p(split["rho"], split["n_parent"], split["k"])
            assert np.isfinite(p) and 0 <= p <= 1
            result[0][i] = (*result[0][i][:2], p)
            result[2][i] = replace(decision, p_value=p)
            split["p_B"], split["p"] = split["p"], p
        assert next(splits, None) is None
    return result


harness.pipeline.apply_split_count_rule_out = sibling_hook


def task(job, arms=ARMS):
    model, replicate = job
    rows = []
    base = {"model": model, "replicate": replicate}
    start = perf_counter()
    try:
        with threadpool_limits(limits=1), parallel_backend("threading", n_jobs=1):
            counts, truth = depth.generate(model, replicate)
            base.update(n=len(truth), true_k=len(np.unique(truth)), count_sha256=digest(counts))
            prepared = prepare(counts)
            data = prepared[0]
            base.update(pca_sha256=digest(data.to_numpy()), preparation_sec=perf_counter()-start)
            for method, coordinate_index in METHODS.items():
                distances = pdist(prepared[coordinate_index])
                geometry_sha = digest(distances)
                topology_sha = None
                params = dict(METHOD_SPECS[method].param_grid[0])
                for arm in arms:
                    row = dict(base, method=method, arm=arm, geometry_sha256=geometry_sha)
                    tick = perf_counter()
                    try:
                        harness.STATE.update(arm=arm, splits=[])
                        result = run_tbs_on_distance(
                            data, distances, .01, edge_alpha=.001,
                            tree_linkage_method=params["tree_linkage_method"],
                            feature_space=continuous_feature_space_from_columns(tuple(data.columns)),
                            branch_length_data_df=data,
                            edge_branch_length_variance_policy=params["edge_branch_length_variance_policy"],
                            adaptive_projection_dimension_energy_fraction=.90,
                            sibling_split_count_rule_out=arm != "A",
                            **_tbs_branch_length_optimization_kwargs(params),
                            extra={"feature_family_diffusion_fallback_reason": None})
                        row["status"] = result.status.value
                        assert row["status"] == "ok", row["status"]
                        labels = np.asarray(result.labels, dtype=int)
                        assert labels.shape == truth.shape and len(np.unique(labels)) == result.found_clusters
                        z = result.extra["linkage_matrix"]
                        sha = digest(z)
                        if topology_sha is None:
                            topology_sha = sha
                        assert sha == topology_sha, "An arm changed the shared linkage"
                        sizes = np.r_[np.ones(len(truth)), z[:, 3]]
                        score, _ = metric_module.metrics(truth, labels, [])
                        row.update(score, linkage_sha256=sha,
                                   root_small=int(sizes[z[-1, :2].astype(int)].min()),
                                   false_partition=bool(len(np.unique(truth)) == 1 and result.found_clusters > 1),
                                   labels=labels.tolist(), truth=truth.tolist(), splits=harness.STATE["splits"])
                    except Exception as error:
                        row.update(status="exception", error=f"{type(error).__name__}: {error}",
                                   traceback=traceback.format_exc())
                    row["fit_sec"] = perf_counter()-tick
                    rows.append(row)
    except Exception as error:
        rows = [dict(base, method=method, arm=arm, status="preparation_exception",
                     error=f"{type(error).__name__}: {error}", traceback=traceback.format_exc())
                for method in METHODS for arm in arms]
    assert len(rows) == len(METHODS)*len(arms)
    return rows


def check():
    old = [json.loads(line) for line in (ANALYSES / "topology_correction_20261107/topology.jsonl").read_text().splitlines()]
    mapping = {"tbs_family_diffusion": "symmetric_average", "tbs_family_diffusion_markov_ward": "markov_ward"}
    matches = []
    for model in MODELS[:3]:
        np.testing.assert_array_equal(depth.generate(model, 0)[0], depth.generate(model, 0)[0])
        for row in task((model, 0), arms=("A",)):
            target = next(r for r in old if r["model"] == model and r["replicate"] == 0 and r["tree"] == mapping[row["method"]])
            assert row["status"] == target["status"] == "ok", row
            assert row["found_clusters"] == target["found_clusters"] and row["root_small"] == target["root_small"], (row, target)
            matches.append({k: row[k] for k in ("model", "method", "status", "found_clusters", "root_small", "count_sha256", "pca_sha256")})
    (HERE / "replay.json").write_text(json.dumps(matches, indent=2))
    print(f"{len(matches)}/6 original count/topology replays matched", flush=True)


def run():
    jobs = [(model, replicate) for model in MODELS
            for replicate in range(10000, 10000+(5 if model.startswith("signal") else 30))]
    assert len(jobs) == 100 and len(set(jobs)) == 100
    with (HERE / "results.jsonl").open("x") as stream, Pool(2) as pool:
        for i, rows in enumerate(pool.imap_unordered(task, jobs), 1):
            for row in rows:
                stream.write(json.dumps(row, default=float)+"\n")
            stream.flush()
            if i % 5 == 0:
                print(f"{i}/{len(jobs)} datasets, {i*8} fit attempts", flush=True)


def summary():
    records = [json.loads(line) for line in (HERE / "results.jsonl").read_text().splitlines()]
    assert len(records) == 800
    assert len({(r["model"], r["replicate"], r["method"], r["arm"]) for r in records}) == 800
    data = pd.DataFrame([{k:v for k,v in r.items() if k not in ("labels", "truth", "splits", "traceback")} for r in records])
    data.to_csv(HERE / "fit_metrics.csv", index=False)
    rows = []
    for keys, frame in data.groupby(["model", "method", "arm"], sort=False):
        good = frame.status == "ok"
        hits = int(frame.loc[good, "false_partition"].sum())
        out = dict(zip(("model", "method", "arm"), keys), attempts=len(frame),
                   ok=int(good.sum()), failed=int((~good).sum()), false_partitions=hits)
        if not keys[0].startswith("signal"):
            ci = binomtest(hits, len(frame)).proportion_ci()
            out.update(rate_lower_accounting=hits/len(frame), rate_upper_accounting=(hits+(~good).sum())/len(frame),
                       ci_low=ci.low, ci_high=ci.high,
                       upper_95_one_sided=binomtest(hits, len(frame), alternative="less").proportion_ci().high)
        for col in ("found_clusters", "ari", "ami", "macro_f1", "homogeneity", "completeness", "fit_sec", "preparation_sec", "root_small"):
            out[col+"_mean"] = float(frame.loc[good, col].mean())
        rows.append(out)
    pd.DataFrame(rows).to_csv(HERE / "summary.csv", index=False)
    print(json.dumps({"attempts": len(data), "status": data.status.value_counts().to_dict(), "rows": len(rows)}))

"""Deterministic root geometry checks. Run with .venv/bin/python from the project."""
import json

import numpy as np
from scipy.optimize import linprog
from scipy.stats import beta, binomtest, f, kstest

SEED = 20261109
REPS = 12000
rng = np.random.default_rng(SEED)


def rank_projected_energy(x, contrast):
    """Return the squared projection contributions of a unit contrast on left singular vectors."""
    x = x - x.mean(axis=-2, keepdims=True)
    vectors, singular, _ = np.linalg.svd(x, full_matrices=False)
    return np.square(contrast @ vectors), singular


def rate(p_values, alpha):
    hits = int(np.count_nonzero(p_values <= alpha))
    interval = binomtest(hits, len(p_values)).proportion_ci()
    return [hits, len(p_values), float(interval.low), float(interval.high)]


def fixed_and_adaptive_rank():
    n, p = 25, 6
    contrast = np.r_[np.full(8, 1 / 8), np.full(n - 8, -1 / (n - 8))]
    contrast /= np.linalg.norm(contrast)
    x = rng.standard_normal((REPS, n, p)) * np.sqrt([7, 4, 2, 1, 0.7, 0.4])
    energies, singular = rank_projected_energy(x, contrast)
    cumulative = np.cumsum(energies, axis=1)
    eigenvalues = np.square(singular)
    spectrum_rank = 1 + np.argmax(
        np.cumsum(eigenvalues, axis=1) >= 0.75 * eigenvalues.sum(axis=1, keepdims=True),
        axis=1,
    )
    contrast_rank = 1 + np.argmax(cumulative >= 0.90 * cumulative[:, -1:], axis=1)

    def tails(ranks):
        r_squared = cumulative[np.arange(REPS), ranks - 1]
        return beta.sf(r_squared, ranks / 2, (n - 1 - ranks) / 2)

    fixed = tails(np.full(REPS, p))
    r_squared = cumulative[:, -1]
    hotelling = (n - 1 - p) / p * r_squared / (1 - r_squared)
    assert np.allclose(fixed, f.sf(hotelling, p, n - 1 - p), atol=1e-14)
    spectral = tails(spectrum_rank)
    selected = tails(contrast_rank)
    groups = {}
    for rank in np.unique(spectrum_rank):
        selected_group = spectral[spectrum_rank == rank]
        groups[int(rank)] = {
            "replicates": len(selected_group),
            "KS_distance": float(kstest(selected_group, "uniform").statistic),
            "at_0.05": rate(selected_group, 0.05),
        }

    # Haar left rotations preserve the same fixed feature Gram, including an exact tie.
    n_tied, p_tied, rank = 12, 4, 2
    centered_basis = np.linalg.qr(np.ones((n_tied, 1)), mode="complete")[0][:, 1:]
    u = np.r_[np.full(4, 1 / 4), np.full(8, -1 / 8)]
    u /= np.linalg.norm(u)
    q, _ = np.linalg.qr(rng.standard_normal((REPS, n_tied - 1, p_tied)))
    tied_singular = np.array([3.0, 3.0, 1.0, 0.5])
    tied_x = centered_basis @ q * tied_singular
    assert np.allclose(np.swapaxes(tied_x, -1, -2) @ tied_x, np.diag(tied_singular**2))
    tied_r2 = np.sum(np.square(u @ (tied_x[:, :, :rank] / tied_singular[:rank])), axis=1)
    tied_p = beta.sf(tied_r2, rank / 2, (n_tied - 1 - rank) / 2)
    assert kstest(spectral, "uniform").statistic < 0.02
    assert kstest(tied_p, "uniform").statistic < 0.02

    return {
        "fixed_rank": {"at_0.01": rate(fixed, 0.01), "at_0.05": rate(fixed, 0.05)},
        "spectrum_rank": {"at_0.01": rate(spectral, 0.01), "at_0.05": rate(spectral, 0.05),
                          "by_rank": groups},
        "contrast_rank": {"at_0.01": rate(selected, 0.01), "at_0.05": rate(selected, 0.05),
                           "KS_distance": float(kstest(selected, "uniform").statistic)},
        "fixed_Gram_tied_eigenblock": {"at_0.01": rate(tied_p, 0.01),
                                     "at_0.05": rate(tied_p, 0.05)},
    }


def unit_contrasts(n):
    """All nonempty splits with the first cell on the left, identifying complements."""
    bits = np.arange(2 ** (n - 1) - 1)[:, None] >> np.arange(n - 1)
    left = np.column_stack([np.ones(len(bits), dtype=bool), (bits & 1).astype(bool)])
    share = left.mean(axis=1, keepdims=True)
    return left, (left - share) / np.sqrt(n * share * (1 - share))


def exhaustive_check():
    n, k = 8, 2
    left, contrasts = unit_contrasts(n)
    x = rng.standard_normal((REPS, n, k))
    centered = x - x.mean(axis=1, keepdims=True)
    basis, _ = np.linalg.qr(centered)
    r_squared = np.square(contrasts @ basis).sum(axis=-1)
    best = r_squared.argmax(axis=1)
    maximum = r_squared[np.arange(REPS), best]
    union = np.minimum(1, len(left) * beta.sf(maximum, k / 2, (n - 1 - k) / 2))
    flat_counts = []
    for sample, winner in zip(x[:4], best[:4], strict=True):
        design = np.column_stack([sample, np.ones(n)])
        flat = np.array([
            linprog(np.zeros(k + 1), A_ub=-(2 * side[:, None].astype(int) - 1) * design,
                    b_ub=-np.ones(n), bounds=(None, None), method="highs").success
            for side in left
        ])
        assert flat[winner]
        flat_counts.append(int(flat.sum()))
    assert flat_counts == [n - 1 + (n - 1) * (n - 2) // 2] * 4

    saturation = {}
    for ambient in (n - 1, n, n + 4):
        saturated_x = rng.standard_normal((n, ambient))
        saturated_x -= saturated_x.mean(axis=0)
        full_basis = np.linalg.svd(saturated_x, full_matrices=False)[0][:, :n - 1]
        error = np.max(np.abs(np.square(contrasts @ full_basis).sum(axis=1) - 1))
        assert error < 1e-12
        saturation[ambient] = float(error)

    limiting_cutoff = {}
    for size in (30, 100, 300, 900):
        cutoff = beta.isf(0.01 / (2 ** (size - 1) - 1), k / 2, (size - 1 - k) / 2)
        limiting_cutoff[size] = [float(cutoff), float((size - 2) / size * cutoff / (1 - cutoff))]
    return {
        "all_unordered_splits": len(left),
        "exact_flat_counts_first_four_samples": flat_counts,
        "exhaustive_max_Bonferroni_at_0.01": rate(union, 0.01),
        "exhaustive_max_Bonferroni_at_0.05": rate(union, 0.05),
        "saturation_max_abs_error_by_ambient_dimension": saturation,
        "Bonferroni_cutoff_R_squared_and_rho": limiting_cutoff,
    }

def generate(model, replicate):
    if model in FROZEN:
        return validate.generate(list(validate.MODELS).index(FROZEN[model]), replicate)
    if model.startswith("signal_"):
        return validate.generate(list(validate.MODELS).index(model), replicate)
    tag, sigma = MATCHED[model]
    rng = np.random.default_rng(np.random.SeedSequence([20261107, tag, replicate]))
    labels = np.zeros(400, dtype=int)
    means = np.exp(rng.normal(1., 1., 3000))
    depth = rng.lognormal(-sigma**2 / 2., sigma, 400)

    return counts, labels


def baseline_pcs(counts):
    totals = counts.sum(axis=1)
    a = ad.AnnData(sparse.csr_matrix(counts, dtype=np.float64))
    a.X = sparse.diags(10000. / totals) @ a.X
    sc.pp.log1p(a)
    sc.pp.highly_variable_genes(a, n_top_genes=2000, flavor="seurat")
    a = a[:, a.var.highly_variable].copy()
    sc.pp.scale(a, max_value=10)
    sc.pp.pca(a, n_comps=30, svd_solver="arpack", random_state=0)
    return np.asarray(a.obsm["X_pca"], dtype=np.float64)


def pearson_pcs(counts):
    a = ad.AnnData(sparse.csr_matrix(counts, dtype=np.float64))
    sc.experimental.pp.highly_variable_genes(a, flavor="pearson_residuals", n_top_genes=2000)
    a = a[:, a.var.highly_variable].copy()
    sc.experimental.pp.normalize_pearson_residuals(a, theta=100)
    sc.pp.pca(a, n_comps=30, svd_solver="arpack", random_state=0)
    return np.asarray(a.obsm["X_pca"], dtype=np.float64)


def regress_depth(pcs, counts):
    design = np.column_stack([np.ones(len(pcs)), np.log(counts.sum(axis=1))])
    beta, *_ = np.linalg.lstsq(design, pcs, rcond=None)
    return pcs - design @ beta


def downsample(counts, replicate):
    rng = np.random.default_rng(np.random.SeedSequence([20261107, 9, replicate]))
    totals = counts.sum(axis=1)
    target = np.quantile(totals, 0.10)
    keep = np.minimum(1.0, target / totals)
    return rng.binomial(counts, keep[:, None])


def geometry(pcs):
    """The frozen `prepare` kernel and symmetric coordinates for given PCs."""
    k = _resolve_pydiffmap_neighbor_search_k(pcs, k_neighbors=15, bandwidth_type="-1/(d+2)")
    fitted = kernel.Kernel(epsilon=1., k=k, metric="euclidean",
                           neighbor_params={"n_jobs": 1}, bandwidth_type="-1/(d+2)")
    fitted.fit(pcs)
    fitted.scaled_dists.setdiag(0.)
    epsilon, _ = resolve_adaptive_epsilon(fitted, "median", metric="euclidean")
    fitted.epsilon_fitted = epsilon
    w = fitted.compute().tocsr()
    w.setdiag(0.)
    w.eliminate_zeros()
    w = w.maximum(w.T).tocsr()
    symmetric = compute_diffusion_coordinates(w, diffusion_time=3, n_components=30)
    data = pd.DataFrame(pcs, index=[f"cell{i:06d}" for i in range(len(pcs))],
                        columns=[f"PC{i+1}" for i in range(pcs.shape[1])])
    return data, pdist(symmetric)


def arm_pcs(arm, counts, replicate):
    if arm == "baseline":
        return baseline_pcs(counts)
    if arm == "pc_regress":
        return regress_depth(baseline_pcs(counts), counts)
    if arm == "pearson":
        return pearson_pcs(counts)
    return baseline_pcs(downsample(counts, replicate))


def task(job):
    model, replicate = job
    counts, truth = generate(model, replicate)
    sha = hashlib.sha256(counts.tobytes()).hexdigest()
    loglib = np.log(counts.sum(axis=1))
    rows = []
    with threadpool_limits(limits=1):
        for arm in ARMS:
            start = perf_counter()
            row = {"model": model, "replicate": replicate, "arm": arm, "count_sha256": sha}
            try:
                pcs = arm_pcs(arm, counts, replicate)
                row["max_abs_spearman_pc_loglib"] = float(max(
                    abs(pd.Series(pcs[:, j]).corr(pd.Series(loglib), method="spearman"))
                    for j in range(pcs.shape[1])))
                data, distances = geometry(pcs)
                row.update(summarize(fit(data, distances, "symmetric_average"), truth))
            except Exception as error:
                row.update(status="exception", error=f"{type(error).__name__}: {error}")
            row["sec"] = perf_counter() - start
            row.pop("stage_timings", None)
            rows.append(row)
    return rows

def main(which):
    if which == "nulls":
        jobs = [(m, r) for m in ("low", "high", "matched_median", "matched_max") for r in range(300)]
    else:
        jobs = [(m, r) for m in validate.MODELS if m.startswith("signal_")
                for r in range(validate.MODELS[m]["replicates"])]

    done = set()
    if path.exists():

    jobs = [j for j in jobs if j not in done]

def generate(model_index, replicate):
    spec = list(MODELS.values())[model_index]
    rng = np.random.default_rng(np.random.SeedSequence([20261103, model_index, replicate]))
    labels = np.repeat(np.arange(len(spec["sizes"])), spec["sizes"])
    means = np.exp(rng.normal(1., 1., 3000))
    profiles = np.tile(means, (len(spec["sizes"]), 1))
    if len(profiles) > 1:
        for group in range(len(profiles)):
            profiles[group, group*60:(group+1)*60] *= np.exp(2.5)
    sigma = spec["library_log_sigma"]
    depth = rng.lognormal(-sigma**2 / 2., sigma, len(labels))
    mu = profiles[labels] * depth[:, None]
    theta = np.asarray(spec["dispersion"])[labels, None]
    counts = rng.negative_binomial(theta, theta / (theta + mu))
    assert np.all(counts.sum(axis=1) > 0)
    return counts, labels


def prepare(counts):
    totals = counts.sum(axis=1)
    a = ad.AnnData(sparse.csr_matrix(counts, dtype=np.float64))
    a.X = sparse.diags(10000. / totals) @ a.X
    sc.pp.log1p(a)
    sc.pp.highly_variable_genes(a, n_top_genes=2000, flavor="seurat")
    a = a[:, a.var.highly_variable].copy()
    sc.pp.scale(a, max_value=10)
    sc.pp.pca(a, n_comps=30, svd_solver="arpack", random_state=0)
    pca = np.asarray(a.obsm["X_pca"], dtype=np.float64)
    k = _resolve_pydiffmap_neighbor_search_k(pca, k_neighbors=15, bandwidth_type="-1/(d+2)")
    fitted = kernel.Kernel(epsilon=1., k=k, metric="euclidean",
        neighbor_params={"n_jobs": -1}, bandwidth_type="-1/(d+2)")
    fitted.fit(pca)
    fitted.scaled_dists.setdiag(0.)
    epsilon, _ = resolve_adaptive_epsilon(fitted, "median", metric="euclidean")
    fitted.epsilon_fitted = epsilon
    w = fitted.compute().tocsr()
    w.setdiag(0.)
    w.eliminate_zeros()
    w = w.maximum(w.T).tocsr()
    symmetric = compute_diffusion_coordinates(w, diffusion_time=3, n_components=30)
    data = pd.DataFrame(pca, index=[f"cell{i:06d}" for i in range(len(counts))],
                        columns=[f"PC{i+1}" for i in range(30)])
    return data, symmetric, markov_coordinates(symmetric, w)


def run(model_names):
    shard = int(sys.argv[2]) if len(sys.argv) > 2 else None
    suffix = "" if shard is None else f"_{shard}"
    path = OUT / ("validation_" + sys.argv[1] + suffix + ".jsonl")
    completed = set()
    for existing in OUT.glob("validation_*.jsonl"):
        for line in existing.read_text().splitlines():
            r = json.loads(line)
            completed.add((r["model"], r["replicate"], r["variant"]))
    with threadpool_limits(limits=2), path.open("a", buffering=1) as output:
        for model in model_names:
            model_index = list(MODELS).index(model)
            for replicate in range(MODELS[model]["replicates"]):
                if shard is not None and replicate % 3 != shard:
                    continue
                if all((model, replicate, v) in completed for v in VARIANTS):
                    continue
                start = perf_counter()
                counts, truth = generate(model_index, replicate)
                fingerprint = hashlib.sha256(counts.tobytes()).hexdigest()
                try:
                    data, symmetric, markov = prepare(counts)
                    prepare_error = None
                except Exception as error:
                    prepare_error = {"type": type(error).__name__, "message": str(error)}
                preparation_sec = perf_counter()-start
                del counts
                for coordinates, variants in ((symmetric if prepare_error is None else None, VARIANTS[:2]),
                                              (markov if prepare_error is None else None, VARIANTS[2:])):
                    distances = pdist(coordinates) if prepare_error is None else None
                    for variant in variants:
                        if (model, replicate, variant) in completed:
                            continue
                        run_start = perf_counter()
                        r = {"model": model, "replicate": replicate, "variant": variant,
                             "seed_entropy": [20261103, model_index, replicate],
                             "count_sha256": fingerprint, "n": len(truth), "true_k": len(np.unique(truth)),
                             "preparation_sec": preparation_sec}
                        try:
                            if prepare_error is not None:
                                raise RuntimeError(str(prepare_error))
                            result = fit(data, distances, variant)
                            r.update(summarize(result, truth))
                            del result
                        except Exception as error:
                            r.update({"status": "exception", "error_type": type(error).__name__, "error": str(error)})
                        r["fit_sec"] = perf_counter()-run_start
                        output.write(json.dumps(r, default=str) + "\n")
                    del distances
                if prepare_error is None:
                    del data, symmetric, markov
                gc.collect()
                if replicate % 10 == 0:
                    print(f"{model} replicate {replicate+1}/{MODELS[model]['replicates']}, {perf_counter()-start:.2f}s", flush=True)

def markov_coordinates(symmetric, weights):
    degrees = np.asarray(weights.sum(axis=1)).ravel()
    if not np.isfinite(degrees).all() or (degrees <= 0).any():
        raise ValueError("Markov variant requires finite, positive kernel degrees")
    return symmetric / np.sqrt(degrees)[:, None]

def summarize(result, truth):
    extra = result.extra or {}
    summary = {
        "status": result.status.value, "skip_reason": result.skip_reason,
        "unsupported_reason": result.unsupported_reason,
        "found_clusters": result.found_clusters,
    }
    summary.update({k: v for k, v in extra.items() if k.startswith("branch_length_")})
    summary["stage_timings"] = extra.get("stage_timings")
    summary["traversal_counters"] = extra.get("traversal_counters")
    z = extra.get("linkage_matrix")
    if z is not None:
        n = len(truth)
        sizes = np.r_[np.ones(n), z[:, 3]]
        root_children = z[-1, :2].astype(int)
        summary["root_child_sizes"] = sizes[root_children].astype(int).tolist()
        top = sizes[z[-min(100, len(z)):, :2].astype(int)].min(axis=1)
        summary["top100_singleton_splits"] = int((top == 1).sum())
        summary["top100_min_child_size_median"] = float(np.median(top))
        summary["root_guarded_covariance_fallback"] = bool(sizes[root_children].min() < 8)
    a = extra.get("annotations")

        labels = np.asarray(result.labels, dtype=int)
        assert labels.shape == (len(truth),) and (labels >= 0).all()
        assert len(np.unique(labels)) == result.found_clusters

        pass
    else:
        raise AssertionError("Zero degrees must not silently change the Markov variant")

def rates(rows, fields):
    out = []

def swapped(tree, leaf_data, records, tested_parent_ids, summaries, method_labels, decisions):
    """rule_out: production as is. ceiling3: p from the at-the-mode ceiling 3, same rows and statistic."""
    by_parent = {record.parent: record for record in records}
    shape = list(leaf_data.shape)
    if STATE["arm"] == "rule_out":
        out = ORIGINAL(tree, leaf_data, records, tested_parent_ids, summaries, method_labels, decisions)
        for parent, label, decision in zip(tested_parent_ids, out[1], out[2], strict=True):
            if decision is not None:
                STATE["splits"].append({"shape": shape, "node": str(parent), "n_parent": int(by_parent[parent].n_parent),
                                        "reason": label.split(":")[-1], "p": float(decision.p_value)})

    row_of_label = {label: row for row, label in enumerate(leaf_data.index)}
    matrix = leaf_data.to_numpy(dtype=np.float64)
    new = ([], [], [])
    for parent, summary, label, decision in zip(tested_parent_ids, summaries, method_labels, decisions, strict=True):
        if decision is None:
            for column, value in zip(new, (summary, label, decision), strict=True):
                column.append(value)
            continue
        record = by_parent[parent]
        left, right = _leaf_rows(tree, record.left, row_of_label), _leaf_rows(tree, record.right, row_of_label)
        points = matrix[left + right]
        in_left = np.r_[np.ones(len(left), bool), np.zeros(len(right), bool)]
        directions = invariant_directions(points)
        p, rho = cut_bound_p(points @ directions.T, in_left, unimodal_bound)
        assert np.isfinite(p), (parent, rho)  # a NaN tail would silently read as p = 1
        statistic = float(record.stat) / float(record.reference_scale)

        new[1].append("split_count_rule_out:ceiling3")  # the FDR step accepts this prefix
        new[2].append(replace(decision, c_hat=1.0, p_value=p))
    return new

def recovered_lines(truth, labels):

# SHA256 b663e02fc7a4ef7b94c4aa0bea3c4f9c4aaa6d440470410dc78d95f7f45c5fdf
"""protocol.md D2-D6: the largest split one population allows, and the p-value that a split goes beyond it."""

import numpy as np
from scipy.special import gammaln, logsumexp
from scipy.stats import ncx2, norm

UNIMODAL_BOUND = 3.0


def gaussian_bound(pi):
    """g(pi): between/within ratio of a flat cut of one Gaussian population into shares pi and 1 - pi."""
    r = norm.pdf(norm.ppf(pi)) ** 2 / (pi * (1 - pi))
    return r / (1 - r)


def unimodal_bound(pi):
    return UNIMODAL_BOUND


def bayes_error(pi, delta):
    """Error of the best rule for two unit Gaussians at separation delta with shares pi and 1 - pi."""
    c = np.log((1 - pi) / pi) / delta  # the boundary moves toward the smaller group
    return float(pi * norm.cdf(c - delta / 2) + (1 - pi) * norm.cdf(-c - delta / 2))


def fisher_ratio(points, in_left):
    """rho = pi (1 - pi) D^2; D = Mahalanobis distance of the side means, pooled within covariance (n - 2)."""
    left, right = points[in_left], points[~in_left]
    n, pi = len(points), len(left) / len(points)
    centered = np.vstack([left - left.mean(0), right - right.mean(0)])
    d = left.mean(0) - right.mean(0)
    return float(pi * (1 - pi) * d @ np.linalg.solve(centered.T @ centered / (n - 2), d))


def log_flat_cuts(n, k):
    """log sum_{i<=k} binom(n-1, i): splits of n points in general position by a flat cut in k dimensions."""
    i = np.arange(0, min(k, n - 1) + 1)
    return float(logsumexp(gammaln(n) - gammaln(i + 1) - gammaln(n - i)))


def cut_bound_p(points, in_left, bound):

    rho = fisher_ratio(points, in_left)
    log_p = log_flat_cuts(n, k) + ncx2.logsf(n * rho, k, n * bound(float(in_left.mean())))
    return float(np.exp(min(0.0, log_p))), rho

"""protocol.md: tail of the best flat cut of one population, in place of the count of all flat cuts.

Null: n cells from one population F whose best flat cut has ratio rho* = b. The tree's split has ratio at
most the best flat cut's (D1), so p = P(sup over flat cuts of rho_hat >= rho_obs). To first order the sup sits
on the population's best cuts (envelope argument; Hartigan 1978, Pollard 1982): in whitened coordinates
R_hat(v) = mean |v'x| / sd(v'x) at the cut through the centre, R^2 = rho/(1 + rho), and

    sqrt(n) (rho_hat(v) - b) ~ sigma_rho W(v),  W a unit-variance even Gaussian field on the sphere S^{k-1}.

Its sup tail is the expected Euler characteristic (Adler and Taylor):
    p = 1/2 sum_j L_j(S^{k-1} of radius sqrt(lambda)) rho_j(u),  u = sqrt(n) (rho_obs - b) / sigma_rho,
halved because W(v) = W(-v). Constants (derivations in protocol.md):
    Gaussian F:  b = 2/(pi - 2) = 1.752, sigma_rho^2 = a^2 (1 - 3/pi) = 6.582, lambda = (1 - 2/pi)/(1 - 3/pi) = 8.063
    ceiling 3 (uniform marginal at the cut): b = 3, sigma_rho^2 = a^2 / 40 = 19.2, lambda = 13.0 (isotropic law with
    uniform marginals, which exists for k = 3; used for every k, an assumption)
with a = d rho / d R = 2 R* / (1 - R*^2)^2.
"""

from functools import lru_cache

import numpy as np
from scipy.special import gammaln
from scipy.stats import norm

SHAPES = {
    "gaussian": {"b": 2 / (np.pi - 2), "R": np.sqrt(2 / np.pi), "var_R": 1 - 3 / np.pi, "lam": (1 - 2 / np.pi) / (1 - 3 / np.pi)},
    "ceiling3": {"b": 3.0, "R": np.sqrt(3) / 2, "var_R": 1 / 40, "lam": 13.0},
}


def sigma_rho(shape):
    s = SHAPES[shape]
    return 2 * s["R"] / (1 - s["R"] ** 2) ** 2 * np.sqrt(s["var_R"])


def log_sphere_area(m):
    """log s_m, the surface area of the unit sphere in R^m."""
    return np.log(2) + (m / 2) * np.log(np.pi) - gammaln(m / 2)


def lk_sphere(j, k, radius):
    """L_j of the sphere S^{k-1} of the given radius (Adler-Taylor eq. 3.6.4); 0 unless k - 1 - j is even."""
    if (k - 1 - j) % 2:
        return 0.0
    log_binom = gammaln(k) - gammaln(j + 1) - gammaln(k - j)
    return float(2 * np.exp(log_binom + log_sphere_area(k) - log_sphere_area(k - j) + j * np.log(radius)))


def ec_density(j, u):
    """rho_j(u) of a unit-variance Gaussian field: 1 - Phi(u) for j = 0, else (2 pi)^{-(j+1)/2} H_{j-1}(u) e^{-u^2/2}."""
    u = np.asarray(u, dtype=float)
    if j == 0:
        return norm.sf(u)
    h_prev, h = np.zeros_like(u), np.ones_like(u)  # probabilists' Hermite: H_{-1} unused, H_0 = 1
    for m in range(1, j):
        h_prev, h = h, u * h - (m - 1) * h_prev
    return (2 * np.pi) ** (-(j + 1) / 2) * h * np.exp(-u * u / 2)


def sup_tail(u, k, lam, even=True):
    """Expected-EC approximation of P(sup over S^{k-1} of a unit-variance field >= u)."""
    total = sum(lk_sphere(j, k, np.sqrt(lam)) * ec_density(j, u) for j in range(k))
    return total / 2 if even else total


U_GRID = np.arange(-10.0, 60.0, 0.002)


@lru_cache(maxsize=None)
def _envelope(k, lam):
    tail = np.asarray(sup_tail(U_GRID, k, lam), dtype=float)
    return np.maximum.accumulate(tail[::-1])[::-1]  # largest value at this u or above


def sup_tail_envelope(u, k, lam):
    """Non-increasing upper envelope of the even-field EC tail: its largest value at u or above (deviations.md).

    The EC expansion is accurate only for large u; for k >= 4 it swings between large positive and negative values
    at moderate u. The envelope keeps the large-u values and replaces the swings by their largest later value.
    """
    env = _envelope(int(k), float(lam))
    if u >= U_GRID[-1]:
        return float(max(sup_tail(u, k, lam), 0.0))
    return float(env[max(0, int(np.searchsorted(U_GRID, u, side="left")) - 1)])  # grid point at or below u: larger p


def sup_p_rho_scale(rho_obs, n, k, shape="ceiling3"):
    """Frozen version (protocol.md): linearised on the rho scale. Failed C1 (too many small p); kept for the record."""
    if k == 0 or not np.isfinite(rho_obs):
        return 1.0
    s = SHAPES[shape]
    u = np.sqrt(n) * (rho_obs - s["b"]) / sigma_rho(shape)
    return float(min(1.0, max(0.0, sup_tail(u, k, s["lam"]))))


def sup_p(rho_obs, n, k, shape="ceiling3"):
    """p that a split of n cells with ratio rho_obs on k invariant directions goes beyond one population's best cut.

    Amended (deviations.md): u on the scale of R = sqrt(rho'/(1 + rho')), rho' = rho n/(n - 2), where the
    linearisation of E3 holds; rho is a convex function of R and its tail is heavier than the linear one.
    """
    if k == 0 or not np.isfinite(rho_obs):
        return 1.0
    s = SHAPES[shape]
    rho_total = rho_obs * n / (n - 2)
    u = np.sqrt(n) * (np.sqrt(rho_total / (1 + rho_total)) - s["R"]) / np.sqrt(s["var_R"])
    return float(min(1.0, max(0.0, sup_tail_envelope(u, k, s["lam"]))))

def directions(k, rng):
    if k == 1:
        return np.ones((1, 1))
    if k == 2:
        a = np.linspace(0, np.pi, 720, endpoint=False)
        return np.c_[np.cos(a), np.sin(a)]
    i = np.arange(2000) + 0.5  # Fibonacci points on the half sphere

def best_cut(y, v):
    """(R^2, order, i) of the best threshold along v; y whitened and centred (total covariance, divisor n)."""
    n = len(y)
    order = np.argsort(y @ v)
    prefix = np.cumsum(y[order], axis=0)[:-1]
    sizes = np.arange(1, n)
    r2 = np.einsum("ij,ij->i", prefix, prefix) / (sizes * (n - sizes))
    i = int(np.argmax(r2))
    return r2[i], order, i + 1


def sup_rho(x, rng):
    n, k = x.shape
    y = x - x.mean(0)
    y = y @ np.linalg.inv(np.linalg.cholesky(y.T @ y / n)).T
    proj = y @ directions(k, rng).T
    order = np.argsort(proj, axis=0)
    best = (-1.0, None, None)
    for j in range(proj.shape[1]):
        prefix = np.cumsum(y[order[:, j]], axis=0)[:-1]
        sizes = np.arange(1, n)
        r2 = np.einsum("ij,ij->i", prefix, prefix) / (sizes * (n - sizes))
        i = int(np.argmax(r2))
        if r2[i] > best[0]:
            best = (r2[i], order[:, j], i + 1)
    for _ in range(5 if k > 1 else 0):  # Fisher re-sort: the best split's own direction, then its best threshold
        left = np.zeros(n, bool)
        left[best[1][:best[2]]] = True
        d = y[left].mean(0) - y[~left].mean(0)
        cand = best_cut(y, d / np.linalg.norm(d))
        if cand[0] <= best[0] + 1e-12:
            break
        best = cand
    r2 = best[0]
    left = np.zeros(n, bool)
    left[best[1][:best[2]]] = True
    return (n - 2) / n * r2 / (1 - r2), left

def metrics(truth, labels, rare=None):
    table = pd.crosstab(truth, labels)
    counts = table.to_numpy()
    truth_n, pred_n = counts.sum(axis=1), counts.sum(axis=0)
    f1 = 2 * counts / (truth_n[:, None] + pred_n)
    rr, cc = linear_sum_assignment(f1, maximize=True)
    cc, rr = cc[counts[rr, cc] > 0], rr[counts[rr, cc] > 0]
    group = pd.DataFrame({"group": table.index, "truth_n": truth_n})
    group["predicted"] = pd.Series(dtype=object)
    group[["precision", "recall", "f1"]] = 0.0
    group.loc[rr, "predicted"] = table.columns[cc].to_numpy(dtype=object)
    group.loc[rr, "precision"] = counts[rr, cc] / pred_n[cc]
    group.loc[rr, "recall"] = counts[rr, cc] / truth_n[rr]
    group.loc[rr, "f1"] = f1[rr, cc]
    if rare is None:
        rare = group.sort_values(["truth_n", "group"]).head(math.ceil(len(group) / 4)).group
    group["rare"] = group.group.isin(rare)
    group["matched_90_90"] = (group.precision >= 0.9) & (group.recall >= 0.9)
    biggest = counts.argmax(axis=1)
    group["legacy_90_90"] = (counts.max(axis=1) / truth_n >= 0.9) & (
        counts.max(axis=0)[biggest] / pred_n[biggest] >= 0.9)
    group["predicted_clusters_touched"] = (counts > 0).sum(axis=1)
    shared_pairs = (counts * (counts - 1)).sum()
    truth_pairs = (truth_n * (truth_n - 1)).sum()
    predicted_pairs = (pred_n * (pred_n - 1)).sum()
    homogeneity, completeness, _ = homogeneity_completeness_v_measure(truth, labels)
    row = {
        "n": len(truth), "true_clusters": len(truth_n), "found_clusters": len(pred_n),
        "cluster_count_error": len(pred_n) - len(truth_n),
        "ari": adjusted_rand_score(truth, labels), "ami": adjusted_mutual_info_score(truth, labels),
        "homogeneity": homogeneity, "completeness": completeness,
        "truth_pair_split": 1 - shared_pairs / truth_pairs if truth_pairs else np.nan,
        "predicted_pair_merge": 1 - shared_pairs / predicted_pairs if predicted_pairs else np.nan,
        "matched_90_90": group.matched_90_90.sum(), "legacy_90_90": group.legacy_90_90.sum(),
        "rare_matched_90_90": group.loc[group.rare, "matched_90_90"].sum(),
    }
    for column in ("precision", "recall", "f1"):
        row[f"macro_{column}"] = group[column].mean()
        row[f"rare_{column}"] = group.loc[group.rare, column].mean()
    return row, group

            assert abs(normalized_mutual_info_score(assigned.Cell_line, assigned.cluster) - old["nmi_cell_line"]) <= 1e-12
            assert abs(adjusted_rand_score(meta.CHOIR_1, assigned.cluster) - old["ari_published_CHOIR_1"]) <= 1e-12
            rows.append(dict(method=method, arm=arm, archive_runtime_s=old["sec"], **row))
            groups.append(group.assign(method=method, arm=arm))
    pd.DataFrame(rows).to_csv(HERE / "choir_metrics.csv", index=False)
    pd.concat(groups, ignore_index=True).to_csv(HERE / "choir_groups.csv", index=False)
    saved_splits(paths)
    write_manifest("saved", paths, {"verified_assignments": len(rows), "metadata_cells": len(meta)})


def saved_splits(paths):
    spec = ANALYSES / "subtree_spectrum_topology_20261108/real_nodes.jsonl"
    ties = ANALYSES / "subspace_distance_ties_20261108/real_splits.jsonl"
    paths.extend([spec, ties])
    spectrum = pd.read_json(spec, lines=True)
    distances = pd.read_json(ties, lines=True)
    joins = ["tree", "node", "n", "n_small"]
    ancillary = spectrum[joins + ["top_multiplicity"]].merge(
        distances[joins + ["kind", "test_tie", "tree_tie", "invariant_tie", "fuzzy_tie", "lines_cut"]],
        on=joins, validate="one_to_one")
    frames = []
    for method, tree in METHODS.items():
        split_b = AUDIT / f"{method}_B_splits.jsonl"
        split_c = AUDIT / f"{method}_C_splits.jsonl"
        linkage_file = CHOIR / method / "linkage.npy"
        paths.extend([split_b, split_c, linkage_file])
        b = pd.read_json(split_b, lines=True).rename(columns={"n_parent": "n", "p": "p_B"})
        c = pd.read_json(split_c, lines=True).rename(columns={"n_parent": "n", "p": "p_C"})
        pd.testing.assert_frame_equal(b[["node", "n", "n_small", "k", "rho"]], c[["node", "n", "n_small", "k", "rho"]])
        linkage = np.load(linkage_file)
        n = len(linkage) + 1
        sizes = np.r_[np.ones(n, dtype=int), linkage[:, 3].astype(int)]
        depth = np.zeros(2 * n - 1, dtype=int)
        for i in reversed(range(len(linkage))):
            depth[linkage[i, :2].astype(int)] = depth[n + i] + 1
        node = b.node.str.removeprefix("N").astype(int)
        assert (sizes[node] == b.n).all()
        assert (sizes[linkage[node - n, :2].astype(int)].min(axis=1) == b.n_small).all()
        b = b.assign(tree=tree, node=node, depth=depth[node], p_C=c.p_C)
        frames.append(b.merge(ancillary, on=joins, how="left", validate="one_to_one"))
    merged = pd.concat(frames, ignore_index=True)
    merged["small_share"] = merged.n_small / merged.n
    merged["B_pass"] = merged.p_B <= 0.01
    merged["C_pass"] = merged.p_C <= 0.01
    merged.to_csv(HERE / "choir_split_relationships.csv", index=False)


REGIMES = {
    "base": (240, 12, 5, "balanced"), "larger_n": (480, 12, 5, "balanced"),
    "wider_p": (240, 72, 5, "balanced"), "imbalanced": (240, 12, 5, "imbalanced"),
    "lower_gap": (240, 12, 2.5, "balanced"), "anisotropic": (240, 12, 5, "anisotropic"),
    "one_population": (240, 12, 0, "null"),
}


def simulate(n, p, gap, shape, seed):
    rng = np.random.default_rng(seed)
    x = rng.standard_normal((n, p))
    if shape == "anisotropic":
        x[:, :2] = (x[:, :2] * [2, 0.5]) @ (np.array([[1, -1], [1, 1]]) / np.sqrt(2))
    sizes = [n] if shape == "null" else ([132, 60, 36, 12] if shape == "imbalanced" else [n // 4] * 4)
    truth = np.repeat(np.arange(len(sizes)), sizes)
    x[:, :2] += gap * (np.array([[-1, -1], [-1, 1], [1, -1], [1, 1]]) / 2)[truth]
    order = rng.permutation(n)
    return x[order], truth[order]


def load_audit():
    sys.path.insert(0, str(ROOT))
    spec = importlib.util.spec_from_file_location("matched_audit", AUDIT / "rerun.py")
    audit = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(audit)
    return audit, audit.load_real_harness()


def fresh():
    audit, harness = load_audit()
    from benchmarks.shared.runners.dispatch import run_clustering_result
    from benchmarks.shared.runners.method_registry import METHOD_SPECS
    from tree_break_selection.tree.feature_space import continuous_feature_space_from_columns

    def sibling_hook(*args):
        result = harness.swapped(*args)
        if harness.STATE["arm"] == "C":
            splits = iter(harness.STATE["splits"])
            for i, decision in enumerate(result[2]):
                if decision is None:
                    continue
                split = next(splits)
                p = audit.sup_p(split["rho"], split["n_parent"], split["k"])
                result[0][i] = (*result[0][i][:2], p)
                result[2][i] = replace(decision, p_value=p)
                split["p"] = p
            assert next(splits, None) is None
        return result

    harness.pipeline.apply_split_count_rule_out = sibling_hook
    sources = [ROOT / path for path in json.loads((AUDIT / "manifest.json").read_text())["hashes"]
               if (ROOT / path).is_file() and not path.endswith((".npy", ".csv"))]
    sources += [AUDIT / "rerun.py"]
    write_manifest("fresh", sources, {"seeds": list(range(731901, 731905)), "BLAS_threads": 1})
    rows, groups, assignments = [], [], []
    for regime, (n, p, gap, shape) in REGIMES.items():
        for seed in range(731901, 731905):
            x, truth = simulate(n, p, gap, shape, seed)
            jitter = np.random.default_rng([seed, 42]).normal(0, 0.1, x.shape)
            for method in METHODS:
                for arm in ARMS:
                    original = None
                    for perturbed in (False, True):
                        df = pd.DataFrame(x + jitter if perturbed else x, columns=[f"x{i}" for i in range(p)])
                        fs = continuous_feature_space_from_columns(tuple(df.columns))
                        harness.STATE.update(arm=arm, splits=[])
                        params = dict(METHOD_SPECS[method].param_grid[0], sibling_split_count_rule_out=arm != "A")
                        start = perf_counter()
                        out = run_clustering_result(df, method, params, seed=seed, significance_level=0.01,
                                                    edge_alpha=0.001, feature_space=fs)
                        seconds = perf_counter() - start
                        assert out.status.value == "ok", (regime, seed, method, arm, out.status)
                        assert out.extra["feature_family_diffusion_fallback_reason"] is None
                        if hasattr(out.labels, "index"):
                            assert out.labels.index.equals(df.index)
                        labels = np.asarray(out.labels)
                        scores, group = metrics(truth, labels, [3] if shape == "imbalanced" else [])
                        assert scores["found_clusters"] == out.found_clusters
                        if not perturbed:
                            original = labels.copy()
                        ids = dict(regime=regime, seed=seed, method=method, arm=arm, perturbed=perturbed)
                        splits = harness.STATE["splits"]
                        rows.append(dict(**ids, p=p, runtime_s=seconds, **scores,
                                         stability_ari=adjusted_rand_score(original, labels) if perturbed else np.nan,
                                         tested_opt_in_splits=len(splits),
                                         raw_opt_in_hits=sum(s["p"] <= 0.01 for s in splits)))
                        groups.append(group.assign(**ids))
                        assignments.append(pd.DataFrame(dict(cell=np.arange(n), truth=truth, cluster=labels)).assign(**ids))
            print(regime, seed, flush=True)
        pd.DataFrame(rows).to_csv(HERE / "fresh_metrics.csv", index=False)
    pd.concat(groups, ignore_index=True).to_csv(HERE / "fresh_groups.csv", index=False)
    pd.concat(assignments, ignore_index=True).to_csv(HERE / "fresh_assignments.csv.gz", index=False)
    assert len(rows) == len(REGIMES) * 4 * len(METHODS) * len(ARMS) * 2


if __name__ == "__main__":

        {"saved": saved, "fresh": fresh}[sys.argv[1]]()

# Execution qualifications

- The frozen protocol requested a one-job threading backend. `prepare` retains
  an explicit neighbor-search `n_jobs=-1`, which may override the backend's
  default `n_jobs=1`. The run used two worker processes and explicit one-thread
  BLAS/OpenMP/Numba environment settings; actual neighbor-search thread counts
  were not instrumented. Do not describe all internal work as single-threaded.
  The six original replays matched, and all paired input/geometry hashes agree.
- Review identified that an exception outside the per-fit handler could replace
  completed rows within that dataset, and that a failure-inclusive upper
  endpoint would require its own binomial upper bound. All 800 attempts completed
  with status `ok`; neither failure branch was used. No samples were dropped,
  replaced or retried, and the reported intervals therefore concern all planned
  datasets directly.
- A bypasses the B/C/rule-out diagnostic hook, so its split trace is empty. Empty
  traces for other arms can result from inherited eligibility. They cannot be
  used as evidence that the candidate tail was calibrated at every node.
