# ===== protocol.md =====
# # Protocol: Gaussian best-flat-cut reference on the benchmark panel (frozen 2026-11-11, before the run)
#
# - Cases: `get_test_cases_by_suite("continuous")`, all 9, and `get_test_cases_by_suite("cancer_pc")`, all 4.
#   The other benchmark cases are binary or categorical; the reference is derived for Gaussian coordinates only.
# - Methods: `tbs_family_diffusion` and `tbs_family_diffusion_markov_ward`, each with its registered parameter
#   cell and the same cell plus `sibling_best_flat_cut_reference=True`; run through the shared
#   `benchmark_cluster_algorithm` (default alphas), plots off, method errors recorded rather than raised.
# - Endpoints (descriptive): found clusters against the true number, ARI and NMI per case, method and cell.
# - Expected limit, stated before running: a subtree of n <= p + 1 cells has no invariant directions and a
#   subtree below 15 cells is outside the table, so both get p = 1 and are not split. The split-count rule-out
#   lost real groups for this reason in gauss_clear_medium (60 cells, 40 features), gauss_moderate_3c (45, 30),
#   gauss_dense_signal_highd (40, 20,000) and dim_diffuse_6c_136f (180, 136); the same loss is expected here.
#   The cancer_pc cases (496 cells, 30 features) are outside this limit at the first splits.
# - One run, no tuning of the module after seeing results.


# ===== run.py =====

# ===== reasons.py =====
"""Replay of the reference cell, one case and method per call, recording every call of best_flat_cut_reference."""
import json
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parents[2]))

from benchmarks.shared.benchmark_runs.runtime import apply_single_thread_runtime_defaults  # noqa: E402
from benchmarks.shared.cases import get_test_cases_by_suite  # noqa: E402
from benchmarks.shared.pipeline import benchmark_cluster_algorithm  # noqa: E402
from benchmarks.shared.runners.method_registry import METHOD_SPECS  # noqa: E402
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence import best_flat_cut_reference as mod  # noqa: E402

METHODS = ["tbs_family_diffusion", "tbs_family_diffusion_markov_ward"]
calls = []
original = mod.best_flat_cut_reference


def recording(points, in_left):
    r = original(points, in_left)
    calls.append(dict(n=int(len(points)), left=int(in_left.sum()), reason=r.reason, k=int(r.invariant_directions),
                      rho=None if r.rho is None else float(r.rho), p=float(r.p_value)))
    return r


mod.best_flat_cut_reference = recording
apply_single_thread_runtime_defaults()
out = []
for case in get_test_cases_by_suite("continuous") + get_test_cases_by_suite("cancer_pc"):
    for m in METHODS:
        calls.clear()
        res = benchmark_cluster_algorithm(test_cases=[case], methods=[m],
                                          method_params={m: [{**METHOD_SPECS[m].param_grid[0], "sibling_best_flat_cut_reference": True}]},
                                          verbose=False, plot_umap=False, include_cover_pages=False,
                                          include_validation_page=False, allow_method_errors=True)
        row = dict(case=case["name"], method=m, found=int(res["found_clusters"].iloc[0]), ari=float(res["ari"].iloc[0]),
                   calls=sorted(calls, key=lambda c: -c["n"]))
        out.append(row)
        print(row["case"], m, row["found"], round(row["ari"], 3), len(calls), flush=True)
json.dump(out, open(HERE / "reasons.json", "w"), indent=1)

# ===== boundaries.py =====
"""Do the refused splits in the cancer_pc cases follow a true group boundary?

Replays the reference cell and, for every call of best_flat_cut_reference, records the true-label counts on both
sides, and for each true group inside the subtree the rho and reference p-value of "this group against the rest"
in the same invariant directions, plus that split's rho in all 30 features.
"""
import json
import sys
from pathlib import Path

import numpy as np

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parents[2]))

from benchmarks.shared.benchmark_runs.runtime import apply_single_thread_runtime_defaults  # noqa: E402
from benchmarks.shared.cases import get_test_cases_by_suite  # noqa: E402
from benchmarks.shared.generators import generate_case_data  # noqa: E402
from benchmarks.shared.pipeline import benchmark_cluster_algorithm  # noqa: E402
from benchmarks.shared.runners.method_registry import METHOD_SPECS  # noqa: E402
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence import best_flat_cut_reference as mod  # noqa: E402
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import invariant_directions  # noqa: E402

METHODS = ["tbs_family_diffusion", "tbs_family_diffusion_markov_ward"]
calls, label_of_row = [], {}
original = mod.best_flat_cut_reference


def recording(points, in_left):
    r = original(points, in_left)
    y = np.array([label_of_row[row.tobytes()] for row in np.ascontiguousarray(points, dtype=np.float64)])
    count = lambda mask: {int(g): int((y[mask] == g).sum()) for g in np.unique(y[mask])}
    row = dict(n=int(len(points)), reason=r.reason, k=int(r.invariant_directions), p=float(r.p_value),
               rho=None if not np.isfinite(r.rho) else float(r.rho), left=count(in_left), right=count(~in_left), groups=[])
    directions = invariant_directions(points)
    groups = [g for g in np.unique(y) if 5 <= (y == g).sum() <= len(y) - 5]
    for g in groups if len(np.unique(y)) > 1 else []:
        mask = y == g
        entry = dict(group=int(g), size=int(mask.sum()))
        if len(points) > points.shape[1] + 2:
            entry["rho_all_features"] = float(mod.split_rho(points, mask))
        if 1 <= len(directions) <= mod.MAX_DIRECTIONS:
            rho = float(mod.split_rho(points @ directions.T, mask))
            entry.update(rho_invariant=rho, p_invariant=float(mod.reference_p_value(rho, len(points), len(directions))[0]))
        row["groups"].append(entry)
    calls.append(row)
    return r


mod.best_flat_cut_reference = recording
apply_single_thread_runtime_defaults()
out = []
for case in get_test_cases_by_suite("cancer_pc"):
    data_df, y, _, _ = generate_case_data(case)
    label_of_row.clear()
    label_of_row.update({row.tobytes(): int(g) for row, g in zip(np.ascontiguousarray(data_df.to_numpy(dtype=np.float64)), y)})
    for m in METHODS:
        calls.clear()
        res = benchmark_cluster_algorithm(test_cases=[case], methods=[m],
                                          method_params={m: [{**METHOD_SPECS[m].param_grid[0], "sibling_best_flat_cut_reference": True}]},
                                          verbose=False, plot_umap=False, include_cover_pages=False,
                                          include_validation_page=False, allow_method_errors=False)
        out.append(dict(case=case["name"], hub_scale=case["hub_scale"], isolation_offset=case["isolation_offset"], method=m,
                        found=int(res["found_clusters"].iloc[0]), ari=float(res["ari"].iloc[0]), calls=sorted(calls, key=lambda c: -c["n"])))
        print(case["name"], m, out[-1]["found"], round(out[-1]["ari"], 3), len(calls), flush=True)
json.dump(out, open(HERE / "boundaries.json", "w"), indent=1)
