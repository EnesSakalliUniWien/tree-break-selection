# ===== protocol.md =====
# # Protocol: the spr measure per eigenvalue block of the invariant directions (frozen 2026-11-11, before the run)
#
# Descriptive; no gate or default is changed. Data, saved Markov/Ward tree and the 1,285 tested splits (32 cells
# or more, 1 to 10 invariant directions) of `spr_alternatives_best_flat_cut_20261111`.
#
# ## Blocks
#
# The invariant directions of node P are eigenvectors of P's covariance. Blocks are the numerical eigenvalue
# groups of `tied_eigenvalue_neighbors` (`tree_break_selection/eigenvalue_groups.py`, the rule
# `spectral_subspaces` uses). No statistical test of equal eigenvalues exists in the code; near-equal neighbours
# (ratio of adjacent eigenvalues below 1.1) are counted as a stated convention only, and directions inside such a
# pair are not identified separately by the data.
#
# ## Trace (algebra, checked in the self-check)
#
# The invariant directions are uncorrelated in P, so whitening scales each direction by its own standard
# deviation. Squared distances add over directions. Therefore, for any split and any clade:
#
# - the between/total ratio is the sum of the blocks' ratios: r2 = sum over blocks of r2_block;
# - the transfer cost of an spr is the sum of the blocks' transfer costs: dW = sum over blocks of dW_block.
#
# So every split has a share per block, and every spr is helped by some blocks and opposed by others.
#
# ## Measured per split and block
#
# 1. Share of each block in r2 for the tree split, the greedy spr result and the best flat cut (all found in the
#    k invariant directions jointly, as before); the block with the largest share for each.
# 2. The block alone (one direction): the tree split's `rho`; the block's own best flat cut (exact: best
#    threshold of the sorted coordinate) with `rho`, reference p at (n, 1), line cut share; greedy sprs from the
#    tree split using this block only, with `rho` and p. Passing is reported at 0.01 and at 0.01 / k (Bonferroni
#    over the k blocks of the node).
# 3. Crossing: for every pair of blocks of one node whose own cuts both pass at 0.01 / k, the ARI between the two
#    cuts. A low ARI means two separations that one split of the tree cannot hold together.
# 4. Equidistance per block: share of cells tied (|m| < 0.1) and on the wrong side in the block's coordinate.
# 5. Sprs per block: for the clades whose spr raises `rho` in all invariant directions (dW < 0), in how many
#    blocks dW_block < 0, and whether the first block alone has the sign of the total.
#
# Summaries by block rank (first, second, third, later) and by the verdict classes of the joint reference (tree
# and best pass 57, only best passes and greedy sprs pass 122, only best passes and greedy sprs fail 99, neither
# 1,007).
#
# ## Self-check before the run (selfcheck.py)
#
# (a) sum of block r2 equals the joint r2 and sum of block dW equals the joint dW, to rounding error; (b) the
# exact one-direction best flat cut equals the table's search on the same coordinate; (c) four groups on a 2 x 2
# grid: two blocks, both own cuts pass, ARI between them near 0; (d) one Gaussian population: count of passing
# blocks in 200 datasets at 0.01 / k.
#
# ## Expected before running
#
# - Real data have no numerical ties, so every invariant direction is its own block.
# - The additivity holds to rounding error.
# - The joint reference at (n, k) asks for more than the reference at (n, 1). So some splits whose joint best flat
#   cut is refused may have one block whose own cut passes; how many remain at 0.01 / k is not predicted.
# - Which block carries the tree split, and whether nodes hold crossing cuts, is not predicted.
#
# ## Limits known before running
#
# One dataset and one tree; dependent splits; blocks are chosen from the same cells that are then cut, and the
# reference does not cover that choice; Bonferroni over blocks covers only the choice of block within one node;
# near-equal eigenvalues make single directions arbitrary; one direction per block, so a separation along a
# mixture of two blocks is not seen by a single block; nothing is applied to the fit.

# ===== block_lib.py =====
"""Eigenvalue blocks of a node's invariant directions and one-direction cuts (protocol.md)."""
import numpy as np


def standardized_blocks(points, directions):
    """Return the node's cells in its invariant directions, each scaled to variance 1 (divisor n), and the eigenvalues."""
    coordinates = points @ directions.T
    centred = coordinates - coordinates.mean(0)
    return centred / np.sqrt((centred**2).mean(0)), coordinates.var(0, ddof=1)


def best_cut_one_direction(column):
    """Exact best flat cut of one standardized coordinate: (r2, cells on the low side)."""
    n = len(column)
    order = np.argsort(column, kind="stable")
    prefix = np.cumsum(column[order])[:-1]
    s = np.arange(1, n)
    r2 = prefix**2 / (s * (n - s))
    best = int(r2.argmax())
    mask = np.zeros(n, dtype=bool)
    mask[order[: best + 1]] = True
    return float(r2[best]), mask


def r2_per_block(y, mask):
    s = int(mask.sum())
    return y[mask].sum(0) ** 2 / (s * (len(y) - s))

"""protocol.md self-check: additivity, exact one-direction cut, crossing cuts on a grid, one population."""

from block_lib import best_cut_one_direction, r2_per_block, standardized_blocks  # noqa: E402
from margin_lib import clade_margins, layout  # noqa: E402
from measure_lib import best_flat_cut, r2_of  # noqa: E402
from spr_lib import index_tree, subtree  # noqa: E402

from tree_break_selection.eigenvalue_groups import tied_eigenvalue_neighbors  # noqa: E402
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.best_flat_cut_reference import reference_p_value, to_rho, whiten  # noqa: E402
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import invariant_directions  # noqa: E402

rng = np.random.default_rng([20261111, 51])
n = 400
grid = np.c_[np.arange(n) % 2, (np.arange(n) // 2) % 2]
x = rng.standard_normal((n, 12))
x[:, 0] += 9.0 * grid[:, 0]
x[:, 1] += 6.0 * grid[:, 1]
directions = invariant_directions(x)
y, values = standardized_blocks(x, directions)
print(f"(c) grid: k = {len(directions)}, numerical ties {int(tied_eigenvalue_neighbors(values).sum())}, "
      f"largest |standardized - whiten| {np.abs(y - whiten(x @ directions.T)).max():.1e}")

z = linkage(x, "ward")
parent = np.full(2 * n - 1, -1, dtype=np.int64)
for k, (a, b) in enumerate(z[:, :2].astype(int)):
    parent[a] = parent[b] = n + k
order, pre, sub_all = index_tree(parent)
nodes, local_parent, is_leaf, sub = subtree(parent, order, pre, sub_all, 2 * n - 2, n)
rows = nodes[is_leaf]
lo, sizes, second, below_first = layout(is_leaf, sub)
tree_mask = np.arange(n) < sizes[1]
yr = y[rows]
_, dw = clade_margins(yr, lo, sizes, second, below_first)
dw_blocks = np.column_stack([clade_margins(yr[:, [i]], lo, sizes, second, below_first)[1] for i in range(yr.shape[1])])
print(f"(a) |sum of block r2 - joint r2| {abs(r2_per_block(yr, tree_mask).sum() - r2_of(yr, tree_mask)):.1e}; "
      f"largest |sum of block dW - joint dW| {np.nanmax(np.abs(dw_blocks.sum(1) - dw)):.1e}")

cuts = []
for i in range(y.shape[1]):
    r2, mask = best_cut_one_direction(y[:, i])
    searched, _ = best_flat_cut(x @ directions[[i]].T)
    p = reference_p_value(to_rho(r2, n), n, 1)[0]
    cuts.append(mask)
    print(f"    block {i + 1}: exact rho {to_rho(r2, n):.4f}, table's search {searched:.4f}, p {p:.4f}, "
          f"ARI with grid axis 1 {adjusted_rand_score(grid[:, 0], mask):.3f}, axis 2 {adjusted_rand_score(grid[:, 1], mask):.3f}")
print(f"    ARI between the two block cuts {adjusted_rand_score(cuts[0], cuts[1]):.3f}")

passing = tested = with_directions = 0
for b in range(200):
    g = np.random.default_rng([20261111, 52, b]).standard_normal((200, 12)) * np.sqrt(np.r_[9.0, 4.0, np.ones(10)])
    d = invariant_directions(g)
    if len(d) == 0:
        continue
    with_directions += 1
    yb, _ = standardized_blocks(g, d)
    for i in range(yb.shape[1]):
        tested += 1
        passing += reference_p_value(to_rho(best_cut_one_direction(yb[:, i])[0], 200), 200, 1)[0] <= 0.01 / yb.shape[1]
print(f"(d) one Gaussian population, 200 datasets of 200 cells: {with_directions} with invariant directions, "
      f"{tested} blocks, passing at 0.01 / k: {passing}")

"""protocol.md: block shares, one-direction cuts, crossing, equidistance and sprs per eigenvalue block."""
import json
import sys
from itertools import combinations
from multiprocessing import Pool
from pathlib import Path

import numpy as np
import pandas as pd
from sklearn.metrics import adjusted_rand_score

HERE = Path(__file__).resolve().parent
SPR = HERE.parent / "spr_alternatives_best_flat_cut_20261111"
sys.path.insert(0, str(HERE.parents[2]))
sys.path.insert(0, str(HERE))
for other in ("spr_subspaces_equidistances_20261111", "spr_alternatives_best_flat_cut_20261111", "split_vs_best_flat_cut_20261111"):
    sys.path.insert(0, str(HERE.parent / other))

from block_lib import best_cut_one_direction, r2_per_block, standardized_blocks  # noqa: E402
from margin_lib import TIE, clade_margins, layout  # noqa: E402
from measure_lib import best_flat_cut, r2_of  # noqa: E402
from spr_lib import greedy_sprs, index_tree, subtree  # noqa: E402

from tree_break_selection.eigenvalue_groups import tied_eigenvalue_neighbors  # noqa: E402
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.best_flat_cut_reference import reference_p_value, to_rho  # noqa: E402
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import invariant_directions  # noqa: E402

METHOD = "tbs_family_diffusion_markov_ward"
STATE = {}


def init():
    source = HERE.parent / "choir_cell_lines_tbs_lbfgsb_20261106"
    STATE["x"] = np.load(source / "pca.npy")
    STATE["line"] = pd.factorize(pd.read_csv(source / "cell_metadata.csv", index_col=0, dtype=str).Cell_line)[0]
    z = np.load(SPR / f"tree_{METHOD}.npz")
    STATE["parent"], STATE["leaf_row"] = z["parent"], z["leaf_row"]
    STATE["index"] = index_tree(z["parent"])


def line_cut_share(lines, mask):
    left = np.bincount(lines[mask], minlength=lines.max() + 1)
    right = np.bincount(lines[~mask], minlength=lines.max() + 1)
    return float(np.minimum(left, right).sum() / len(lines))


def job(node):
    order, pre, sub_all = STATE["index"]
    nodes, local_parent, is_leaf, sub = subtree(STATE["parent"], order, pre, sub_all, node, len(STATE["leaf_row"]))
    rows = STATE["leaf_row"][nodes[is_leaf]]
    points = STATE["x"][rows]
    n = len(rows)
    directions = invariant_directions(points)
    k = len(directions)
    y, values = standardized_blocks(points, directions)
    lines = STATE["line"][rows]
    lo, sizes, second, below_first = layout(is_leaf, sub)
    tree_mask = np.arange(n) < sizes[1]

    def p_of(r2, directions_used):
        return float(reference_p_value(to_rho(r2, n), n, directions_used)[0])

    final = tree_mask
    for _, _, left in greedy_sprs(y, local_parent, is_leaf, sub):
        final = left
    rho_best, best = best_flat_cut(points @ directions.T)
    out = {"node": f"N{node}", "n": n, "k": k, "lines_in_subtree": int(len(np.unique(lines))),
           "eigenvalues": [float(v) for v in values], "numerical_ties": int(tied_eigenvalue_neighbors(values).sum()),
           "near_equal_pairs": int((values[:-1] / values[1:] < 1.1).sum()),
           "p_tree": p_of(r2_of(y, tree_mask), k), "p_spr": p_of(r2_of(y, final), k), "p_best": float(reference_p_value(rho_best, n, k)[0]),
           "rho_tree": to_rho(r2_of(y, tree_mask), n), "rho_best": rho_best,
           "share": {name: [float(v) for v in r2_per_block(y, mask) / r2_per_block(y, mask).sum()] for name, mask in (("tree", tree_mask), ("spr", final), ("best", best))},
           "cut_tree": line_cut_share(lines, tree_mask), "cut_best": line_cut_share(lines, best)}

    _, dw = clade_margins(y, lo, sizes, second, below_first)
    dw_blocks = np.empty((len(sub), k))
    blocks, cuts = [], []
    for i in range(k):
        column = y[:, [i]]
        m, dw_blocks[:, i] = clade_margins(column, lo, sizes, second, below_first)
        cell = m[is_leaf]
        r2_cut, cut = best_cut_one_direction(y[:, i])
        steps = list(greedy_sprs(column, local_parent, is_leaf, sub))
        r2_tree = float(r2_per_block(y, tree_mask)[i])
        cuts.append(cut)
        blocks.append({"rho_tree": to_rho(r2_tree, n), "p_tree": p_of(r2_tree, 1), "rho_cut": to_rho(r2_cut, n), "p_cut": p_of(r2_cut, 1),
                       "small_cut": int(min(cut.sum(), n - cut.sum())), "line_cut": line_cut_share(lines, cut),
                       "ari_cut_tree": float(adjusted_rand_score(tree_mask, cut)), "ari_cut_best": float(adjusted_rand_score(best, cut)),
                       "rho_spr": to_rho(steps[-1][0], n), "p_spr": p_of(steps[-1][0], 1), "sprs": len(steps) - 1,
                       "tied": int((np.abs(cell) < TIE).sum()), "wrong": int((cell < 0).sum())})
    out["blocks"] = blocks
    passing = [i for i in range(k) if blocks[i]["p_cut"] <= 0.01 / k]
    out["crossing"] = [[i, j, float(adjusted_rand_score(cuts[i], cuts[j]))] for i, j in combinations(passing, 2)]

    out["spr_blocks"] = {}
    valid = ~np.isnan(dw)
    for label, group in (("all", valid), ("big", valid & (sizes >= 10))):
        raises = group & (dw < 0)
        agree = (dw_blocks[raises] < 0).sum(1)
        out["spr_blocks"][label] = {"raises": int(raises.sum()), "agreeing_blocks": int(agree.sum()), "all_agree": int((agree == k).sum()),
                                    "first_agrees": int((dw_blocks[raises, 0] < 0).sum()), "only_one_agrees": int((agree == 1).sum())}
    return out


if __name__ == "__main__":
    init()
    earlier = [json.loads(line) for line in (SPR / "measures.jsonl").read_text().splitlines()]
    jobs = [int(r["node"][1:]) for r in earlier if "rho_tree" in r]
    limit = int(sys.argv[1]) if len(sys.argv) > 1 else None
    jobs = sorted(jobs, key=lambda node: STATE["index"][2][node])[:limit] if limit else jobs
    with Pool(10, initializer=init) as pool, (HERE / ("trial.jsonl" if limit else "measures.jsonl")).open("w") as f:
        for done, row in enumerate(pool.imap_unordered(job, jobs), 1):
            f.write(json.dumps(row) + "\n")
            if done % 200 == 0:
                print(done, "of", len(jobs), flush=True)
    print("DONE", len(jobs), flush=True)

"""protocol.md: summary tables from measures.jsonl."""
import json
from pathlib import Path

import numpy as np

HERE = Path(__file__).resolve().parent
rows = [json.loads(line) for line in (HERE / "measures.jsonl").read_text().splitlines()]
CLASSES = [
    ("all", lambda r: True),
    ("tree and best pass", lambda r: r["p_tree"] <= 0.01 and r["p_best"] <= 0.01),
    ("only best passes, greedy sprs pass", lambda r: r["p_tree"] > 0.01 and r["p_best"] <= 0.01 and r["p_spr"] <= 0.01),
    ("only best passes, greedy sprs fail", lambda r: r["p_tree"] > 0.01 and r["p_best"] <= 0.01 and r["p_spr"] > 0.01),
    ("neither passes", lambda r: r["p_tree"] > 0.01 and r["p_best"] > 0.01),
]
groups = [(name, [r for r in rows if f(r)]) for name, f in CLASSES]
print("splits:", {name: len(g) for name, g in groups})
ks = np.array([r["k"] for r in rows])
print(f"blocks per split: median {np.median(ks):.0f}, 1 block in {(ks == 1).sum()}, 2-3 in {((ks >= 2) & (ks <= 3)).sum()}, 4-6 in {((ks >= 4) & (ks <= 6)).sum()}, 7-10 in {(ks >= 7).sum()}; "
      f"blocks in all {ks.sum()}; numerical ties {sum(r['numerical_ties'] for r in rows)}; near-equal neighbour pairs (ratio < 1.1) {sum(r['near_equal_pairs'] for r in rows)} of {int((ks - 1).sum())}, "
      f"splits with one or more {sum(r['near_equal_pairs'] > 0 for r in rows)}")


def med(values):
    values = list(values)
    return float(np.median(values)) if values else float("nan")


print("\n1. Share of r2 per block (splits with 2 or more blocks)")
for name, g in groups:
    g = [r for r in g if r["k"] >= 2]
    if not g:
        continue
    parts = []
    for kind in ("tree", "spr", "best"):
        top = np.array([int(np.argmax(r["share"][kind])) for r in g])
        parts.append(f"{kind}: median largest share {med(max(r['share'][kind]) for r in g):.2f}, first block's share {med(r['share'][kind][0] for r in g):.2f}, "
                     f"largest is block 1 / 2 / 3 / later in {(top == 0).sum()} / {(top == 1).sum()} / {(top == 2).sum()} / {(top >= 3).sum()}")
    same = sum(int(np.argmax(r["share"]["tree"])) == int(np.argmax(r["share"]["best"])) for r in g)
    print(f"  {name} ({len(g)}): " + "; ".join(parts) + f"; tree split and best flat cut have the same largest block in {same}")

print("\n2. The block alone, by block rank: tree split / own best cut / greedy sprs in the block")
for name, g in groups:
    print(f"  {name}:")
    for label, pick in (("block 1", lambda b: b[:1]), ("block 2", lambda b: b[1:2]), ("block 3", lambda b: b[2:3]), ("later blocks", lambda b: b[3:])):
        items = [(b, r) for r in g for b in pick(r["blocks"])]
        if not items:
            continue
        n_cells = sum(r["n"] for _, r in items)
        print(f"      {label} ({len(items)}): median rho tree {med(b['rho_tree'] for b, _ in items):.2f}, own cut {med(b['rho_cut'] for b, _ in items):.2f}, sprs {med(b['rho_spr'] for b, _ in items):.2f}; "
              f"pass at 0.01: tree {sum(b['p_tree'] <= 0.01 for b, _ in items)}, own cut {sum(b['p_cut'] <= 0.01 for b, _ in items)}, sprs {sum(b['p_spr'] <= 0.01 for b, _ in items)}; "
              f"pass at 0.01/k: tree {sum(b['p_tree'] <= 0.01 / r['k'] for b, r in items)}, own cut {sum(b['p_cut'] <= 0.01 / r['k'] for b, r in items)}, sprs {sum(b['p_spr'] <= 0.01 / r['k'] for b, r in items)}; "
              f"cells tied {100 * sum(b['tied'] for b, _ in items) / n_cells:.1f}%, wrong side {100 * sum(b['wrong'] for b, _ in items) / n_cells:.1f}%")

print("\n3. Splits by the number of blocks whose own cut passes at 0.01/k (and at 0.01)")
for name, g in groups:
    strict = np.array([sum(b["p_cut"] <= 0.01 / r["k"] for b in r["blocks"]) for r in g])
    loose = np.array([sum(b["p_cut"] <= 0.01 for b in r["blocks"]) for r in g])
    tree_block = sum(any(b["p_tree"] <= 0.01 / r["k"] for b in r["blocks"]) for r in g)
    spr_block = sum(any(b["p_spr"] <= 0.01 / r["k"] for b in r["blocks"]) for r in g)
    print(f"  {name} ({len(g)}): 0 / 1 / 2 / 3+ blocks at 0.01/k: {(strict == 0).sum()} / {(strict == 1).sum()} / {(strict == 2).sum()} / {(strict >= 3).sum()}; "
          f"at 0.01: {(loose == 0).sum()} / {(loose == 1).sum()} / {(loose == 2).sum()} / {(loose >= 3).sum()}; "
          f"tree split passes in some block at 0.01/k: {tree_block}; greedy sprs in some block: {spr_block}")
passing = [(b, r) for r in rows for b in r["blocks"] if b["p_cut"] <= 0.01 / r["k"]]
print(f"  passing own cuts at 0.01/k: {len(passing)}; median line cut share {med(b['line_cut'] for b, _ in passing):.3f}; cut at most 5% of cells from their line: {sum(b['line_cut'] <= 0.05 for b, _ in passing)}; "
      f"smaller side under 5 cells: {sum(b['small_cut'] < 5 for b, _ in passing)}; median ARI with the joint best flat cut {med(b['ari_cut_best'] for b, _ in passing):.3f}, with the tree split {med(b['ari_cut_tree'] for b, _ in passing):.3f}")
for label, f in (("several lines", lambda r: r["lines_in_subtree"] > 1), ("one line", lambda r: r["lines_in_subtree"] == 1)):
    sel = [(b, r) for b, r in passing if f(r)]
    nodes = {r["node"] for _, r in sel}
    print(f"      {label}: {len(sel)} passing cuts in {len(nodes)} of {sum(f(r) for r in rows)} splits; cut at most 5%: {sum(b['line_cut'] <= 0.05 for b, _ in sel)}")

print("\n4. Crossing: pairs of blocks of one node whose own cuts both pass at 0.01/k")
for name, g in groups:
    pairs = [c[2] for r in g for c in r["crossing"]]
    nodes = sum(len(r["crossing"]) > 0 for r in g)
    if pairs:
        pairs = np.array(pairs)
        print(f"  {name}: {len(pairs)} pairs in {nodes} splits; median ARI {np.median(pairs):.3f}; ARI below 0.1 in {(pairs < 0.1).sum()}, 0.1 to 0.5 in {((pairs >= 0.1) & (pairs < 0.5)).sum()}, 0.5 and more in {(pairs >= 0.5).sum()}")
    else:
        print(f"  {name}: 0 pairs")

print("\n5. Clades whose spr raises rho in all invariant directions: blocks with dW_block < 0 (splits with 2 or more blocks)")
for label in ("all", "big"):
    for name, g in groups:
        g = [r for r in g if r["k"] >= 2]
        total = sum(r["spr_blocks"][label]["raises"] for r in g)
        if not total:
            continue
        blocks = sum(r["spr_blocks"][label]["raises"] * r["k"] for r in g)
        print(f"  {label} clades, {name}: {total} sprs; blocks agreeing {100 * sum(r['spr_blocks'][label]['agreeing_blocks'] for r in g) / blocks:.1f}%; "
              f"all blocks agree {100 * sum(r['spr_blocks'][label]['all_agree'] for r in g) / total:.1f}%; only one block agrees {100 * sum(r['spr_blocks'][label]['only_one_agrees'] for r in g) / total:.1f}%; "
              f"first block agrees {100 * sum(r['spr_blocks'][label]['first_agrees'] for r in g) / total:.1f}%")

"""Follow-up to protocol.md: cell lines against the two crossing block cuts in the nodes with two passing blocks."""
import json
import sys
from pathlib import Path

import numpy as np
import pandas as pd

HERE = Path(__file__).resolve().parent
SPR = HERE.parent / "spr_alternatives_best_flat_cut_20261111"
sys.path.insert(0, str(HERE.parents[2]))
sys.path.insert(0, str(HERE))
sys.path.insert(0, str(SPR))

from block_lib import best_cut_one_direction, standardized_blocks  # noqa: E402
from spr_lib import index_tree, subtree  # noqa: E402

from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.best_flat_cut_reference import reference_p_value, to_rho  # noqa: E402
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import invariant_directions  # noqa: E402

METHOD = "tbs_family_diffusion_markov_ward"
source = HERE.parent / "choir_cell_lines_tbs_lbfgsb_20261106"
x = np.load(source / "pca.npy")
line = pd.read_csv(source / "cell_metadata.csv", index_col=0, dtype=str).Cell_line.to_numpy()
z = np.load(SPR / f"tree_{METHOD}.npz")
parent, leaf_row = z["parent"], z["leaf_row"]
order, pre, sub_all = index_tree(parent)
rows_all = [json.loads(text) for text in (HERE / "measures.jsonl").read_text().splitlines()]
fit = {s["node"]: s for s in json.loads((SPR / f"splits_{METHOD}.json").read_text())["splits"]}
totals = pd.Series(line).value_counts()


def side_summary(lines, mask):
    """Lines by where their cells in the node fall: low side, high side, or cut (each side at least 10%)."""
    table = pd.crosstab(lines, mask).reindex(columns=[True, False], fill_value=0)
    share_low = table[True] / table.sum(1)
    low, high, cut = table[share_low >= 0.9], table[share_low <= 0.1], table[(share_low > 0.1) & (share_low < 0.9)]
    return low, high, cut


for record in sorted((r for r in rows_all if r["crossing"]), key=lambda r: -r["n"]):
    node = int(record["node"][1:])
    nodes, _, is_leaf, _ = subtree(parent, order, pre, sub_all, node, len(leaf_row))
    rows = leaf_row[nodes[is_leaf]]
    n, k = len(rows), record["k"]
    y, values = standardized_blocks(x[rows], invariant_directions(x[rows]))
    lines = line[rows]
    counts = pd.Series(lines).value_counts()
    whole = [(name, int(c), int(totals[name])) for name, c in counts.items()]
    verdict = "tree and best pass" if record["p_tree"] <= 0.01 else ("only best passes" if record["p_best"] <= 0.01 else "neither passes")
    print(f"\n=== {record['node']}: {n} cells, k = {k}, {len(counts)} lines, joint verdict: {verdict}, fit: sibling p {fit[record['node']]['p']}, BH different {fit[record['node']]['bh_different']}")
    print("    lines in the node (cells here / cells of the line in the data): " + ", ".join(f"{a} {b}/{c}" for a, b, c in whole[:8]) + (f", and {len(whole) - 8} more" if len(whole) > 8 else ""))
    blocks = sorted({i for pair in record["crossing"] for i in pair[:2]})
    masks = {}
    for i in blocks:
        r2, mask = best_cut_one_direction(y[:, i])
        masks[i] = mask
        low, high, cut = side_summary(lines, mask)
        p = reference_p_value(to_rho(r2, n), n, 1)[0]
        print(f"    block {i + 1} (eigenvalue {values[i]:.2f}): rho {to_rho(r2, n):.2f}, p {p:.4f}, sides {int(mask.sum())} | {int((~mask).sum())}")
        for label, t in (("low side", low), ("high side", high)):
            print(f"        {label}: {len(t)} lines, {int(t.sum().sum())} cells: " + ", ".join(f"{name} {int(r[True] + r[False])}" for name, r in t.sort_values(True, ascending=False).head(6).iterrows() if r.sum() >= 3))
        print(f"        cut lines: {len(cut)}, {int(cut.sum().sum())} cells: " + ", ".join(f"{name} {int(r[True])}|{int(r[False])}" for name, r in cut.iterrows() if r.sum() >= 3))
    for i, j, ari in record["crossing"]:
        a, b = masks[i], masks[j]
        print(f"    blocks {i + 1} x {j + 1}: ARI {ari:.3f}; cells low/low {int((a & b).sum())}, low/high {int((a & ~b).sum())}, high/low {int((~a & b).sum())}, high/high {int((~a & ~b).sum())}")
        quadrant = pd.Series(np.where(a, "L", "H")) + pd.Series(np.where(b, "L", "H"))
        table = pd.crosstab(lines, quadrant.to_numpy())
        major = table[table.sum(1) >= max(5, 0.02 * n)]
        for name, r in major.iterrows():
            top = r.idxmax()
            print(f"        {name}: {int(r.sum())} cells, {100 * r.max() / r.sum():.0f}% in {top}  ({', '.join(f'{q} {int(v)}' for q, v in r.items() if v)})")

# ===== node_splits.py (follow-up) =====
"""Follow-up: one node's tree split and block cuts as leaf-set splits (Partition). usage: node_splits.py N97405"""
import sys
from pathlib import Path

import numpy as np
import pandas as pd

HERE = Path(__file__).resolve().parent
SPR = HERE.parent / "spr_alternatives_best_flat_cut_20261111"
sys.path.insert(0, str(HERE.parents[2]))
sys.path.insert(0, str(HERE))
sys.path.insert(0, str(SPR))

from block_lib import best_cut_one_direction, standardized_blocks  # noqa: E402
from spr_lib import index_tree, subtree  # noqa: E402

from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.best_flat_cut_reference import reference_p_value, to_rho  # noqa: E402
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import invariant_directions  # noqa: E402
from tree_break_selection.tree.elements import Partition  # noqa: E402
from tree_break_selection.tree.poset_tree import PosetTree  # noqa: E402

METHOD = "tbs_family_diffusion_markov_ward"
name = sys.argv[1]
source = HERE.parent / "choir_cell_lines_tbs_lbfgsb_20261106"
x = np.load(source / "pca.npy")
metadata = pd.read_csv(source / "cell_metadata.csv", index_col=0, dtype=str)
z = np.load(SPR / f"tree_{METHOD}.npz")
parent, leaf_row = z["parent"], z["leaf_row"]
n_leaves = len(leaf_row)
order, pre, sub_all = index_tree(parent)
nodes, local_parent, is_leaf, _ = subtree(parent, order, pre, sub_all, int(name[1:]), n_leaves)
rows = leaf_row[nodes[is_leaf]]
cells = metadata.index[rows].to_numpy()
lines = metadata.Cell_line.to_numpy()[rows]
n = len(rows)


def node_id(index):
    return f"L{index}" if index < n_leaves else f"N{index}"


tree = PosetTree()
for index in nodes:
    tree.add_node(node_id(index), is_leaf=bool(index < n_leaves), label=metadata.index[leaf_row[index]] if index < n_leaves else node_id(index))
for index, up in zip(nodes[1:], local_parent[1:], strict=True):
    tree.add_edge(node_id(nodes[up]), node_id(index))
tree.graph["root"] = name
tree.initialize_split_indices({cell: position for position, cell in enumerate(cells)})
encoding = tree.graph["taxa_encoding"]
splits = tree.to_splits(with_leaves=True)
everything = set(range(n))


def partition(mask):
    return Partition(tuple(int(i) for i in np.flatnonzero(mask)), encoding)


def composition(mask):
    counts = pd.Series(lines[mask]).value_counts()
    return ", ".join(f"{line} {count}" for line, count in counts.items())


def describe(label, side):
    """Is this leaf set a node of the tree? If not, which nodes of the tree make it up?"""
    mask = np.zeros(n, dtype=bool)
    mask[list(side.indices)] = True
    node = tree.find_node_by_split(side)
    print(f"  {label}: {len(side)} leaves ({composition(mask)})")
    if node is not None:
        print(f"      is the tree node {node}")
        return
    pieces = sorted(splits.maximals_under(side), key=lambda p: -len(p))
    singles = sum(len(p) == 1 for p in pieces)
    print(f"      is not a node of the tree; it is the union of {len(pieces)} tree nodes ({singles} of them single leaves). Largest:")
    for piece in pieces[:4]:
        piece_mask = np.zeros(n, dtype=bool)
        piece_mask[list(piece.indices)] = True
        print(f"        {tree.find_node_by_split(piece)}: {len(piece)} leaves ({composition(piece_mask)})")


print(f"=== {name}: {n} leaves; lines: {composition(np.ones(n, dtype=bool))}")
children = list(tree.successors(name))
tree_sides = [tree.nodes[child]["split_indices"] for child in children]
print("\nTree split (the node's two children)")
for child, side in zip(children, tree_sides, strict=True):
    describe(f"child {child}", side)

print("\nHow the tree holds each cell line of the node")
for line in pd.Series(lines).value_counts().index:
    describe(f"all leaves of {line}", partition(lines == line))

y, values = standardized_blocks(x[rows], invariant_directions(x[rows]))
table = pd.DataFrame({"cell": cells, "cell_line": lines, "tree_split": np.where(np.isin(np.arange(n), tree_sides[0].indices), children[0], children[1])})
cuts = {}
for block in range(y.shape[1]):
    r2, mask = best_cut_one_direction(y[:, block])
    p = float(reference_p_value(to_rho(r2, n), n, 1)[0])
    if p > 0.01 / y.shape[1]:
        continue
    cuts[block + 1] = (partition(mask), partition(~mask))
    table[f"block_{block + 1}_cut"] = np.where(mask, "low", "high")
    print(f"\nBlock {block + 1} cut (eigenvalue {values[block]:.2f}, rho {to_rho(r2, n):.2f}, p {p:.4f})")
    describe("low side", cuts[block + 1][0])
    describe("high side", cuts[block + 1][1])

print("\nCompatibility (two splits fit in one tree when one of the four intersections of their sides is empty)")
named = {"tree split": tree_sides[0], **{f"block {b} cut": low for b, (low, _) in cuts.items()}}
labels = list(named)
for i, a in enumerate(labels):
    for b in labels[i + 1 :]:
        first, second = named[a], named[b]
        rest_first, rest_second = partition(~np.isin(np.arange(n), first.indices)), partition(~np.isin(np.arange(n), second.indices))
        sizes = [len(u & v) for u in (first, rest_first) for v in (second, rest_second)]
        print(f"  {a} x {b}: compatible {first.is_compatible_with(second, everything)}; intersections {sizes}; smallest {min(sizes)} leaves")
out = HERE / f"leaves_{name}.csv"
table.to_csv(out, index=False)
print(f"\nleaf table: {out.relative_to(HERE.parents[2])} ({len(table)} rows)")

#   tree split x block 2 cut: compatible True; intersections [7, 0, 0, 36]; smallest 0 leaves

# ===== follow-up protocol (appended to protocol.md before the spr run) =====
# ## Follow-up protocol: apply the sprs in N96475, N96842 and N96808 and re-test (written before the run)
#
# In each of the three nodes the fit refuses the tree split, and the smaller side of block 1's cut is a cell-line
# boundary that is a union of 2 tree nodes (TE14 10 leaves in N96475; KYSE520 6 plus BICR16 1 in N96842 and in
# N96808). `apply_sprs.py` builds the node's subtree as a `PosetTree` with set based identity, joins the pieces of
# that side with `subtree_prune_regraft` and then regrafts the joined node on the edge above the node, so that
# the node's two children are (that side | the rest). Each node is treated on the original tree.
#
# Recorded: the sprs applied, the leaf sets changed (`to_splits` before against after), and the sibling reference
# (`best_flat_cut_reference`: k, `rho`, p) of the node's split before and after; the same for the split of the
# new "rest" child where it has 15 cells or more. The node keeps its cells, so its k and its best flat cut do not
# change; the joint best flat cut passes in all three nodes, so the new split can pass but need not.
#
# Expected: block 1's own cut has `rho` 4.27, 3.86 and 5.93 in one direction. In all k directions the same split
# has at least that between/total ratio, but the reference at (n, k) asks for more than at (n, 1); whether the new
# split passes at 0.01 is not predicted. Limits: three nodes; the side was chosen from the same cells that are
# tested; the per-split p-value only, without the edge gate, the BH step and pass-through; the fit is not rerun.

# ===== apply_sprs.py =====
"""Follow-up protocol in protocol.md: apply the sprs in one node and re-test its split. usage: apply_sprs.py N96475"""
import json
import sys
from pathlib import Path

import numpy as np
import pandas as pd

HERE = Path(__file__).resolve().parent
SPR = HERE.parent / "spr_alternatives_best_flat_cut_20261111"
sys.path.insert(0, str(HERE.parents[2]))
sys.path.insert(0, str(HERE))
sys.path.insert(0, str(SPR))

from block_lib import best_cut_one_direction, standardized_blocks  # noqa: E402
from spr_lib import index_tree, subtree  # noqa: E402

from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.best_flat_cut_reference import best_flat_cut_reference  # noqa: E402
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import invariant_directions  # noqa: E402
from tree_break_selection.tree.elements import Partition  # noqa: E402
from tree_break_selection.tree.poset_tree import PosetTree  # noqa: E402
from tree_break_selection.tree.spr import subtree_prune_regraft  # noqa: E402

METHOD = "tbs_family_diffusion_markov_ward"
name = sys.argv[1]
source = HERE.parent / "choir_cell_lines_tbs_lbfgsb_20261106"
x = np.load(source / "pca.npy")
metadata = pd.read_csv(source / "cell_metadata.csv", index_col=0, dtype=str)
z = np.load(SPR / f"tree_{METHOD}.npz")
parent, leaf_row = z["parent"], z["leaf_row"]
n_leaves = len(leaf_row)
order, pre, sub_all = index_tree(parent)
nodes, local_parent, is_leaf, _ = subtree(parent, order, pre, sub_all, int(name[1:]), n_leaves)
rows = leaf_row[nodes[is_leaf]]
cells = metadata.index[rows].to_numpy()
lines = metadata.Cell_line.to_numpy()[rows]
points = x[rows]
n = len(rows)
earlier = {r["node"]: r for r in map(json.loads, (SPR / "measures.jsonl").read_text().splitlines())}[name]


def node_id(index):
    return f"L{index}" if index < n_leaves else f"N{index}"


# The node's subtree below an extra root with one leaf for everything outside, so an spr can regraft above the node.
tree = PosetTree()
tree.add_node("ROOT", is_leaf=False, label="ROOT")
tree.add_node("OUT", is_leaf=True, label="outside")
for index in nodes:
    tree.add_node(node_id(index), is_leaf=bool(index < n_leaves), label=metadata.index[leaf_row[index]] if index < n_leaves else node_id(index))
for index, up in zip(nodes[1:], local_parent[1:], strict=True):
    tree.add_edge(node_id(nodes[up]), node_id(index))
tree.add_edge("ROOT", name)
tree.add_edge("ROOT", "OUT")
tree.graph["root"] = "ROOT"
encoding = {cell: position for position, cell in enumerate(cells)} | {"outside": n}
tree.initialize_split_indices(encoding)


def partition(mask):
    return Partition(tuple(int(i) for i in np.flatnonzero(mask)), encoding)


def mask_of(split):
    mask = np.zeros(n, dtype=bool)
    mask[[i for i in split.indices if i < n]] = True
    return mask


def composition(mask):
    return ", ".join(f"{line} {count}" for line, count in pd.Series(lines[mask]).value_counts().items())


def line_cut_share(mask):
    codes = pd.factorize(lines)[0]
    left, right = np.bincount(codes[mask], minlength=codes.max() + 1), np.bincount(codes[~mask], minlength=codes.max() + 1)
    return float(np.minimum(left, right).sum() / n)


def test(label, cell_mask, in_left):
    """Sibling reference of the split of the cells in ``cell_mask`` into ``in_left`` and the rest."""
    if cell_mask.sum() < 3 or in_left.sum() in (0, cell_mask.sum()):
        print(f"  {label}: not testable")
        return
    result = best_flat_cut_reference(points[cell_mask], in_left[cell_mask])
    print(f"  {label}: {int(in_left[cell_mask].sum())} | {int((~in_left)[cell_mask].sum())} leaves; {result.reason}, k {result.invariant_directions}, "
          f"rho {result.rho:.3f}, p {result.p_value:.4f}")


everyone = np.ones(n, dtype=bool)
node_set = partition(everyone)
before = tree.to_splits()
first_child = sorted(tree.successors(name), key=lambda c: -len(tree.nodes[c]["split_indices"]))[0]
tree_mask = mask_of(tree.nodes[first_child]["split_indices"])
y, _ = standardized_blocks(points, invariant_directions(points))
_, cut = best_cut_one_direction(y[:, 0])
target_mask = cut if cut.sum() <= n / 2 else ~cut
target = partition(target_mask)

print(f"=== {name}: {n} leaves ({composition(everyone)})")
print(f"joint best flat cut (earlier run): rho {earlier['rho_best']:.3f}, p {earlier['p_best']:.4f}")
print("\nBefore")
print(f"  tree split: [{composition(tree_mask)}] | [{composition(~tree_mask)}]; line cut share {line_cut_share(tree_mask):.3f}")
test("sibling reference of the tree split", everyone, tree_mask)
print(f"\nSide to make a node: {int(target_mask.sum())} leaves ({composition(target_mask)}); a tree node already: {tree.find_node_by_split(target) is not None}")

pieces = sorted(tree.to_splits(with_leaves=True).maximals_under(target), key=lambda p: -len(p))
print("  it is the union of the tree nodes: " + "; ".join(f"{tree.find_node_by_split(p)} ({composition(mask_of(p))})" for p in pieces))
sprs = 0
joined = pieces[0]
for piece in pieces[1:]:
    prune, regraft = tree.find_node_by_split(piece), tree.find_node_by_split(joined)
    subtree_prune_regraft(tree, prune, regraft)
    joined = Partition(tuple(sorted(set(joined.indices) | set(piece.indices))), encoding)
    sprs += 1
    print(f"  spr {sprs}: pruned {prune} ({len(piece)} leaves), regrafted on the edge above {regraft}")
target_node = tree.find_node_by_split(target)
holder = tree.find_node_by_split(node_set)
if next(iter(tree.predecessors(target_node))) != holder:
    subtree_prune_regraft(tree, target_node, holder)
    sprs += 1
    print(f"  spr {sprs}: pruned {target_node} ({len(target)} leaves), regrafted on the edge above {holder}")

after = tree.to_splits()
holder = tree.find_node_by_split(node_set)
children = sorted(tree.successors(holder), key=lambda c: len(tree.nodes[c]["split_indices"]))
child_masks = [mask_of(tree.nodes[c]["split_indices"]) for c in children]
assert (child_masks[0] == target_mask).all() and (child_masks[1] == ~target_mask).all()
print(f"\nAfter {sprs} sprs")
print(f"  the node's leaf set is now held by {holder}; children {children[0]} [{composition(child_masks[0])}] | {children[1]} [{composition(child_masks[1])}]")
print(f"  leaf sets removed {len(before - after)}, added {len(after - before)}, unchanged {len(before & after)} of {len(before)}; line cut share {line_cut_share(target_mask):.3f}")
test("sibling reference of the new split", everyone, target_mask)
rest = children[1]
grand = sorted(tree.successors(rest), key=lambda c: -len(tree.nodes[c]["split_indices"]))
if len(grand) == 2:
    rest_mask, grand_mask = child_masks[1], mask_of(tree.nodes[grand[0]]["split_indices"])
    print(f"  split of the rest ({rest}): [{composition(grand_mask)}] | [{composition(rest_mask & ~grand_mask)}]")
    test("sibling reference of the rest's split", rest_mask, grand_mask)
