#
# - Question: the third version (`results/analyses/variant_a_fixed_20261112/draft_rule_v3.py`) was run on the
#   Markov/Ward tree only. What does it give on the default tree (`tbs_family_diffusion`: symmetric operator,
#   average linkage)?
# - Tree: rebuilt from the saved linkage of 6 November
#   (`results/analyses/choir_cell_lines_tbs_lbfgsb_20261106/tbs_family_diffusion/linkage.npy`) as a parent array
#   (`build_tree.py`). Check of the conversion, done before this protocol: the Markov/Ward tree rebuilt the same
#   way from its saved linkage equals the saved Markov/Ward tree. The default tree's root splits 1 cell from
#   48,878.
# - Runs (`run_fit.py`): the harness of `spr_refit_20261111` with method `tbs_family_diffusion`, 48,879 cells, 30
#   PCs, opt-in reference on: (i) the current rule (table, k up to 15); (ii) the third version in the data order
#   of the rows and in two random orders (seeds 1 and 2).
# - Known before the run (wiki page `choir-cell-lines-recovery-plan`, `best-flat-cut-reference-real`): the default
#   tree holds 42 of the 190 lines as clean subtrees; today's test gives 75 clusters and ARI 0.0275 on it; the
#   reference with k up to 10 gave 53 clusters, ARI 0.0179, largest cluster 32,890.
# - Predictions, stated before the run:
#   - X1 the third version recovers at least as many lines as the current rule on this tree.
#   - X2 the ARI of both rules stays below 0.3.
#   - X3 over the three orders the third version's found clusters differ by at most 10.
# - Limits known before the run: one dataset; the tree is the one of 6 November, not rebuilt from the cells today.


# ===== build_tree.py =====
"""protocol.md: the default tree of the CHOIR cells as a parent array, from the saved linkage."""
from pathlib import Path

import numpy as np

HERE = Path(__file__).resolve().parent
Z = np.load(HERE.parent / "choir_cell_lines_tbs_lbfgsb_20261106" / "tbs_family_diffusion" / "linkage.npy")
n = len(Z) + 1
parent = np.full(2 * n - 1, -1, dtype=np.int64)
for i, (a, b) in enumerate(Z[:, :2].astype(np.int64)):
    parent[a] = parent[b] = n + i
assert (parent < 0).sum() == 1 and n == 48879
np.savez_compressed(HERE / "tree_default.npz", parent=parent, leaf_row=np.arange(n, dtype=np.int64))
print("nodes", len(parent), "root", int(np.flatnonzero(parent < 0)[0]))

"""Follow-up: the default-tree fit with the third version again, saving the edge and sibling columns of every node."""
import os
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parents[2]))
sys.path.insert(0, str(HERE.parent / "spr_refit_20261111"))
sys.path.insert(0, str(HERE.parent / "variant_a_fixed_20261112"))

import draft_rule_v3  # noqa: E402
import harness  # noqa: E402

import benchmarks.shared.runners.tbs_runner as tbs_runner  # noqa: E402

COLUMNS = ["leaf_count", "Child_Parent_Divergence_Test_Statistic", "Child_Parent_Divergence_df", "Child_Parent_Divergence_P_Value",
           "Child_Parent_Divergence_P_Value_BH", "Child_Parent_Divergence_Significant", "Child_Parent_Divergence_Tested",
           "Child_Parent_Divergence_Ancestor_Blocked", "Sibling_Divergence_Skipped", "Sibling_Divergence_P_Value", "Sibling_BH_Different"]
Original = tbs_runner.TreeDecomposition


class Recording(Original):
    def decompose_tree(self):
        out = super().decompose_tree()
        self.annotations_df[COLUMNS].to_csv(HERE / "default_data_annotations.csv.gz")
        edges = [(str(a), str(b), self.tree.edges[a, b].get("branch_length")) for a, b in self.tree.edges]
        import pandas as pd

        pd.DataFrame(edges, columns=["parent", "child", "branch_length"]).to_csv(HERE / "default_data_edges.csv.gz", index=False)
        return out


tbs_runner.TreeDecomposition = Recording
draft_rule_v3.install(None)
harness.HERE = HERE
harness.main("results/analyses/default_tree_third_version_20261112/tree_default.npz", "default_data_again", method="tbs_family_diffusion")
os._exit(0)  # shortcut: interpreter teardown stalled after earlier full-data runs (harness.py)
