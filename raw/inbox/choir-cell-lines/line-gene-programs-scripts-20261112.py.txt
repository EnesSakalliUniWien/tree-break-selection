# ===== protocol.md =====
# # Protocol: which gene programs carry the directions inside a cell line (frozen 2026-11-12, before the run)
#
# ## Why
#
# Inside one CHOIR cell line the cells are far from spherical for reasons specific to the line; a common
# covariance, sequencing depth and pools do not explain it (wiki page `line-non-sphericity`). Open question: which
# genes carry the leading directions inside a line?
#
# ## Looked up before this protocol
#
# - Kinker et al. 2020, Nature Genetics 52(11), 1208-1218, https://doi.org/10.1038/s41588-020-00726-6 (PubMed
#   record, abstract): the paper of this dataset reports 12 expression programs that are recurrently
#   heterogeneous within multiple cell lines, among them cell cycle, senescence, stress and interferon responses,
#   epithelial-mesenchymal transition and protein metabolism.
# - Tirosh et al. 2016, Science 352(6282), 189-196, https://doi.org/10.1126/science.aad0501 (PubMed record): the
#   publication the scanpy documentation cites for the cell-cycle gene list of 97 genes, 43 for S phase and 54
#   for G2M (https://scanpy.readthedocs.io/en/1.11.x/how-to/cell-cycle.html).
# - The gene symbols themselves could not be retrieved from the original file (the documented GitHub paths return
#   "not found"). They are taken from a secondary page that prints lists it attributes to the Seurat package,
#   https://natian-and-ryabhatta.web.app/cell-cyle-scores.html : 42 S phase symbols and 51 G2M symbols as returned
#   by the lookup (`cell_cycle_genes.json`). This is 4 symbols fewer than the documented 97; the list is marked as
#   a secondary copy, and symbols absent from the data are reported.
#
# ## Data and definitions
#
# - Expression: `local_data/choir/kinker_2020/kinker_2020_choir_cells_cpm.h5ad`, transformed as for the PCs:
#   log1p(CPM / 100), no scaling. PCs and labels as in `line_non_sphericity_20261112`. Lines: the 175 cell lines
#   with at least 100 cells.
# - Per line: the three leading eigenvectors of the line's covariance in the 30 PCs and the cells' scores on them
#   (direction 1, 2, 3).
# - Cell-cycle scores per cell and line: for the S list and for the G2M list, the mean over the list's genes of
#   the expression standardized within the line (genes with zero variance in the line are left out).
#
# ## Measurements (`measure.py`, outputs `measure.txt`, `measure.json`)
#
# - M1 genes of the directions, without any gene list: per line and direction the 50 genes with the largest
#   absolute correlation with the direction's score, among genes detected in at least 10% of the line's cells.
#   Recurrence: for every gene the number of lines in which it is among the 50 of direction 1 (and of direction
#   2, 3, and of any of the three); the 40 most recurrent genes with their counts; the count expected if the 50
#   were drawn at random.
# - M2 cell cycle: per line the absolute correlation of the scores of directions 1, 2, 3 with the S score and the
#   G2M score; the share of the within-line variance of the 30 PCs explained by a linear fit on both scores; T1
#   (largest eigenvalue over median eigenvalue), k current rule and k prototype rule after removing that fit; the
#   number of the 50 genes of direction 1 that are in the cell-cycle list.
#   Control: the same share for 20 random pairs of gene sets of the same sizes (genes detected in at least 10% of
#   all cells).
# - M3 after removing the cell-cycle fit: M1 repeated on the remaining within-line variation, direction 1 only:
#   the 40 most recurrent genes.
# - Self-checks (asserts): T1 of the 175 lines in PC coordinates has median 25.64; the cell order of the
#   expression file equals the metadata; at least 80 of the 93 cell-cycle symbols are gene names of the data.
#
# ## Predictions, stated before the run
#
# - P1 the cell-cycle fit explains more than 10% of the within-line variance at the median (depth: 3.0%).
# - P2 in more than half of the lines one of the directions 1 to 3 correlates with the S or the G2M score at 0.5
#   or more in absolute value.
# - P3 removing the cell-cycle fit does not make the lines spherical: the median T1 stays above 10.
#
# ## Limits known before the run
#
# The cell-cycle list is a secondary copy with 93 of 97 symbols. No other program of the Kinker paper has a
# looked-up gene list here, so genes found in M1 and M3 are reported by symbol without a biological label. One
# dataset; lines with at least 100 cells; linear fits only.


# ===== deviations.md =====
# # Deviations (written 2026-11-12, after the result)
#
# - `after.py` was added after `measure.py` showed that the most recurrent genes of direction 1 are not in the
#   cell-cycle list. It scores the 86 genes whose symbols are RPL or RPS followed by a number (plus RPLP0 to
#   RPLP2 and RPSA), and fits cell cycle, that score and log depth together. It is not part of the protocol and
#   the gene set was chosen after seeing the gene names in M1 and M3.
# - `measure.py` was not changed after its run.

# ===== measure.py =====
"""protocol.md M1 to M3: the genes of the leading directions inside each cell line, and the cell cycle."""
import json
import sys
from collections import Counter
from pathlib import Path

import anndata as ad
import numpy as np
import pandas as pd

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[2]
sys.path.insert(0, str(ROOT))
sys.path.insert(0, str(HERE.parent / "k_margin_prototype_20261112"))

from k_rule import directions_with_margin  # noqa: E402

from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import (  # noqa: E402
    invariant_directions,
)

TOP, DIRECTIONS = 50, 3
rng = np.random.default_rng(20261112)
lines_out = []


def say(text=""):
    print(text, flush=True)
    lines_out.append(text)


def spread(values, digits=3):
    values = np.asarray(values, dtype=float)
    return f"median {np.median(values):.{digits}f} (min {values.min():.{digits}f}, max {values.max():.{digits}f})"


def t1(cells):
    values = np.linalg.eigvalsh(np.cov(cells.T))
    return float(values[-1] / np.median(values))


def leading(cells):
    """Scores of the cells on the three leading eigenvectors of their covariance."""
    centred = cells - cells.mean(axis=0)
    _, vectors = np.linalg.eigh(np.cov(centred.T))
    return centred @ vectors[:, ::-1][:, :DIRECTIONS]


def correlations(expression, scores):
    """Pearson correlation of every gene (columns of the standardized expression) with every score column."""
    z = (scores - scores.mean(axis=0)) / scores.std(axis=0)
    return expression.T @ z / len(z)


def remove_fit(cells, covariates):
    design = np.column_stack([np.ones(len(cells)), covariates])
    return cells - design @ np.linalg.lstsq(design, cells, rcond=None)[0]


source = HERE.parent / "choir_cell_lines_tbs_lbfgsb_20261106"
metadata = pd.read_csv(source / "cell_metadata.csv", index_col=0, dtype=str)
pcs = np.load(source / "pca.npy")
cpm = ad.read_h5ad(ROOT / "local_data/choir/kinker_2020/kinker_2020_choir_cells_cpm.h5ad")
assert (cpm.obs_names.to_numpy() == metadata.index.to_numpy()).all(), "cell order differs"
genes = np.asarray(cpm.var_names)
matrix = cpm.X.tocsr()
cycle = json.loads((HERE / "cell_cycle_genes.json").read_text())
position = {gene: i for i, gene in enumerate(genes)}
s_index = np.array([position[g] for g in cycle["s"] if g in position])
g2m_index = np.array([position[g] for g in cycle["g2m"] if g in position])
missing = [g for g in cycle["s"] + cycle["g2m"] if g not in position]
say(f"cells {matrix.shape[0]}, genes {matrix.shape[1]}; cell-cycle symbols found: S {len(s_index)} of {len(cycle['s'])}, "
    f"G2M {len(g2m_index)} of {len(cycle['g2m'])}; not in the data: {missing}")
assert len(s_index) + len(g2m_index) >= 80
cycle_set = set(s_index) | set(g2m_index)
detected_all = np.asarray((matrix > 0).mean(axis=0)).ravel() >= 0.10
random_sets = [(rng.choice(np.flatnonzero(detected_all), len(s_index), replace=False),
                rng.choice(np.flatnonzero(detected_all), len(g2m_index), replace=False)) for _ in range(20)]

groups = {line: index for line, index in metadata.groupby("Cell_line").indices.items() if len(index) >= 100}
recurrence = {name: Counter() for name in ("direction 1", "direction 2", "direction 3", "any of the three", "direction 1 after the cell cycle")}
rows, eligible_counts = [], []
for line, index in groups.items():
    cells = pcs[index]
    dense = np.log1p(matrix[index].toarray() / 100.0)
    keep = (dense > 0).mean(axis=0) >= 0.10
    sd = dense.std(axis=0)
    usable = keep & (sd > 0)
    standardized = np.zeros_like(dense)
    standardized[:, sd > 0] = (dense[:, sd > 0] - dense[:, sd > 0].mean(axis=0)) / sd[sd > 0]
    eligible = np.flatnonzero(usable)
    eligible_counts.append(len(eligible))
    scores = leading(cells)

    # M1
    r = correlations(standardized[:, eligible], scores)
    tops = [eligible[np.argsort(-np.abs(r[:, j]))[:TOP]] for j in range(DIRECTIONS)]
    for j in range(DIRECTIONS):
        recurrence[f"direction {j + 1}"].update(tops[j].tolist())
    recurrence["any of the three"].update(set(np.concatenate(tops).tolist()))

    # M2
    def set_scores(a, b):
        return np.column_stack([standardized[:, a][:, sd[a] > 0].mean(axis=1), standardized[:, b][:, sd[b] > 0].mean(axis=1)])

    cc = set_scores(s_index, g2m_index)
    z = (scores - scores.mean(axis=0)) / scores.std(axis=0)
    zc = (cc - cc.mean(axis=0)) / cc.std(axis=0)
    corr = np.abs(z.T @ zc / len(z))  # directions x (S, G2M)
    trace = np.trace(np.cov(cells.T))
    residual = remove_fit(cells, cc)
    share = 1 - np.trace(np.cov(residual.T)) / trace
    control = [1 - np.trace(np.cov(remove_fit(cells, set_scores(a, b)).T)) / trace for a, b in random_sets]

    # M3
    r_after = correlations(standardized[:, eligible], leading(residual)[:, :1])
    recurrence["direction 1 after the cell cycle"].update(eligible[np.argsort(-np.abs(r_after[:, 0]))[:TOP]].tolist())

    rows.append(dict(line=line, n=len(index), t1=t1(cells), t1_after=t1(residual), share_cell_cycle=float(share),
                     share_random_median=float(np.median(control)), share_random_max=float(np.max(control)),
                     corr_s=corr[:, 0].tolist(), corr_g2m=corr[:, 1].tolist(), max_corr=float(corr.max()),
                     max_corr_direction_1=float(corr[0].max()), top50_in_cell_cycle=int(len(set(tops[0].tolist()) & cycle_set)),
                     k_current=len(invariant_directions(cells)), k_current_after=len(invariant_directions(residual)),
                     k_prototype=len(directions_with_margin(cells)), k_prototype_after=len(directions_with_margin(residual))))
    if len(rows) % 25 == 0:
        print(len(rows), "lines done", flush=True)

table = pd.DataFrame(rows)
assert len(table) == 175 and abs(table["t1"].median() - 25.64) < 0.01, table["t1"].median()
expected = TOP / np.median(eligible_counts) * len(table)
say(f"lines {len(table)}; genes eligible per line {spread(eligible_counts, 0)}; a gene drawn at random is among the {TOP} of a direction in "
    f"{expected:.2f} of {len(table)} lines on average")

say("\n== M1: genes of the directions, most recurrent across the 175 lines (gene: lines; * = in the cell-cycle list) ==")
for name in ("direction 1", "direction 2", "direction 3", "any of the three"):
    top = recurrence[name].most_common(40)
    say(f"{name}: " + ", ".join(f"{genes[g]}{'*' if g in cycle_set else ''}: {c}" for g, c in top))
    say(f"  of these 40, in the cell-cycle list: {sum(g in cycle_set for g, _ in top)}; genes among the {TOP} in at least 20 lines: "
        f"{sum(c >= 20 for c in recurrence[name].values())}")

say("\n== M2: cell cycle ==")
say(f"share of the within-line variance explained by the fit on the S and G2M scores: {spread(table['share_cell_cycle'])}")
say(f"control, 20 random pairs of gene sets of the same sizes: per-line median share {spread(table['share_random_median'])}; "
    f"per-line largest share {spread(table['share_random_max'])}")
say(f"largest absolute correlation of directions 1 to 3 with the S or G2M score: {spread(table['max_corr'])}; "
    f"at least 0.5 in {int((table['max_corr'] >= 0.5).sum())} of {len(table)} lines; at least 0.8 in {int((table['max_corr'] >= 0.8).sum())}")
say(f"direction 1 alone: {spread(table['max_corr_direction_1'])}; at least 0.5 in {int((table['max_corr_direction_1'] >= 0.5).sum())} of {len(table)}")
for j in range(DIRECTIONS):
    best = np.array([max(row["corr_s"][j], row["corr_g2m"][j]) for row in rows])
    say(f"  direction {j + 1}: the larger of the two correlations is at least 0.5 in {int((best >= 0.5).sum())} lines; {spread(best)}")
say(f"of the {TOP} genes of direction 1, in the cell-cycle list: {spread(table['top50_in_cell_cycle'], 0)}; at least 10 in "
    f"{int((table['top50_in_cell_cycle'] >= 10).sum())} lines")
say(f"T1 in PC coordinates: {spread(table['t1'], 2)}; after removing the cell-cycle fit: {spread(table['t1_after'], 2)}")
say(f"k current rule: {dict(sorted(Counter(table['k_current']).items()))} -> {dict(sorted(Counter(table['k_current_after']).items()))}")
say(f"k prototype rule: {dict(sorted(Counter(table['k_prototype']).items()))} -> {dict(sorted(Counter(table['k_prototype_after']).items()))}")

say("\n== M3: direction 1 after removing the cell-cycle fit, most recurrent genes ==")
top = recurrence["direction 1 after the cell cycle"].most_common(40)
say(", ".join(f"{genes[g]}{'*' if g in cycle_set else ''}: {c}" for g, c in top))
say(f"of these 40, in the cell-cycle list: {sum(g in cycle_set for g, _ in top)}")

say("\n== predictions ==")
say(f"P1 share above 0.10 at the median: {'held' if table['share_cell_cycle'].median() > 0.10 else 'not held'} ({table['share_cell_cycle'].median():.3f})")
say(f"P2 correlation of at least 0.5 in more than half of the lines: {'held' if (table['max_corr'] >= 0.5).sum() > len(table) / 2 else 'not held'} "
    f"({int((table['max_corr'] >= 0.5).sum())} of {len(table)})")
say(f"P3 median T1 after the cell-cycle fit above 10: {'held' if table['t1_after'].median() > 10 else 'not held'} ({table['t1_after'].median():.2f})")
(HERE / "measure.txt").write_text("\n".join(lines_out) + "\n")
(HERE / "measure.json").write_text(json.dumps(dict(lines=rows, recurrence={name: {str(genes[g]): c for g, c in counter.most_common(200)}
                                                                              for name, counter in recurrence.items()}), indent=1))

g2m_index = np.array([position[g] for g in cycle["g2m"] if g in position])
rp_index = np.array([i for i, g in enumerate(genes) if re.fullmatch(r"RP[LS]\d+[AXYL]?\d*|RPLP\d|RPSA", g)])
print("genes named RPL/RPS used:", len(rp_index), list(genes[rp_index][:12]), "...")
depth = np.array([matrix[i].data.sum() / matrix[i].data.min() for i in range(matrix.shape[0])])  # as on 7 November
log_depth = np.log(depth)


def spread(values, digits=3):
    values = np.asarray(values, dtype=float)
    return f"median {np.median(values):.{digits}f} (min {values.min():.{digits}f}, max {values.max():.{digits}f})"


def remove_fit(cells, covariates):
    design = np.column_stack([np.ones(len(cells)), covariates])
    return cells - design @ np.linalg.lstsq(design, cells, rcond=None)[0]


def t1(cells):
    values = np.linalg.eigvalsh(np.cov(cells.T))
    return float(values[-1] / np.median(values))


rows = []
for line, index in metadata.groupby("Cell_line").indices.items():
    if len(index) < 100:
        continue
    cells = pcs[index]
    dense = np.log1p(matrix[index].toarray() / 100.0)
    sd = dense.std(axis=0)

    def score(columns):
        columns = columns[sd[columns] > 0]
        return ((dense[:, columns] - dense[:, columns].mean(axis=0)) / sd[columns]).mean(axis=1)

    factors = {"cell cycle": np.column_stack([score(s_index), score(g2m_index)]), "RPL/RPS": score(rp_index)[:, None],
               "log depth": log_depth[index][:, None]}
    factors["all three"] = np.column_stack(list(factors.values()))
    centred = cells - cells.mean(axis=0)
    _, vectors = np.linalg.eigh(np.cov(centred.T))
    first = centred @ vectors[:, -1]
    trace = np.trace(np.cov(cells.T))
    row = dict(line=line, corr_first_rp=abs(float(np.corrcoef(first, factors["RPL/RPS"][:, 0])[0, 1])),
               corr_rp_depth=float(np.corrcoef(factors["RPL/RPS"][:, 0], factors["log depth"][:, 0])[0, 1]))
    for name, covariates in factors.items():
        residual = remove_fit(cells, covariates)
        row[f"share {name}"] = float(1 - np.trace(np.cov(residual.T)) / trace)
        row[f"t1 {name}"] = t1(residual)
    rows.append(row)
table = pd.DataFrame(rows)
print("lines", len(table))
for name in ("cell cycle", "RPL/RPS", "log depth", "all three"):
    print(f"{name}: share of the within-line variance explained {spread(table[f'share {name}'])}; T1 after removing it {spread(table[f't1 {name}'], 2)}")
print(f"direction 1 against the RPL/RPS score, absolute correlation: {spread(table['corr_first_rp'])}; at least 0.5 in "
      f"{int((table['corr_first_rp'] >= 0.5).sum())} of {len(table)} lines")
print(f"RPL/RPS score against log depth, correlation: {spread(table['corr_rp_depth'])}")
table.to_json(HERE / "after.json", orient="records", indent=1)
