# ===== protocol.md =====
# # Protocol: how far is each tested split from the best flat cut in the rebuilt directions (frozen 2026-11-11, before the run)
#
# Descriptive; no production change. Data and setup of `best_flat_cut_reference_real_20261111` (48,879 CHOIR
# cells, 30 PCs, saved diffusion coordinates, registry parameters, both registered methods,
# `sibling_best_flat_cut_reference=True`). The fit is repeated only to capture the cells of every tested split;
# replay check: 138 and 53 clusters.
#
# For every tested split whose subtree has at least one invariant direction (`invariant_directions` of the
# subtree's own cells, the rebuilt geometry), in those directions:
#
# - tree split: its `rho`, and the size of its smaller side;
# - nearest flat cut: start at the tree split, re-sort along its mean difference and move single cells until
#   stable (`resort`, `swap` of the table builder); its `rho` and the adjusted Rand index (ARI) with the tree split;
# - best flat cut: the table's search (grid, top 50 re-sorts, single-cell moves); its `rho`, its smaller side,
#   and the ARI with the tree split. For k above 10 the search gives a lower bound only;
# - reference p-values of the tree split and of the best flat cut where n >= 15 and k <= 10;
# - cell lines: for each of the three partitions the share of the subtree's cells that sit on the minority side
#   of their own cell line ("line cut share"; 0 means no line is cut).
#
# Summaries by subtree size (32-159, 160-999, 1,000 and more) and tree, with medians and counts.
#
# Expected before running: the 9 November run gave a median tree-split `rho` of 0.97 (32-159 cells) and 1.00
# (160-999 cells), below the 1.75 that the median cut of one Gaussian population reaches, and 3.13 at 1,000
# cells and more. So in small subtrees the best flat cut should differ from the tree split in most cases (low
# ARI) and the tree split should rarely be a flat cut itself; in large subtrees agreement should be higher.
# Whether the best flat cut cuts fewer cell lines than the tree split is not predicted. The best flat cut may
# isolate a few outlying cells; its smaller side is recorded for that reason.
#
# Limits known before running: dependent splits of two trees on one dataset; cell lines are the only truth; the
# best flat cut is found by search, not proved.


# ===== measure_lib.py =====
"""Best flat cut of a subtree with its partition, and the flat cut nearest to a given split."""
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parents[3] / "scripts"))
from build_best_flat_cut_reference import _along_one, directions, resort, swap  # noqa: E402

from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.best_flat_cut_reference import (  # noqa: E402
    to_rho,
    whiten,
)


def r2_of(y, mask):
    s = int(mask.sum())
    return float((y[mask].sum(0) ** 2).sum() / (s * (len(y) - s)))


def best_flat_cut(coordinates, top=50):
    """The table's search (grid, top re-sorts, single-point moves), returning rho and the partition."""
    y = whiten(coordinates)
    n, k = y.shape
    dirs = directions(k)
    s = np.arange(1, n)[:, None]
    chunk = max(1, min(2000, int(2e7 // (n * k))))
    best = np.empty(len(dirs))
    for a in range(0, len(dirs), chunk):
        g = dirs[a : a + chunk]
        order = np.argsort(y @ g.T, axis=0)
        prefix = np.cumsum(y[order], axis=0)[:-1]
        best[a : a + chunk] = ((prefix**2).sum(-1) / (s * (n - s))).max(0)
    out_r2, out_mask = -1.0, None
    for j in np.argsort(-best, kind="stable")[:top]:
        r2, mask = _along_one(y, dirs[j])
        r2, mask = resort(y, mask, r2)
        r2, mask = swap(y, mask, r2)
        if r2 > out_r2:
            out_r2, out_mask = float(r2), mask
    return float(to_rho(out_r2, n)), out_mask


def nearest_flat_cut(coordinates, in_left):
    """Start at the given split, re-sort along its mean difference and move single cells until stable."""
    y = whiten(coordinates)
    start = r2_of(y, in_left)
    r2, mask = resort(y, in_left, start)
    r2, mask = swap(y, mask, r2)
    return float(to_rho(start, len(y))), float(to_rho(r2, len(y))), mask


# ===== fit.py =====
"""protocol.md: refit one method with the reference and save the cells of every tested split. usage: fit.py METHOD"""
import json
import os
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parents[2]))

import numpy as np  # noqa: E402
import pandas as pd  # noqa: E402

import tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.inflated_projected_wald_annotation.pipeline as pipeline  # noqa: E402,E501
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import _leaf_rows  # noqa: E402

ORIGINAL = pipeline.apply_best_flat_cut_reference
SPLITS = []


def recording(tree, leaf_data, records, tested_parent_ids, summaries, method_labels, decisions):
    out = ORIGINAL(tree, leaf_data, records, tested_parent_ids, summaries, method_labels, decisions)
    row_of_label = {label: row for row, label in enumerate(leaf_data.index)}
    by_parent = {record.parent: record for record in records}
    for parent, summary, label, decision in zip(tested_parent_ids, out[0], out[1], out[2], strict=True):
        if decision is None:
            continue
        record = by_parent[parent]
        SPLITS.append({"node": str(parent), "reason": label.split(":")[-1], "k": int(summary[1]), "p": float(decision.p_value),
                       "left": _leaf_rows(tree, record.left, row_of_label), "right": _leaf_rows(tree, record.right, row_of_label)})
    return out


pipeline.apply_best_flat_cut_reference = recording


def main(method):
    from scipy.spatial.distance import pdist
    from threadpoolctl import threadpool_limits

    from benchmarks.shared.runners.dispatch import _tbs_branch_length_optimization_kwargs
    from benchmarks.shared.runners.method_registry import METHOD_SPECS
    from benchmarks.shared.runners.tbs_runner import run_tbs_on_distance
    from tree_break_selection.tree.feature_space import continuous_feature_space_from_columns

    source = HERE.parent / "choir_cell_lines_tbs_lbfgsb_20261106"
    metadata = pd.read_csv(source / "cell_metadata.csv", index_col=0, dtype=str)
    data = pd.DataFrame(np.load(source / "pca.npy"), index=metadata.index, columns=[f"PC{i + 1}" for i in range(30)])
    params = dict(METHOD_SPECS[method].param_grid[0])
    distance_path = HERE / f"diffusion_distance_{method}.npy"
    distances = np.lib.format.open_memmap(distance_path, mode="w+", dtype=np.float64, shape=(len(data) * (len(data) - 1) // 2,))
    pdist(np.load(source / method / "coordinates.npy"), metric="euclidean", out=distances)
    distances.flush()
    try:
        with threadpool_limits(limits=4):
            result = run_tbs_on_distance(
                data, distances, 0.01, edge_alpha=0.001, tree_linkage_method=params["tree_linkage_method"],
                feature_space=continuous_feature_space_from_columns(tuple(data.columns)), branch_length_data_df=data,
                edge_branch_length_variance_policy=params["edge_branch_length_variance_policy"],
                adaptive_projection_dimension_energy_fraction=0.90, sibling_best_flat_cut_reference=True,
                **_tbs_branch_length_optimization_kwargs(params),
                extra={"feature_family_diffusion_fallback_reason": None})
        bh = result.extra["annotations"]["Sibling_BH_Different"].astype(bool)
        rows = np.concatenate([np.r_[s["left"], s["right"]] for s in SPLITS]).astype(np.int32)
        sizes = np.array([[len(s["left"]), len(s["right"])] for s in SPLITS], dtype=np.int64)
        np.savez_compressed(HERE / f"splits_{method}.npz", rows=rows, sizes=sizes)
        meta = [{"node": s["node"], "reason": s["reason"], "k": s["k"], "p": s["p"], "bh_different": bool(bh[s["node"]])} for s in SPLITS]
        (HERE / f"splits_{method}.json").write_text(json.dumps({"found_clusters": int(result.found_clusters), "splits": meta}))
        print(method, "clusters", result.found_clusters, "tested splits", len(SPLITS), flush=True)
    finally:
        distances._mmap.close()
        distance_path.unlink()


if __name__ == "__main__":
    main(sys.argv[1])
    os._exit(0)  # shortcut: interpreter teardown stalled after earlier full-data runs (run_lbfgsb.py)


# ===== measure.py =====
"""protocol.md: tree split, nearest flat cut and best flat cut of every tested split with invariant directions."""
import json
import sys
from multiprocessing import Pool
from pathlib import Path

import numpy as np
import pandas as pd
from sklearn.metrics import adjusted_rand_score

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parents[2]))
sys.path.insert(0, str(HERE))

from measure_lib import best_flat_cut, nearest_flat_cut  # noqa: E402

from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.best_flat_cut_reference import (  # noqa: E402
    MAX_DIRECTIONS,
    MIN_CELLS,
    reference_p_value,
    split_rho,
)
from tree_break_selection.hierarchy_analysis.statistics.sibling_divergence.split_count_rule_out import invariant_directions  # noqa: E402

METHODS = ["tbs_family_diffusion_markov_ward", "tbs_family_diffusion"]
STATE = {}


def init():
    source = HERE.parent / "choir_cell_lines_tbs_lbfgsb_20261106"
    STATE["x"] = np.load(source / "pca.npy")
    lines = pd.read_csv(source / "cell_metadata.csv", index_col=0, dtype=str).Cell_line
    STATE["line"] = pd.factorize(lines)[0]
    for method in METHODS:
        z = np.load(HERE / f"splits_{method}.npz")
        STATE[method] = (z["rows"], z["sizes"], np.r_[0, np.cumsum(z["sizes"].sum(1))])


def line_cut_share(lines, mask):
    left = np.bincount(lines[mask], minlength=lines.max() + 1)
    right = np.bincount(lines[~mask], minlength=lines.max() + 1)
    return float(np.minimum(left, right).sum() / len(lines))


def job(args):
    method, i = args
    rows, sizes, offsets = STATE[method]
    idx = rows[offsets[i] : offsets[i + 1]]
    n_left = int(sizes[i, 0])
    points = STATE["x"][idx]
    in_left = np.arange(len(idx)) < n_left
    d = invariant_directions(points)
    out = {"method": method, "i": i, "n": len(idx), "k": len(d), "small_tree": int(min(n_left, len(idx) - n_left))}
    if len(d) == 0:
        return out
    c = points @ d.T
    lines = STATE["line"][idx]
    rho_tree, rho_near, near = nearest_flat_cut(c, in_left)
    rho_best, best = best_flat_cut(c)
    out.update(rho_tree=rho_tree, rho_tree_check=float(split_rho(c, in_left)), rho_near=rho_near, rho_best=rho_best,
               ari_near=float(adjusted_rand_score(in_left, near)), ari_best=float(adjusted_rand_score(in_left, best)),
               moved_near=float(min((near != in_left).mean(), (near == in_left).mean())),
               small_best=int(min(best.sum(), (~best).sum())),
               cut_tree=line_cut_share(lines, in_left), cut_near=line_cut_share(lines, near), cut_best=line_cut_share(lines, best),
               lines_in_subtree=int(len(np.unique(lines))))
    if len(idx) >= MIN_CELLS and len(d) <= MAX_DIRECTIONS:
        out.update(p_tree=float(reference_p_value(rho_tree, len(idx), len(d))[0]), p_best=float(reference_p_value(rho_best, len(idx), len(d))[0]))
    return out


if __name__ == "__main__":
    init()
    jobs = sorted(((m, i) for m in METHODS for i in range(len(STATE[m][1]))), key=lambda a: -int(STATE[a[0]][1][a[1]].sum()))
    meta = {m: json.loads((HERE / f"splits_{m}.json").read_text())["splits"] for m in METHODS}
    with Pool(10, initializer=init) as pool, (HERE / "measures.jsonl").open("w") as f:
        for done, row in enumerate(pool.imap_unordered(job, jobs), 1):
            row.update({key: meta[row["method"]][row["i"]][key] for key in ("node", "reason", "p", "bh_different")})
            f.write(json.dumps(row) + "\n")
            if done % 200 == 0:
                print(done, "of", len(jobs), flush=True)
    print("DONE", flush=True)

    splits=("n", "size"), k=q("k"), rho_tree=q("rho_tree"), rho_near=q("rho_near"), rho_best=q("rho_best"),
    ari_near=q("ari_near"), ari_best=q("ari_best"), moved_near=q("moved_near"), tree_is_flat=("tree_is_flat", "sum"),

    ari_best_lt_05=("ari_best", lambda x: (x < 0.5).sum()), best_tiny=("best_tiny", "sum"),
    cut_tree=q("cut_tree"), cut_near=q("cut_near"), cut_best=q("cut_best")).round(3).T.to_string())

    "splits": len(g), "tree_only": int(((g.p_tree <= 0.01) & (g.p_best > 0.01)).sum()), "both": int(((g.p_tree <= 0.01) & (g.p_best <= 0.01)).sum()),

acc = d[d.bh_different]
print("\nsplits accepted by BH:", len(acc), "| median ARI best vs tree:", round(acc.ari_best.median(), 3), "| tree split is a flat cut:", int(acc.tree_is_flat.sum()))
print("\nk above 10 (search gives a lower bound):", int((d.k > 10).sum()), "splits; median ARI best vs tree", round(d[d.k > 10].ari_best.median(), 3),
      "median rho tree / best", round(d[d.k > 10].rho_tree.median(), 2), round(d[d.k > 10].rho_best.median(), 2))
m.to_csv(HERE / "measures.csv", index=False)
