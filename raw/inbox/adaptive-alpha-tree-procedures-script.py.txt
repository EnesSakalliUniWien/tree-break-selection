"""Tree-constrained alpha procedures from protocol.md, on one family's node list.

Each node is a dict with position, depth, parent (position or None), p_value (None if
unresolved), rows, n and merge_height, in breadth-first order (parents first).
"""

from itertools import product

import numpy as np

ALPHA = 0.05


def tree_reject(nodes, levels):
    """Reject a node when its parent was rejected (or it is the root) and p <= its level."""
    rejected = set()
    for node in nodes:
        parent_open = node["parent"] is None or node["parent"] in rejected
        p = node["p_value"]
        if parent_open and p is not None and p <= levels[node["position"]]:
            rejected.add(node["position"])
    return rejected


def bonferroni7(nodes):
    return tree_reject(nodes, {v["position"]: ALPHA / 7 for v in nodes})


def bonferroni3(nodes):
    return tree_reject(nodes, {v["position"]: ALPHA / 3 if v["depth"] < 2 else 0.0 for v in nodes})


def allocation_levels(nodes, c, lam):
    """alpha * c_d * (lam_d * n_v / n + (1 - lam_d) * 2^-d): sums to <= alpha on any tree."""
    return {
        v["position"]: ALPHA * c[v["depth"]]
        * (lam[v["depth"]] * v["rows"] / v["n"] + (1 - lam[v["depth"]]) * 2.0 ** -v["depth"])
        for v in nodes
    }


CA_C, CA_LAM = (1 / 3, 1 / 3, 1 / 3), (1.0, 1.0, 1.0)


def chain_conditioned(nodes):
    return tree_reject(nodes, allocation_levels(nodes, CA_C, CA_LAM))


def manifold_weighted(nodes):
    """Exploratory: CA's depth share split by merge height within each depth."""
    totals = {}
    for v in nodes:
        totals[v["depth"]] = totals.get(v["depth"], 0.0) + v["merge_height"]
    return tree_reject(nodes, {
        v["position"]: ALPHA * CA_C[v["depth"]] * v["merge_height"] / totals[v["depth"]] for v in nodes
    })


def alpha_investing(nodes, alpha=ALPHA):
    """Foster-Stine form (Ramdas et al. 2017, Table 2): W0 = B0 = alpha, phi = W / 2."""
    children = {}
    for v in nodes:
        children.setdefault(v["parent"], []).append(v)
    queue, wealth, rejected, levels = list(children[None]), alpha, set(), {}
    while queue:
        node = queue.pop(0)
        phi = wealth / 2
        level = phi / (1 + phi)
        levels[node["position"]] = level
        wealth -= phi
        if node["p_value"] is not None and node["p_value"] <= level:
            wealth += phi + alpha
            rejected.add(node["position"])
            queue += children.get(node["position"], [])
    return rejected, levels


def simplex_grid(step=0.05):
    k = round(1 / step)
    return [(a / k, b / k, (k - a - b) / k) for a in range(k + 1) for b in range(k + 1 - a)]


LAM_GRID = (0.0, 0.25, 0.5, 0.75, 1.0)


def learn_allocation(families):
    """Grid point maximising total mixed nodes rejected; earliest grid point wins ties."""
    best, best_score = None, -1
    for c in simplex_grid():
        for lam1, lam2 in product(LAM_GRID, LAM_GRID):
            lam = (1.0, lam1, lam2)  # depth 0 has n_v / n = 2^0 = 1, so lam_0 is irrelevant
            score = sum(
                sum(1 for v in nodes if not v["null"] and v["position"] in tree_reject(nodes, allocation_levels(nodes, c, lam)))
                for nodes in families
            )
            if score > best_score:
                best, best_score = (c, lam), score
    return best, best_score


def _self_check():
    # Alpha-investing by hand: root p=0.001 at level 0.025/1.025 rejects, W = 0.05-0.025+0.075 = 0.1;
    # child p=0.04 at level 0.05/1.05 = 0.0476 rejects; second child p=0.2 does not.
    nodes = [
        dict(position=0, depth=0, parent=None, p_value=0.001),
        dict(position=1, depth=1, parent=0, p_value=0.04),
        dict(position=2, depth=1, parent=0, p_value=0.2),
    ]
    rejected, levels = alpha_investing(nodes)
    assert rejected == {0, 1}
    assert np.isclose(levels[0], 0.025 / 1.025) and np.isclose(levels[1], 0.05 / 1.05)
    assert np.isclose(levels[2], 0.075 / 1.075)  # W = 0.1 - 0.05 + 0.1 = 0.15, phi = 0.075
    # Allocation sums stay <= alpha on random binary trees for any grid point.
    rng = np.random.default_rng(0)
    for _ in range(200):
        n = int(rng.integers(4, 60))
        groups = [[i] for i in range(n)]
        signature = []
        while len(groups) > 1:
            a, b = sorted(rng.choice(len(groups), 2, replace=False))
            signature.append((groups[a], groups[b]))
            groups = [g for i, g in enumerate(groups) if i not in (a, b)] + [groups[a] + groups[b]]
        family, frontier = [], [(groups[0], 0, None)]
        children = {tuple(sorted(l + r)): (l, r) for l, r in signature}
        while frontier:
            rows, depth, parent = frontier.pop(0)
            if len(rows) < 2 or depth > 2:
                continue
            family.append(dict(position=len(family), depth=depth, parent=parent, rows=len(rows), n=n))
            frontier += [(child, depth + 1, len(family) - 1) for child in children[tuple(sorted(rows))]]
        c = rng.dirichlet(np.ones(3))
        lam = (1.0, *rng.uniform(size=2))
        assert sum(allocation_levels(family, c, lam).values()) <= ALPHA + 1e-12
    assert len(simplex_grid()) == 231


_self_check()
